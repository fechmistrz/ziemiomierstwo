\pol{\section{Okrąg Spiekera}}
\ita{\section{Cerchio di Spieker}}

\begin{proposition}[\pol{punkt i okrąg Spiekera}\ita{punto e cerchio di Spieker}]
\label{punkt_spiekera}%
    \pol{Niech $\triangle ABC$ będzie trójkątem.}
    \ita{Sia $\triangle ABC$ un triangolo.}
    \pol{Środek okręgu wpisanego w trójkąt, którego wierzchołkami są środki boków trójkąta $ABC$, nazywamy jego punktem Spiekera.}
    \ita{Il centro della circonferenza inscritta nel triangolo avente per vertici i punti medi dei lati del triangolo $ABC$ è detto punto di Spieker.}
\end{proposition}

\pol{O punkcie Spiekera pisze też Zetel \cite[s. 23]{zetel_2020}, chociaż nie nazywa go tak, być może dlatego, że Theodor Spieker będzie dziewiętnastowiecznym geometrą z Niemiec (Poczdamu)?}
\ita{Anche Zetel \cite[p.~23]{zetel_2020} scrive del punto di Spieker, pur senza chiamarlo così, forse perché Theodor Spieker fu un geometra tedesco del XIX secolo (di Potsdam)?}
