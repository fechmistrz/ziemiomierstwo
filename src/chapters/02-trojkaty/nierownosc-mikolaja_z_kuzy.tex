\pol{\subsection{Nierówność Mikołaja z Kuzy}}
\ita{\subsection{Disuguaglianza di Niccolò Cusano}}

\pol{\index{nierówność!Mikołaja z Kuzy}}
\ita{\index{disuguaglianza!di Niccolò Cusano}}

\pol{Mikołaj z Kuzy odkryje wzmocnienie nierówności $\frac{\sin x} x < 1$ na podstawie rozważań geometrycznych, tak jak opisze to Guzicki \cite[s. 390]{guzicki_2021}:}
\ita{Niccolò Cusano scoprì un rafforzamento della disuguaglianza $\frac{\sin x} x < 1$ sulla base di considerazioni geometriche, come descritto da Guzicki \cite[p. 390]{guzicki_2021}:}

\begin{proposition}[\pol{nierówność Mikołaja z Kuzy}\ita{disuguaglianza di Niccolò Cusano}]
    \pol{Jeśli $x \in (0, \pi/2)$, to}
    \ita{Se $x \in (0, \pi/2)$, allora}
\begin{equation}
    \label{nierownosc_mikolaja_z_kuzy}
    \frac{\sin x}{x} < \frac{2 + \cos x}{3}.
\end{equation}
\end{proposition}
