\pol{\subsection{Nierówność Leibniza}}
\ita{\subsection{Disuguaglianza di Leibniz}}

\pol{U Zetela \cite[s. 72]{zetel_2020} znajdziemy wniosek z twierdzenia Stewarta (\ref{twierdzenie_stewarta}):}
\ita{In Zetel \cite[p. 72]{zetel_2020} troviamo una conseguenza del teorema di Stewart (\ref{twierdzenie_stewarta}):}

\begin{proposition}
    \pol{Odległość między środkiem ciężkości i środkiem okręgu opisanego na trójkącie wynosi}
    \ita{La distanza tra il baricentro e il circocentro di un triangolo è}
    \begin{equation}
        \sqrt{R^2 - \frac 19 \left(a^2 + b^2 + c^2\right)},
    \end{equation}
\end{proposition}

\pol{Wynika z niego:}
\ita{Ne consegue:}

\begin{proposition}[\pol{nierówność Leibniza}\ita{disuguaglianza di Leibniz}]
    \pol{Niech...}
    \ita{Sia...}
    \begin{equation}
        a^2 + b^2 + c^2 \le 9R^2,
    \end{equation}
    \pol{z równością dla trójkąta równobocznego.}
    \ita{con uguaglianza nel caso del triangolo equilatero.}
    \pol{\index{nierówność!Leibniza}}
    \ita{\index{disuguaglianza!di Leibniz}}
\end{proposition}

\pol{Guzicki \cite[s. 362, 363, 371-375]{guzicki_2021} dowodzi tego trygonometrycznie:}
\ita{Guzicki \cite[p. 362, 363, 371-375]{guzicki_2021} dimostra questo risultato mediante la trigonometria:}
\begin{equation}
    \sin^2 \alpha + \sin^2 \beta + \sin^2 \gamma = 2 (1 + \cos \alpha \cos \beta \cos \gamma) \le \frac 94.
\end{equation}
