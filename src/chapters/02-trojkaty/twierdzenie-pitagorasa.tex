\pol{Do sformułowania twierdzenia Pitagorasa potrzebne jest pojęcie pola, którego z pewnych powodów nigdzie formalnie nie opisaliśmy.}
\ita{Per formulare il teorema di Pitagora è necessario il concetto di area, che per varie ragioni non abbiamo mai definito formalmente.}
\pol{Wystarczy nam wiedzieć, że dla każdej odpowiednio regularnej figury na płaszczyźnie $F$ istnieje liczba rzeczywista $S_F \ge 0$ (zwana polem) taka, że:}
\ita{Ci basterà sapere che per ogni figura sufficientemente regolare del piano $F$ esiste un numero reale $S_F \ge 0$ (detto area) tale che:}
\begin{itemize}
    \item \pol{pole prostokąta o bokach długości $a$ oraz $b$ to $ab$;}
          \ita{l'area di un rettangolo di lati $a$ e $b$ è uguale ad $ab$;}
    \item \pol{pole sumy (,,unii'') dwóch rozłącznych figur $F_1 \cup F_2$ jest sumą ich pól;}
          \ita{l'area dell'unione di due figure disgiunte $F_1 \cup F_2$ è la somma delle loro aree;}
    \item \pol{pola przystających trójkątów są sobie równe.}
          \ita{i triangoli congruenti hanno la stessa area.}
\end{itemize}

\pol{(Można poczytać o tym u Neugebauera \cite[s. 37]{neugebauer_2018}).}
\ita{(Si può leggere qualcosa in proposito in Neugebauer \cite[p. 37]{neugebauer_2018}).}

\begin{theorem}[\pol{Pitagorasa}\ita{di Pitagora}]
\label{theorem_pythagorean}%
\pol{\index{twierdzenie!Pitagorasa}}
\ita{\index{teorema!di Pitagora}}%
    \pol{Dany jest trójkąt prostokątny $\triangle ABC$, w którym kąt przy wierzchołku $C$ jest prosty.}
    \ita{Sia dato un triangolo rettangolo $\triangle ABC$, nel quale l'angolo al vertice $C$ è retto.}
    \pol{Na jego bokach zbudowano trzy kwadraty.}
    \ita{Sui suoi lati sono costruiti tre quadrati.}
    \begin{center}
\begin{comment}
...
\end{comment}
    \end{center}
    \pol{Suma pól jasnych kwadratów jest równa polu ciemnego kwadratu:}
    \ita{La somma delle aree dei quadrati chiari è uguale all'area del quadrato scuro:}
    \begin{equation}
        |AC|^2 + |BC|^2 = |AB|^2.
    \end{equation}
    \pol{Odwrotnie, jeśli $ABC$ jest trójkątem takim, że $|AC|^2 + |BC|^2 = |AB|^2$, to trójkąt ten jest prostokątny, zaś kąt przy wierzchołku $C$ jest prosty.}
    \ita{Viceversa, se $ABC$ è un triangolo tale che $|AC|^2 + |BC|^2 = |AB|^2$, allora il triangolo è rettangolo e l'angolo al vertice $C$ è retto.}
\end{theorem}

\ita{Forse il teorema di Pitagora possiede più dimostrazioni conosciute di qualsiasi altro teorema (a eccezione della legge di reciprocità quadratica).}
\pol{Będzie ich tak bardzo bez liku, że nie wiadomo, ile dokładnie.}
\ita{Saranno così numerose da rendere impossibile stabilirne il numero esatto.}

\ita{Alcune si basano sul taglio di una figura in pezzi, sul loro riordinamento e sulla costruzione di una nuova figura.}
\pol{Inne korzystają z podobieństwa trójkątów.}
\ita{Altre utilizzano la similitudine dei triangoli.}

\begin{proof}
    \pol{Niech $\triangle ABC$ będzie trójkątem prostokątnym, z kątem prostym przy wierzchołku $C$.}
    \ita{Sia $\triangle ABC$ un triangolo rettangolo con angolo retto nel vertice $C$.}
    \pol{Na bokach $BC$, $AB$, $CA$ kreślimy kolejno kwadraty $BCDE$, $ABFG$, $ACHI$ (konstrukcja kwadratu Euklidesa korzysta z postulatu równoległości).}
    \ita{Sui lati $BC$, $AB$, $CA$ costruiamo rispettivamente i quadrati $BCDE$, $ABFG$, $ACHI$ (la costruzione euclidea del quadrato utilizza il postulato delle parallele).}

    \begin{center}
\begin{comment}
...
\end{comment}
    \end{center}

    \pol{Z punktu $C$ opuszczamy wysokość na przeciwprostokątną $AB$ i przedłużamy tak, by przecięła kwadrat $ABFG$ w punktach $K$ i $L$.}
    \ita{Dal punto $C$ tracciamo l'altezza all'ipotenusa $AB$ e la prolunghiamo in modo che intersechi il quadrato $ABFG$ nei punti $K$ e $L$.}
    \pol{Łączymy punkty $B$ i $I$ oraz $C$ i $G$.}
    \ita{Congiungiamo i punti $B$ e $I$, nonché $C$ e $G$.}
    \pol{Otrzymane trójkąty $\triangle BAI$ oraz $\triangle GAC$ są przystające na mocy cechy bok-kąt-bok.}
    \ita{I triangoli ottenuti $\triangle BAI$ e $\triangle GAC$ sono congruenti per il criterio lato-angolo-lato.}
    \pol{Niebieski prostokąt $AGLK$ (odpowiednio: kwadrat $ACHI$) ma dwukrotnie większe pole niż trójkąt $\triangle GAC$ (trójkąt $\triangle BAI$).}
    \ita{Il rettangolo blu $AGLK$ (rispettivamente il quadrato $ACHI$) ha area doppia rispetto al triangolo $\triangle GAC$ (rispettivamente al triangolo $\triangle BAI$).}
    \pol{(To jest zamaskowana zasada Cavalieriego!).}
    \ita{(Questo è un principio di Cavalieri mascherato!).}
    \pol{\index{zasada Cavalieriego}}
    \ita{\index{principio di Cavalieri}}%
    \pol{Zatem prostokąt $AGLK$ i kwadrat $ACHI$ mają równe pola.}
    \ita{Pertanto il rettangolo $AGLK$ e il quadrato $ACHI$ hanno la stessa area.}

    \pol{Analogicznie pokazujemy, że czerwony prostokąt $BFLK$ i kwadrat $BCDE$ mają równe pola.}
    \ita{In modo analogo si dimostra che il rettangolo rosso $BFLK$ e il quadrato $BCDE$ hanno la stessa area.}
    \pol{Dodajemy dwie równości stronami i otrzymujemy, że suma pól kwadratów $ACIH$ oraz $BCDE$ jest równa polu kwadratu $ABFG$.}
    \ita{Sommando membro a membro le due uguaglianze otteniamo che la somma delle aree dei quadrati $ACIH$ e $BCDE$ è uguale all'area del quadrato $ABFG$.}
\end{proof}

\begin{corollary}
    \pol{Długość przekątnej prostokąta o bokach długości $a$ i $b$ wynosi $c = \sqrt{a^2 + b^2}$.}
    \ita{La lunghezza della diagonale di un rettangolo di lati $a$ e $b$ è $c = \sqrt{a^2+b^2}$.}
\end{corollary}

\pol{Kto umie rozwijać funkcje w szeregi potęgowe może wpaść na pomysł, by zapisać}
\ita{Chi sa sviluppare le funzioni in serie di potenze potrebbe scrivere}
\begin{equation}
    c = a + \frac{b^2}{2a} - \frac{b^4}{8a^3} + \frac{b^6}{16a^5} - \frac{5b^8}{127a^7} + \ldots,
\end{equation}
\pol{co po obcięciu do pierwszych dwóch wyrazów daje przybliżenie znane Babilończykom (\cite[s. 6]{eves1_1972}):}
\ita{che, troncata ai primi due termini, fornisce l'approssimazione già nota ai Babilonesi (\cite[p. 6]{eves1_1972}):}
\begin{equation}
    c \approx a + \frac{b^2}{2a}.
\end{equation}

\pol{Metoda ta daje największy błąd względny, gdy $a = b$ i wynosi on $6.06601\ldots \%$.}
\ita{Questo metodo produce il massimo errore relativo quando $a=b$, e tale errore è pari a $6.06601\ldots \%$.}

\begin{problem}
    \pol{Miasto jest otoczone okrągłym murem o nieznanej średnicy, z czterema bramami.}
    \ita{Una città è circondata da una cinta muraria circolare di diametro sconosciuto, con quattro porte.}
    \pol{Drzewo leży trzy nible na północ od północnej bramy.}
    \ita{Un albero si trova tre nibli a nord della porta settentrionale.}
    \pol{Jeśli idziemy dziewięć nibli na wschód od południowej bramy, drzewo staje się widoczne.}
    \ita{Se ci spostiamo di nove nibli a est della porta meridionale, l'albero diventa visibile.}
    \pol{Znajdź średnicę miasta.}
    \ita{Determinare il diametro della città.}
\end{problem}

\pol{Problem pochodzi z chińskiego tekstu \emph{Shu shu jiu zhang} (,,Traktat matematyczny w dziewięciu sekcjach'') z XIII wieku.}
\ita{Il problema proviene dal testo cinese \emph{Shu shu jiu zhang} (``Trattato matematico in nove sezioni'') del XIII secolo.}
\pol{Jeżeli oznaczymy średnicę miasta przez $x$, odległość drzewa od miasta przez $b$, odległość do przejścia przez $a$, zaś kąt środek miasta -- punkt widoczności -- południowa brama przez $\alpha$, to dostaniemy równanie}
\ita{Se indichiamo con $x$ il diametro della città, con $b$ la distanza dell'albero dalla città, con $a$ la distanza percorsa e con $\alpha$ l'angolo formato dal centro della città, dal punto di osservazione e dalla porta meridionale, otteniamo l'equazione}
\begin{equation}
    \tan 2 \alpha = 2 \tan \alpha + \frac ba,
\end{equation}
\pol{co po prostych przekształceniach i przyjęciu $y = \tan \alpha$ daje}
\ita{che, dopo semplici trasformazioni e ponendo $y=\tan\alpha$, diventa}
$
    2ay^3 + by^2 - b = 0
$,
\pol{albo (pamiętając, że $y = x/2a$)}
\ita{oppure (ricordando che $y=x/2a$)}
$
    x^3 + bx^2 - 4a^2 b = 0
$.
\pol{To równanie ma jedno rozwiązanie rzeczywiste, $x = 9$.}
\ita{Questa equazione possiede un'unica soluzione reale, $x=9$.}

\pol{Względnie pierwsze liczby naturalne $a, b, c$ takie, że $a^2 + b^2 = c^2$ nazywamy (pierwotną) trójką pitagorejską.}
\ita{Chiamiamo terna pitagorica (primitiva) tre numeri naturali coprimi $a,b,c$ tali che $a^2+b^2=c^2$.}
\pol{\index{trójka pitagorejska}}
\ita{\index{terna pitagorica}}%
\pol{Każdą taką trójkę można otrzymać biorąc względnie pierwsze liczby $m, n$ różnej parzystości takie, że $m > n$ i kładąc $a = m^2 - n^2$, $b = 2 mn$, $c = m^2 + n^2$.}
\ita{Ogni tale terna può essere ottenuta scegliendo numeri coprimi $m$, $n$ di parità diversa con $m>n$ e ponendo $a=m^2-n^2$, $b=2mn$, $c=m^2+n^2$.}
\pol{Najmniejszą taką trójką jest $(3, 4, 5)$; inna legenda (nie było w niej ani smoków, ani Hippazosa) głosi, że Egipcjanie używali tego trójkąta do wyznaczania kątów prostych u podstawy piramid.}

\pol{Twierdzenie Pitagorasa znajduje zastosowanie także przy wyznaczaniu niektórych miejsc geometrycznych.}
\ita{Il teorema di Pitagora trova applicazione anche nella determinazione di alcuni luoghi geometrici.}

\begin{proposition}
    \pol{Dane są dwa różne punkty $A$ i $B$ na płaszczyźnie oraz liczba rzeczywista $c$ taka, że $2c > |AB|^2$.}
    \ita{Siano dati due punti distinti $A$ e $B$ del piano e un numero reale $c$ tale che $2c>|AB|^2$.}
    \pol{Miejscem geometrycznym punktów $P$ o własności $|AP|^2 + |BP|^2 = c$ jest okrąg o środku w środku odcinka $AB$ i średnicy $r = \sqrt{2c - |AB|^2}$.}
    \ita{Il luogo geometrico dei punti $P$ che soddisfano $|AP|^2+|BP|^2=c$ è una circonferenza avente centro nel punto medio di $AB$ e diametro $r=\sqrt{2c-|AB|^2}$.}
\end{proposition}

\begin{proposition}
    \pol{Dane są dwa różne punkty $A$ i $B$ na płaszczyźnie oraz liczba rzeczywista $c$.}
    \ita{Siano dati due punti distinti $A$ e $B$ del piano e un numero reale $c$.}
    \pol{Wtedy miejscem geometrycznym punktów $P$ o własności $|AP|^2 - |BP|^2 = c$ jest prosta prostopadła do prostej $AB$.}
    \ita{Allora il luogo geometrico dei punti $P$ tali che $|AP|^2-|BP|^2=c$ è una retta perpendicolare alla retta $AB$.}
\end{proposition}
