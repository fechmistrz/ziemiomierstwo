\pol{\subsection{Nierówność Eulera}}
\ita{\subsection{Disuguaglianza di Eulero}}

\begin{proposition}[\pol{nierówność Eulera}\ita{disuguaglianza di Eulero}]
\pol{\index{nierówność!Eulera}}%
\ita{\index{disuguaglianza!di Eulero}}%
	\pol{Niech $R, r, d$ oznaczają kolejno promienie okręgu opisanego i wpisanego w~trójkąt oraz odległość między środkami tych okręgów.}
	\ita{Siano $R, r, d$ rispettivamente i raggi della circonferenza circoscritta e inscritta a un triangolo e la distanza tra i centri di tali circonferenze.}
	\pol{Wtedy}
	\ita{Allora}
	\begin{equation}
	\frac{1}{R-d} + \frac{1}{R+d} = \frac 1 r,
	\end{equation}
	\pol{lub równoważnie $d^2 = R \cdot (R - 2r)$.}
	\ita{oppure, equivalentemente, $d^2 = R \cdot (R - 2r)$.}
	\pol{Wynika stąd, że $R \ge 2r$.}
	\ita{Ne segue che $R \ge 2r$.}
\end{proposition}

\pol{Leonhard Euler opublikuje ten wynik w 1765 roku, chociaż wcześniej zrobi to William Chapple, w pracy \emph{,,An essay on the properties of triangles inscribed in and circumscribed about two given circles''} z 1746 roku, na dole strony 123.}
\ita{Leonhard Euler pubblicò questo risultato nel 1765, sebbene William Chapple lo avesse già pubblicato in precedenza, nel lavoro \emph{``An essay on the properties of triangles inscribed in and circumscribed about two given circles''} del 1746, in fondo a pagina 123.}

\begin{proposition}[\pol{nierówność Grace'a-Danielssona}\ita{disuguaglianza di Grace-Danielsson}]
\pol{\index{nierówność!Grace'a-Danielssona}}%
\ita{\index{disuguaglianza!di Grace-Danielsson}}%
	\pol{Dla każdej pary sfer o promieniach $r, R$ takich, że $r < R$ istnieje czworościan zawarty w dużej sferze i zawierający małą sferę wtedy i tylko wtedy, gdy}
	\ita{Per ogni coppia di sfere di raggi $r, R$ tali che $r < R$ esiste un tetraedro inscritto nella sfera maggiore e circoscritto alla sfera minore se e solo se}
	\begin{equation}d^2 \le (R+r)(R - 3r).\end{equation}
\end{proposition}
