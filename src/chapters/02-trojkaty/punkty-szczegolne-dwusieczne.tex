\pol{\subsection{Okrąg wpisany i trzy dwusieczne}}
\ita{\subsection{Cerchio inscritto e tre bisettrici}}

\begin{proposition}
    \pol{Trzy dwusieczne kątów trójkąta przecinają się w jednym punkcie, środku okręgu wpisanego w ten trójkąt.}
    \ita{Le tre bisettrici degli angoli di un triangolo si incontrano in un unico punto, il centro del cerchio inscritto nel triangolo.}
\end{proposition}

\pol{Uzasadnienie jest kompletnie analogiczne do istnienia środku okręgu opisanego (tak zrobi Neugebauer \cite[s. 32]{neugebauer_2018}), ale można też podać fikuśny dowód dookoła: z twierdzenia o dwusiecznej i Cevy, jak Zetel \cite[s. 14]{zetel_2020}.}
\ita{La dimostrazione è del tutto analoga a quella dell'esistenza del centro del cerchio circoscritto (come fa Neugebauer \cite[p. 32]{neugebauer_2018}), ma si può anche fornire una dimostrazione più elegante: mediante il teorema della bisettrice e il teorema di Ceva, come in Zetel \cite[p. 14]{zetel_2020}.}
\pol{Hartshorne \cite[s. 16]{hartshorne2000} podaje to w formie ćwiczenia ze wskazówką, by spojrzeć na (IV.4), z czym totalnie się zgadzamy.}
\ita{Hartshorne \cite[p. 16]{hartshorne2000} lo presenta come esercizio con il suggerimento di guardare (IV.4), e siamo pienamente d'accordo.}

\pol{Po angielsku mamy krótkie określenia \emph{excenter, excircle, incenter, incircle}.}
\ita{In inglese si usano i termini brevi \emph{excenter, excircle, incenter, incircle}.}

\begin{proposition}
    \pol{Niech $r$ i $R$ będą promieniami okręgów wpisanego i opisanego na trójkącie o kątach $\alpha, \beta, \gamma$.}
    \ita{Siano $r$ e $R$ i raggi del cerchio inscritto e circoscritto di un triangolo con angoli $\alpha, \beta, \gamma$.}
    \pol{Wtedy}
    \ita{Allora}
    \begin{equation}
        \frac r R = 4 \sin \frac \alpha 2 \sin \frac \beta 2 \sin \frac \gamma 2.
    \end{equation}
\end{proposition}

\pol{Trygonometryczne rozwiązanie opisze Guzicki \cite[s. 351]{guzicki_2021}.}
\ita{Una dimostrazione trigonometrica è presentata da Guzicki \cite[p. 351]{guzicki_2021}.}
