\pol{\subsection{Największy kwadrat w trójkącie}}
\ita{\subsection{Il più grande quadrato in un triangolo}}

\pol{Wewnątrz każdego trójkąta znajduje się co najmniej jeden kwadrat o maksymalnym polu.}
\ita{All'interno di ogni triangolo esiste almeno un quadrato di area massima.}
\pol{Łatwo widać, że jego bok musi leżeć na którymś z boków trójkąta.}
\ita{È facile vedere che uno dei suoi lati deve giacere su un lato del triangolo.}
\pol{(Skonstruowanie go to zadanie \ref{konstrukcja_kwadratu_wpisanego}).}
\ita{(La sua costruzione è l'esercizio \ref{konstrukcja_kwadratu_wpisanego}).}
\pol{W~trójkącie równobocznym są nawet trzy takie kwadraty, każdy oparty o inną podstawę.}
\ita{In un triangolo equilatero esistono addirittura tre quadrati di questo tipo, ciascuno costruito su una base diversa.}

\begin{proposition}
\label{trojkat_calabiego}%
    \pol{W trójkącie o dwóch ramionach długości $1$ oraz podstawie, której długość jest największym dodatnim pierwiastkiem równania}
    \ita{In un triangolo con due lati uguali di lunghezza $1$ e base la cui lunghezza è la maggiore radice positiva dell'equazione}
    \begin{equation}
        2x^3 - 2x^2 - 3x + 2 = 0,
    \end{equation}
    \pol{można wpisać trzy kwadraty o największym możliwym polu.}
    \ita{si possono inscrivere tre quadrati di area massima possibile.}
    $x \approx 1.55138$.
\end{proposition}

\pol{Calabi pokaże około 1997 roku, że jest to jedyny nierównoboczny trójkąt o tej własności.}
\ita{Calabi mostrerà intorno al 1997 che questo è l'unico triangolo non equilatero con tale proprietà.}
