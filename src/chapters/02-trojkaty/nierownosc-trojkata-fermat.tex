\pol{\subsection{Zadanie Fermata}}
\ita{\subsection{Problema di Fermat}}

\begin{problem}[\pol{zadanie Fermata}\ita{problema di Fermat}]
	\label{punkt_fermata}
	\pol{Dany jest trójkąt ostrokątny $ABC$.}
	\ita{Sia dato un triangolo acutangolo $ABC$.}
	\pol{Znaleźć punkt $P$ taki, by suma $|PA| + |PB| + |PC|$ była możliwie najmniejsza.}
	\ita{Determinare un punto $P$ tale che la somma $|PA| + |PB| + |PC|$ sia minima possibile.}
\pol{\index{zadanie!Fermata}}%
\ita{\index{problema!di Fermat}}%
\end{problem}

\pol{Powyższe zadanie rozwiąże Evangelista Torricelli, który dostanie je w formie wyzwania od Fermata (dlatego też punkt $P$ nazywa się czasem punktem Torricellego; robi tak Guzicki \cite[s. 224-228]{guzicki_2021} albo Zetel \cite[s. 157-164]{zetel_2020}).}
\ita{Il problema precedente fu risolto da Evangelista Torricelli, che lo ricevette come sfida da Fermat (per questo motivo il punto $P$ viene talvolta chiamato punto di Torricelli; così fanno Guzicki \cite[pp. 224-228]{guzicki_2021} e Zetel \cite[pp. 157-164]{zetel_2020}).}

\pol{Założenie ostrokątności można osłabić, wystarczy, żeby każdy kąt wewnętrzny trójkąta miał miarę mniejszą niż~$2\pi/3$.}
\ita{L'ipotesi di acutangolarità può essere indebolita: è sufficiente che ogni angolo interno del triangolo abbia ampiezza minore di~$2\pi/3$.}

\pol{Michał Kieza opisze w $\Delta_{13}^5$ przestrzenny odpowiednik punktu Fermata-Torricellego z adnotacją, że najkrótsza sieć łącząca cztery wierzchołki czworościanu często jest łamaną złożoną z pięciu odcinków:}
\ita{Michał Kieza descrive in $\Delta_{13}^5$ l'analogo spaziale del punto di Fermat--Torricelli, osservando che la rete più corta che collega i quattro vertici di un tetraedro è spesso una spezzata composta da cinque segmenti:}

\begin{proposition}
    \pol{Jeśli punkt $F$ wewnątrz czworościanu $ABCD$ minimalizuje sumę odległości od wierzchołków $|AF| + |BF| + |CF| + |DF|$, to}
    \ita{Se un punto $F$ all'interno del tetraedro $ABCD$ minimizza la somma delle distanze dai vertici $|AF| + |BF| + |CF| + |DF|$, allora}
    \begin{itemize}
    \item \pol{dwusieczne kątów płaskich $\angle AFB$ oraz $\angle CFD$ pokrywają się,}
    \ita{le bisettrici degli angoli piani $\angle AFB$ e $\angle CFD$ coincidono,}
    \item $\angle AFB = \angle CFD$,
    \item $\cos \angle AFB + \cos \angle BFC + \cos \angle CFA = -1$.
    \end{itemize}
\end{proposition}

\begin{proposition}
    \pol{Dany jest trójkąt $ABC$, w którym żaden kąt nie ma miary większej niż~$2\pi/3$.}
    \ita{Sia dato un triangolo $ABC$ in cui nessun angolo misura più di~$2\pi/3$.}
    \pol{Wtedy wewnątrz istnieje dokładnie jeden punkt $P$ taki, że kąty $\angle APB$, $\angle BPC$, $\angle CPA$ są przystające (i miarę $2\pi/3$).}
    \ita{Allora esiste al suo interno un unico punto $P$ tale che gli angoli $\angle APB$, $\angle BPC$, $\angle CPA$ sono congruenti (e misurano $2\pi/3$).}
\end{proposition}

\pol{Punkt $P$ jest punktem Fermata i można go łatwo skontruować.}
\ita{Il punto $P$ è il punto di Fermat e può essere costruito facilmente.}
\pol{Na zewnątrz boków trójkąta $ABC$ kreślimy trzy trójkąty równoboczne $ABD$, $BCE$, $CAF$, po czym prowadzimy proste $AE$, $BF$, $CD$.}
\ita{All'esterno dei lati del triangolo $ABC$ costruiamo tre triangoli equilateri $ABD$, $BCE$, $CAF$, quindi tracciamo le rette $AE$, $BF$, $CD$.}
\pol{Punkt $P$ leży na przecięciu wszystkich trzech, co wynika z twierdzenia Jacobiego:}
\ita{Il punto $P$ è il punto d'intersezione di tutte e tre, come segue dal teorema di Jacobi:}

\begin{proposition}
    \pol{Na bokach trójkąta $ABC$, po jego zewnętrznej stronie, zbudowano takie trójkąty $ABD$, $BCE$, $CAF$, że kąty: $\angle BAD = \angle CAF$, $\angle ABD = \angle CBE$, $\angle BCE = ACF$ są równe.}
    \ita{Sui lati del triangolo $ABC$, dal lato esterno, sono costruiti triangoli $ABD$, $BCE$, $CAF$ tali che gli angoli $\angle BAD = \angle CAF$, $\angle ABD = \angle CBE$, $\angle BCE = \angle ACF$ siano uguali.}
    \pol{Wówczas proste $AE$, $BF$, $CD$ przecinają się w jednym punkcie.}
    \ita{Allora le rette $AE$, $BF$, $CD$ passano per uno stesso punto.}
\end{proposition}
