\pol{\section{Podobieństwo trójkątów}}
\ita{\section{Similitudine dei triangoli}}

\pol{O podobieństwie trójkątów piszą Bogdańska, Neugebauer \cite[s. 57-62]{neugebauer_2018}, Hartshorne \cite[s. 175-181]{hartshorne2000} w kontekście płaszczyzny Hilberta spełniającej aksjomat (P).}
\ita{Della similitudine dei triangoli scrivono Bogdańska e Neugebauer \cite[p. 57-62]{neugebauer_2018}, nonché Hartshorne \cite[p. 175-181]{hartshorne2000} nel contesto del piano di Hilbert che soddisfa l'assioma (P).}

\begin{definition}
    \pol{Dwa trójkąty nazywamy podobnymi...}
    \ita{Due triangoli si dicono simili...}
    \pol{Liczbę $\lambda$... nazywamy skalą podobieństwa.}
    \ita{Il numero $\lambda$... si chiama rapporto di similitudine.}
    \pol{\index{podobieństwo}}
    \ita{\index{similitudine}}
\end{definition}

\begin{proposition}[\pol{cecha podobieństwa BKB}\ita{criterio di similitudine LAL}]
    \pol{Jeśli dla danych trójkątów...}
    \ita{Se per i triangoli dati...}
    \pol{\index{cecha podobieństwa!bok-kąt-bok}}
    \ita{\index{criterio di similitudine!lato-angolo-lato}}
\end{proposition}

\begin{proposition}[\pol{cecha podobieństwa BBB}\ita{criterio di similitudine LLL}]
    \pol{Jeśli dla danych trójkątów...}
    \ita{Se per i triangoli dati...}
    \pol{\index{cecha podobieństwa!bok-bok-bok}}
    \ita{\index{criterio di similitudine!lato-lato-lato}}
\end{proposition}
