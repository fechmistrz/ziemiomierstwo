\begin{theorem}[\pol{nierówność Erdősa-Mordella}\ita{disuguaglianza di Erdős-Mordell}]
    \pol{Niech $P$ będzie punktem wewnątrz trójkąta $\triangle ABC$, zaś $A_p, B_p, C_p$ spodkami punktu $P$ na boki trójkąta jak na rysunku \ref{erdos_mordell_barrowa}.}
    \ita{Sia $P$ un punto interno al triangolo $\triangle ABC$, e siano $A_p, B_p, C_p$ le proiezioni ortogonali del punto $P$ sui lati del triangolo, come nella figura \ref{erdos_mordell_barrowa}.}
    \pol{Wtedy}
    \ita{Allora}
    \begin{equation}
        |PA| + |PB| + |PC| \ge 2 (|PA_p| + |PB_p| + |PC_p|).
    \end{equation}
\end{theorem}

\pol{W geometrii absolutnej nierówność Erdősa-Mordella jest równoważna temu, że suma kątów trójkąta nie przekracza $\pi$ (patrz Pambuccian \cite{pambuccian_2008}).}

\begin{theorem}[\pol{nierówność Barrowa}\ita{disuguaglianza di Barrow}]
    \pol{Niech $P$ będzie punktem wewnątrz trójkąta $\triangle ABC$, zaś $A_p$, $B_p$, $C_p$ punktami przecięć dwusiecznych trzech kątów wyznaczanych przez $P$ i pary wierzchołków trójkąta; tak jak na rysunku \ref{erdos_mordell_barrowb}.}
    \ita{Sia $P$ un punto interno al triangolo $\triangle ABC$, e siano $A_p$, $B_p$, $C_p$ i punti d'intersezione delle bisettrici dei tre angoli determinati da $P$ e dalle coppie di vertici del triangolo, come nella figura \ref{erdos_mordell_barrowb}.}
    \pol{Wtedy}
    \ita{Allora}
    \begin{equation}
        |PA| + |PB| + |PC| \ge 2 (|PA_p| + |PB_p| + |PC_p|).
    \end{equation}
\end{theorem}

\ita{La dimostrazione di Barrow e Mordell \cite{mordell_barrow_1937} sarà pubblicata nel 1937, ma il nome \emph{disuguaglianza di Barrow} si affermerà solo più tardi; certamente nel 1961 Alexander Oppenheim lo utilizzava già.}
