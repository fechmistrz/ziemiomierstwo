\pol{\subsection{Kwadraty zbudowane na bokach wielokąta}}
\ita{\subsection{Quadrati costruiti sui lati di un poligono}}

\begin{proposition}[\pol{twierdzenie Napoleona}\ita{teorema di Napoleone}]
    \pol{Środki trójkątów równobocznych zbudowanych na zewnątrz boków dowolnego trójkąta są wierzchołkami trójkąta równobocznego.}
    \ita{I centri dei triangoli equilateri costruiti esternamente sui lati di un qualunque triangolo sono i vertici di un triangolo equilatero.}
\end{proposition}

\pol{Napisze o tym Guzicki \cite[s. 307]{guzicki_2021}, Pompe \cite[s. 29]{pompe_2022} (,,słynne twierdzenie Napoleona'').}
\ita{Ne scrivono Guzicki \cite[p.~307]{guzicki_2021} e Pompe \cite[p.~29]{pompe_2022} (``il celebre teorema di Napoleone'').}
\index[persons]{Bonaparte, Napoleon}%
\pol{Chociaż twierdzenie przypisuje się Napoleonowi Bonapartemu, prawdziwym odkrywcą mógł być Lorenzo Mascheroni albo William Rutherford, matematyk angielski, który zasłynął z obliczenia 208 cyfr $\pi$ w~1841 roku (tylko 152 cyfry były poprawne).}
\ita{Sebbene il teorema venga attribuito a Napoleone Bonaparte, il vero scopritore potrebbe essere stato Lorenzo Mascheroni oppure William Rutherford, matematico inglese noto per aver calcolato nel 1841 ben 208 cifre di $\pi$ (delle quali soltanto 152 erano corrette).}
\index[persons]{Mascheroni, Lorenzo}%
\index[persons]{Rutherford, William}%

\pol{Twierdzenie pojawi się wielokrotnie w Delcie: między innymi jako ćwiczenie od Jaszuńskiej ($\Delta_{09}^5$ z~liczbami zespolonymi, $\Delta_{15}^{12}$ z obrotami) albo ćwiczenie od Bzdęgi ($\Delta_{19}^9$).}
\ita{Il teorema comparirà più volte su \emph{Delta}: per esempio come esercizio proposto da Jaszuńska ($\Delta_{09}^5$ con i numeri complessi, $\Delta_{15}^{12}$ con le rotazioni) oppure come esercizio proposto da Bzdęga ($\Delta_{19}^9$).}

\pol{Victor Thébault zauważy w 1937, że środki kwadratów zbudowanych na zewnątrz boków danego równoległoboku są wierzchołkami kwadratu (i że tylko równoległobok ma tę własność).}
\ita{Nel 1937 Victor Thébault osservò che i centri dei quadrati costruiti esternamente sui lati di un dato parallelogramma sono i vertici di un quadrato (e che soltanto un parallelogramma possiede questa proprietà).}
\index[persons]{Thébault, Victor}%

\pol{Uogólnienie do dowolnych wielokątów znajdzie Adriano Barlotti w 1955 (a później znajdzie jeszcze raz na nowo Leon Gerber w 1980).}
\ita{La generalizzazione a poligoni arbitrari fu trovata da Adriano Barlotti nel 1955 (e successivamente riscoperta da Leon Gerber nel 1980).}
\index[persons]{Barlotti, Adriano}%
\index[persons]{Gerber, Leon}%

\begin{proposition}
    \pol{Środki $n$-kątów foremnych zbudowanych na zewnątrz boków $n$-kąta $F$ są wierzchołkami $n$-kąta foremnego wtedy i tylko wtedy, gdy $F$ jest afinicznym obrazem $n$-kąta foremnego.}
    \ita{I centri degli $n$-goni regolari costruiti esternamente sui lati di un $n$-gono $F$ sono i vertici di un $n$-gono regolare se e solo se $F$ è l'immagine affine di un $n$-gono regolare.}
\end{proposition}
