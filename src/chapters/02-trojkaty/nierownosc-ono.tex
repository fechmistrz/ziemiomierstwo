\pol{\subsection{Nierówność Ono}}
\ita{\subsection{Disuguaglianza di Ono}}

% https://en.wikipedia.org/wiki/Ono%27s_inequality

\pol{W 1914 roku Toda Ono wysnuje hipotezę (nieprawdziwą w ogólności, ale prawdziwą dla wszystkich trójkątów ostrokątnych):}
\ita{Nel 1914 Toda Ono proporrà la seguente congettura (falsa in generale, ma vera per tutti i triangoli acutangoli):}
\index[persons]{Ono, Toda}%

\begin{proposition}[\pol{nierówność Ono}\ita{disuguaglianza di Ono}]
    \pol{W trójkącie ostrokątnym o bokach długości $a, b, c$ oraz polu powierzchni $S$ zachodzi nierówność}
    \ita{In un triangolo acutangolo con lati di lunghezza $a$, $b$, $c$ e area $S$ vale la disuguaglianza}
    \begin{equation}
        \frac{27}{4096} \cdot (b^2 + c^2 - a^2)^2 (c^2 + a^2 - b^2)^2 (a^2 + b^2 - c^2)^2 \le S^6.
    \end{equation}
\end{proposition}

\pol{Dowód znajdzie F. Balitrand w 1916 roku.}
\ita{La dimostrazione sarà trovata da F. Balitrand nel 1916.}
\index[persons]{Balitrand, F?}%
% Ono, T. (1914). "Problem 4417". Interméd. Math. 21: 146.
% https://zbmath.org/?format=complete&q=an:46.0859.06
\pol{Nierówność nie zachodzi dla trójkąta o bokach długości $a = 2$, $b = 3$, $c = 4$, wtedy $S^2 = 135/16$.}
\ita{La disuguaglianza non vale per il triangolo di lati $a = 2$, $b = 3$, $c = 4$; in tal caso $S^2 = 135/16$.}
