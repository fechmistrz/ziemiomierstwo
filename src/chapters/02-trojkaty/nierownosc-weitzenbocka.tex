\pol{\subsection{Nierówność Weitzenböcka i jej uogólnienia}}
\ita{\subsection{Disuguaglianza di Weitzenböck e sue generalizzazioni}}

% https://en.wikipedia.org/wiki/Weitzenböck's_inequality

\pol{Roland Weitzenböck \cite{weitzenbock_1919} udowodni w 1919 roku:}
\ita{Nel 1919 Roland Weitzenböck \cite{weitzenbock_1919} dimostrerà:}

\index[persons]{Weitzenböck, Roland}%

\begin{proposition}[\pol{nierówność Weitzenböcka}\ita{disuguaglianza di Weitzenböck}]
	\pol{W trójkącie o bokach długości $a, b, c$ oraz polu powierzchni $S$ zachodzi nierówność}
    \ita{In un triangolo con lati di lunghezza $a$, $b$, $c$ e area $S$ vale la disuguaglianza}
    \begin{equation}
        a^2 + b^2 + c^2 \ge 4 \sqrt{3} S,
    \end{equation}
    \pol{z równością wtedy i tylko wtedy, gdy trójkąt jest równoboczny: $a = b = c$.}
    \ita{con uguaglianza se e solo se il triangolo è equilatero: $a = b = c$.}
	\pol{\index{nierówność!Weitzenböcka}}
    \ita{\index{disuguaglianza!di Weitzenböck}}
\end{proposition}

% https://en.wikipedia.org/wiki/Weitzenböck%27s_inequality#Further_proofs

\pol{Dowód wykorzystuje punkt Fermata, wzór Herona, nierówność między średnią arytmetyczną i~geometryczną, własności funkcji cotangens albo nierówność}
\ita{La dimostrazione utilizza il punto di Fermat, la formula di Erone, la disuguaglianza tra media aritmetica e geometrica, le proprietà della funzione cotangente oppure la disuguaglianza}

\begin{equation}
    (a^2 - b^2)^2 + (b^2 - c^2)^2 + (c^2 - a^2)^2 \ge 0,
\end{equation}

\pol{patrz na przykład do Guzickiego \cite[s. 363, 365-367]{guzicki_2021}.}
\ita{si veda ad esempio Guzicki \cite[pp.~363, 365--367]{guzicki_2021}.}

\pol{\index{punkt Fermata}}
\ita{\index{punto di Fermat}}

\pol{\index{wzór Herona}}
\ita{\index{formula!di Erone}}

\pol{Nierówność pojawi się też w niezmienionej postaci na III Międzynarodowej Olimpiadzie Matematycznej.}
\ita{La disuguaglianza comparirà inoltre, nella stessa forma, alla III Olimpiade Internazionale della Matematica.}

\pol{Są dwa uogólnienia: nierówność Hadwigera-Finslera oraz nierówność (Neuberga-)Pedoego.}
\ita{Esistono due generalizzazioni: la disuguaglianza di Hadwiger--Finsler e la disuguaglianza di Neuberg--Pedoe.}

\begin{proposition}[\pol{nierówność Hadwigera-Finslera}\ita{disuguaglianza di Hadwiger--Finsler}]
    \pol{W trójkącie o bokach długości $a, b, c$ oraz polu powierzchni $S$ zachodzi nierówność}
    \ita{In un triangolo con lati di lunghezza $a$, $b$, $c$ e area $S$ vale la disuguaglianza}
    \begin{equation}
        a^2 + b^2 + c^2 \ge (a-b)^2 + (b-c)^2 + (a-c)^2 + 4 \sqrt{3} S,
    \end{equation}
\end{proposition}

\begin{proof}
    \pol{Z twierdzenia kosinusów mamy $a^2 = b^2 + c^2 - 2bc \cos \alpha$.}
    \ita{Dal teorema del coseno otteniamo $a^2 = b^2 + c^2 - 2bc \cos \alpha$.}

    \pol{Ponieważ $S = \frac 1 2 b c \sin \alpha$, poprzednią równość możemy przepisać do}
    \ita{Poiché $S = \frac 1 2 b c \sin \alpha$, possiamo riscrivere l'uguaglianza precedente come}
    \begin{align}
        a^2 & = (b-c)^2 + 4S \cdot \frac{1 - \cos \alpha}{\sin \alpha} \\
        & = (b-c)^2 + 4S \cdot \frac{2 \sin^2 (\alpha / 2)}{2 \sin (\alpha / 2) \cos (\alpha / 2)} \\
        & = (b-c)^2 + 4S \cdot \tan \frac \alpha 2.
    \end{align}

    \pol{Po powtórzeniu tego dla pozostałych boków i dodaniu do siebie dostaniemy}
    \ita{Ripetendo lo stesso ragionamento per gli altri lati e sommando le uguaglianze otteniamo}
    \begin{equation}
        a^2 + b^2 + c^2 = (a-b)^2 + (b-c)^2 + (c-a)^2 + 4S \left(\tan \frac \alpha 2 + \tan \frac \beta 2 + \tan \frac \gamma 2\right).
    \end{equation}

    \pol{Ale połówki kątów trójkąta nie przekraczają $\pi/2$, zaś funkcja $\tan$ jest wypukła, więc nawias można ograniczyć z dołu:}
    \ita{Poiché le metà degli angoli del triangolo non superano $\pi/2$ e la funzione $\tan$ è convessa, possiamo stimare dal basso la parentesi:}
    \begin{equation}
        \tan \frac \alpha 2 + \tan \frac \beta 2 + \tan \frac \gamma 2 \ge 3 \tan \frac{\alpha + \beta + \gamma}{6} = 3\tan \frac \pi 6 = \sqrt 3,
    \end{equation}

    \pol{co kończy dowód.}
    \ita{e questo conclude la dimostrazione.}
\end{proof}

\pol{Paul Finsler oraz Hugo Hadwiger \cite{finsler_1937} udowodnią to twierdzenie w 1937 roku.}
\ita{Paul Finsler e Hugo Hadwiger \cite{finsler_1937} dimostreranno questo teorema nel 1937.}

\index[persons]{Finsler, Paul}%
\index[persons]{Hadwiger, Hugo}%

\begin{proposition}[\pol{nierówność Neuberga-Pedoego}\ita{disuguaglianza di Neuberg--Pedoe}]
    \pol{W dwóch trójkątach o bokach $a, b, c$ oraz polu powierzchni $S$ i bokach $a_*, b_*, c_*$ oraz polu powierzchni $S_*$ zachodzi nierówność}
    \ita{Per due triangoli con lati $a$, $b$, $c$ e area $S$, e lati $a_*$, $b_*$, $c_*$ e area $S_*$, vale la disuguaglianza}
    \begin{equation}
        a^2_*(b^2 + c^2 - a^2) + 
        b^2_*(a^2 + c^2 - b^2) + 
        c^2_*(a^2 + b^2 - c^2) \ge 16 SS_*
    \end{equation}
    \pol{z równością wtedy i tylko wtedy, gdy trójkąty są podobne: $a/a_* = b/b_* = c/c_*$.}
    \ita{con uguaglianza se e solo se i triangoli sono simili: $a/a_* = b/b_* = c/c_*$.}
\end{proposition}

\pol{(Aby dostać nierówność Weitzenböcka, wystarczy wziąć $a = b = c = 1$).}
\ita{(Per ottenere la disuguaglianza di Weitzenböck, basta porre $a = b = c = 1$).}

\pol{Daniel Pedoe \cite{pedoe_1941} odkryje nierówność w 1941, a później dowie się, że była znana luksemburskiemu geometrze Józefowi Neubergowi.}
\ita{Daniel Pedoe \cite{pedoe_1941} scoprirà la disuguaglianza nel 1941 e in seguito verrà a sapere che era già nota al geometra lussemburghese Joseph Neuberg.}

\index[persons]{Pedoe, Daniel}%
\index[persons]{Neuberg, Joseph}%

\pol{Ten jednak nie pokaże, że z równości dwóch strony wynika podobieństwo trójkątów.}
\ita{Quest'ultimo, tuttavia, non mostrerà che l'uguaglianza tra i due membri implica la similitudine dei triangoli.}

\pol{Jeden z dowodów wykorzystuje wzór Herona oraz nierówność Cauchy'ego-Schwarza.}
\ita{Una delle dimostrazioni utilizza la formula di Erone e la disuguaglianza di Cauchy--Schwarz.}
