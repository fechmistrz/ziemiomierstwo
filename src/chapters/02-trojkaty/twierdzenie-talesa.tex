\begin{theorem}[\pol{Talesa}\ita{di Talete}]
\pol{\index{twierdzenie!Talesa}}\ita{\index{teorema!di Talete}}%
    \pol{Jeśli ramiona kąta płaskiego przetnie się 2 równoległymi prostymi:}
    \ita{Se i lati di un angolo vengono intersecati da due rette parallele:}

    \pol{to długości odcinków wyznaczonych przez te proste na jednym z ramion kąta są proporcjonalne do długości odpowiednich odcinków na drugim ramieniu kąta, a zatem}
    \ita{allora le lunghezze dei segmenti individuati da tali rette su uno dei lati dell'angolo sono proporzionali alle lunghezze dei segmenti corrispondenti sull'altro lato dell'angolo, e quindi}
    \begin{equation}
        \label{thales_ratio}
        \frac{|OA|}{|OA'|} = \frac{|OB|}{|OB'|} = \frac{|AB|}{|A'B'|}.
    \end{equation}
\end{theorem}

\begin{proposition}[\pol{twierdzenie odwrotne do tw. Talesa}\ita{teorema inverso di Talete}]
    \pol{Jeżeli pewna prosta przecina boki $OA'$, $OB'$ trójkąta $OA'B'$ w różnych punktach $A$ i $B$ odpowiednio, a przy tym zachodzi równość \ref{thales_ratio}, to prosta ta jest równoległa do prostej $A'B'$.}
    \ita{Se una retta interseca i lati $OA'$, $OB'$ del triangolo $OA'B'$ rispettivamente nei punti distinti $A$ e $B$, e vale l'uguaglianza \ref{thales_ratio}, allora tale retta è parallela alla retta $A'B'$.}
\end{proposition}

\begin{theorem}[Varignona]
    \label{theorem_varignon}
    \pol{Czworokąt $PQRS$, którego wierzchołki leżą na środkach boków $AB$, $BC$, $CD$, $DA$ czworokąta $ABCD$, jest równoległobokiem.}
    \ita{Il quadrilatero $PQRS$, i cui vertici giacciono nei punti medi dei lati $AB$, $BC$, $CD$, $DA$ del quadrilatero $ABCD$, è un parallelogramma.}
    \pol{Jego znakowane (!) pole jest równe połowie pola czworokąta $ABCD$.}
    \ita{La sua area orientata (!) è uguale alla metà dell'area del quadrilatero $ABCD$.}
\end{theorem}

\begin{proposition}
    \pol{Następujące warunki są równoważne:}
    \ita{Le seguenti condizioni sono equivalenti:}
    \begin{itemize}
        \item \pol{przekątne czworokąta $ABCD$ są równej długości,}
              \ita{le diagonali del quadrilatero $ABCD$ hanno la stessa lunghezza,}
        \item \pol{równoległobok Varignona $PQRS$ związany z czworokątem $ABCD$ jest rombem.}
              \ita{il parallelogramma di Varignon $PQRS$ associato al quadrilatero $ABCD$ è un rombo.}
    \end{itemize}
\pol{\index{równoległobok!Varignona}}\ita{\index{parallelogramma!di Varignon}}%
\pol{\index{romb}}\ita{\index{rombo}}%
\end{proposition}

    \pol{Następujące warunki są równoważne:}
