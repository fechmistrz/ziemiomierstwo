\pol{Wnioskiem z twierdzenia o dwusiecznej jest:}
\ita{Una conseguenza del teorema della bisettrice è la seguente:}

\begin{theorem}[\pol{Steinera-Lehmusa}\ita{di Steiner-Lehmus}]
    \pol{\index{twierdzenie!Steinera-Lehmusa}}%
    \ita{\index{teorema!di Steiner-Lehmus}}%
    \label{theorem_steiner_lehmus}%
	\pol{Jeżeli dwie dwusieczne trójkąta są równej długości, to trójkąt ten jest równoramienny.}
	\ita{Se due bisettrici di un triangolo hanno la stessa lunghezza, allora il triangolo è isoscele.}
\end{theorem}

\pol{Po raz pierwszy wspomni o nim Christian Lehmus w~liście z~1840 roku do Charlesa Sturma, gdzie poprosi o~czysto geometryczny dowód.}
\ita{Il teorema fu menzionato per la prima volta da Christian Lehmus in una lettera del 1840 a Charles Sturm, nella quale chiedeva una dimostrazione puramente geometrica.}
\index[persons]{Lehmus, Christian}%
\index[persons]{Sturm, Charles}%
\pol{Sturm przekazał prośbę do innych matematyków, jedną z~pierwszych osób, która uporała się z problemem, był Jakob Steiner.}
\ita{Sturm trasmise la richiesta ad altri matematici; una delle prime persone a risolvere il problema fu Jakob Steiner.}
\index[persons]{Steiner, Jakob}%
\pol{Większość znanych uzasadnień to dowody nie wprost: jeśli trójkąt nie jest równoramienny, to ma dwusieczne różnej długości.}
\ita{La maggior parte delle dimostrazioni note procede per assurdo: se un triangolo non è isoscele, allora le sue bisettrici hanno lunghezze diverse.}

\pol{Dowód można znaleźć u Hartshorne'a \cite[s. 11]{hartshorne2000}; Pompego \cite[s. 32, 33]{pompe_2022}; Bogdańskiej, Neugebauera \cite[s. 74]{neugebauer_2018}.}
\ita{Una dimostrazione si trova in Hartshorne \cite[p. 11]{hartshorne2000}, Pompegi \cite[pp. 32, 33]{pompe_2022} e Bogdańska, Neugebauer \cite[p. 74]{neugebauer_2018}.}

\ita{Zdzisław Pogoda, in $\Delta_{11}^1$, descrive l'ingegnosa soluzione trovata da due ingegneri inglesi, G. Gilbert e D. McDonnell.}
