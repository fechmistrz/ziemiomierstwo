\begin{proposition}[\pol{nierówność trójkąta}\ita{disuguaglianza triangolare}]
\index{nierówność!trójkąta}%
\ita{\index{disuguaglianza!triangolare}}%
	\pol{Niech $ABC$ będzie trojkątem.}
	\ita{Sia $ABC$ un triangolo.}
	\pol{Wtedy suma odcinków $AB$ i $BC$ jest dłuższa niż $AC$.}
	\ita{Allora la somma delle lunghezze dei segmenti $AB$ e $BC$ è maggiore della lunghezza di $AC$.}
\end{proposition}

% TODO: https://www.deltami.edu.pl/2013/09/siatka-czworoscianu/ to samo w R3

\begin{proof}
	\pol{Wynika to ze wzoru Herona (fakt \ref{prp_heron}) i tego, że pole trójkąta jest nieujemne.}
	\ita{Segue dalla formula di Erone (fatto \ref{prp_heron}) e dal fatto che l'area di un triangolo è non negativa.}
\end{proof}

\begin{corollary}
	\pol{Niech $a \ge b \ge c$ będą bokami trójkąta.}
	\ita{Siano $a \ge b \ge c$ i lati di un triangolo.}
	\pol{Wtedy}
	\ita{Allora}
	% \begin{equation}
	% 	1 < \frac{a + c}{b} < 3 
	% \end{equation}
	% oraz
	\begin{equation}
		1 \le \min \left(\frac ab, \frac bc\right) \le \phi = \frac {1 + \sqrt 5}{2}.
		% American Mathematical Monthly, pp. 49-50, 1954. 
	\end{equation}
\end{corollary}

\pol{Nierówność trójkąta nie jest wnioskiem z aksjomatów I1-I3, B1-B4, C1-C3, ponieważ nie zachodzi w następującym modelu (ukradniętym Hartshorne'owi \cite[s. 90]{hartshorne2000}):}
\ita{La disuguaglianza triangolare non è una conseguenza degli assiomi I1--I3, B1--B4, C1--C3, poiché non vale nel seguente modello (preso in prestito da Hartshorne \cite[p.~90]{hartshorne2000}):}

\begin{example}
	\pol{Rozpatrujemy zbiór $\mathbb R^2$ jako płaszczyznę ze standardowymi punktami oraz prostymi, ale niestandardową metryką}
	\ita{Consideriamo l'insieme $\mathbb R^2$ come piano con i punti e le rette usuali, ma dotato della seguente metrica non standard}
	\begin{equation}
		d((x_1, y_1), (x_2, y_2)) = \begin{cases}
			\sqrt{(x_1-x_2)^2 + (y_1-y_2)^2} & \text{\pol{jeśli}\ita{se }} x_1 = x_2 \vee y_1 = y_2, \\
			2 \sqrt{(x_1-x_2)^2 + (y_1-y_2)^2} & \text{\pol{w przeciwnym wypadku}\ita{altrimenti}}
		\end{cases}.
	\end{equation}
	\pol{Wtedy nierówność trójkąta nie zachodzi.}
	\ita{In tal caso la disuguaglianza triangolare non vale.}
\end{example}

\pol{Odpowiednikiem nierówności trójkąta w trzech wymiarach zdaje się być ćwiczenie od Kiezy \cite[s. 31]{kieza_2017}:}
\ita{Un analogo tridimensionale della disuguaglianza triangolare sembra essere il seguente esercizio di Kieza \cite[p.~31]{kieza_2017}:}

\begin{proposition}
	\pol{W każdym czworościanie suma pól trzech dowolnych ścian jest większa od pola czwartej ściany.}
	\ita{In ogni tetraedro la somma delle aree di tre facce qualsiasi è maggiore dell'area della quarta faccia.}
\end{proposition}
