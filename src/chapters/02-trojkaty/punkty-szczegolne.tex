\pol{\section{Punkty szczególne}}
\ita{\section{Punti notevoli}}

\pol{Z każdym trójkątem związane są pewne specjalne punkty, internetowa lista \emph{Encyclopedia of Triangle Centers} wymieni ich co najmniej 68 547.}
\ita{A ogni triangolo sono associati alcuni punti notevoli; l'elenco online \emph{Encyclopedia of Triangle Centers} ne riporta almeno 68\,547.}
\pol{My nie mamy aż tyle miejsca, więc ograniczymy się do najważniejszych:}
\ita{Non abbiamo però così tanto spazio, quindi ci limiteremo ai più importanti:}
\begin{itemize}
\item \pol{$X(1)$ to środek okręgu wpisanego,}\ita{$X(1)$ è l'incentro,}
\item \pol{$X(2)$ to środek ciężkości,}\ita{$X(2)$ è il baricentro,}
\item \pol{$X(3)$ to środek okręgu opisanego,}\ita{$X(3)$ è il circocentro,}
\item \pol{$X(4)$ to ortocentrum.}\ita{$X(4)$ è l'ortocentro.}
\end{itemize}
\pol{Te cztery punkty opisujemy poniżej.}
\ita{Questi quattro punti sono descritti qui di seguito.}

\begin{itemize}
\item \pol{$X(5)$ to środek okręgu dziewięciu punktów z faktu \ref{okrag_dziewieciu_punktow},}\ita{$X(5)$ è il centro del cerchio dei nove punti del fatto \ref{okrag_dziewieciu_punktow},}
\item \pol{$X(6)$ to punkt Lemoine'a (Grebego) z faktu \ref{punkt_lemoine},}\ita{$X(6)$ è il punto di Lemoine (Grebe) del fatto \ref{punkt_lemoine},}
\item \pol{$X(7)$ to punkt Gergonne'a z~faktu \ref{punkt_gergonne},}\ita{$X(7)$ è il punto di Gergonne del fatto \ref{punkt_gergonne},}
\item \pol{$X(8)$ to punkt Nagela z faktu \ref{punkt_nagela},}\ita{$X(8)$ è il punto di Nagel del fatto \ref{punkt_nagela},}
\item \pol{$X(9)$ to mittenpunkt (punkt Lemoine'a trójkąta rozpiętego przez środki okręgów dopisanych; leży na prostych łączących środek ciężkości z punktem Gergonne'a, środek okręgu wpisanego z punktem Lemoine'a oraz ortocentrum ze środkiem Spiekera),}\ita{$X(9)$ è il mittenpunkt (il punto di Lemoine del triangolo formato dai centri degli excerchi; appartiene alle rette che uniscono il baricentro con il punto di Gergonne, l'incentro con il punto di Lemoine e l'ortocentro con il centro di Spieker),}
\item \pol{$X(10)$ to środek okręgu Spiekera z faktu \ref{punkt_spiekera},}\ita{$X(10)$ è il centro del cerchio di Spieker del fatto \ref{punkt_spiekera},}
\item \pol{$X(11)$ to punkt Feuerbacha z twierdzenia \ref{punkt_feuerbacha},}\ita{$X(11)$ è il punto di Feuerbach del teorema \ref{punkt_feuerbacha},}
\item \pol{$X(13)$ to punkt Fermata z zadania \ref{punkt_fermata},}\ita{$X(13)$ è il punto di Fermat dell'esercizio \ref{punkt_fermata},}
\item \pol{$X(17)$, $X(18)$ to punkty Napoleona,}\ita{$X(17)$, $X(18)$ sono i punti di Napoleone,}
\item \pol{$X(39)$ to środek odcinka łączącego punkty Brocarda z definicji \ref{punkty_brocarda}.}\ita{$X(39)$ è il punto medio del segmento che unisce i punti di Brocard definiti in \ref{punkty_brocarda}.}
\end{itemize}

\ita{La lista è lunga e sembra non avere fine.}
