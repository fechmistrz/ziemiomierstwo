\begin{proposition}[\pol{nierówność Snelliusa-Huygensa}\ita{disuguaglianza di Snellius-Huygens}]
    \pol{Jeśli $x \in (0, \pi/2)$, to}
    \ita{Se $x \in (0, \pi/2)$, allora}
    \begin{equation}
        2 \sin x + \tan x > 3x.
        \label{nierownosc_snelliusa}
    \end{equation}
\end{proposition}
