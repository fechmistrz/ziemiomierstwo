\pol{\subsection{Pole trójkąta, wzór Herona}}
\ita{\subsection{Area del triangolo, formula di Erone}}

\pol{W tej podsekcji znajdziemy różne wzory na pole trójkąta.}
\ita{In questa sottosezione presenteremo diverse formule per l'area di un triangolo.}

\begin{proposition}
    \pol{Pole trójkąta o podstawie długości $a$ oraz wysokości opuszczonej na tę podstawę długości $h$ wyraża się wzorem}
    \ita{L'area di un triangolo con base di lunghezza $a$ e altezza relativa a tale base di lunghezza $h$ è data dalla formula}
    \begin{equation}
        S_\triangle = \frac 1 2 a h.
    \end{equation}
    \pol{Jeśli kąt naprzeciw podstawy ma miarę $\alpha$, a ramiona długości $b, c$, to zachodzi też}
    \ita{Se l'angolo opposto alla base misura $\alpha$ e gli altri due lati hanno lunghezze $b$ e $c$, allora vale anche}
    \begin{equation}
        S_\triangle = \frac 1 2 bc \sin \alpha.
    \end{equation}
\end{proposition}

\pol{Pisze o tym Guzicki \cite[s. 255]{guzicki_2021}.}
\ita{Se ne occupa Guzicki \cite[p.~255]{guzicki_2021}.}

\begin{proposition}[\pol{wzór Herona, 60 r.n.e.?}\ita{formula di Erone, 60 a.C.?}]
\label{prp_heron}%
    \pol{Pole trójkąta o bokach długości $a, b, c$ wyraża się wzorem}
    \ita{L'area di un triangolo con lati di lunghezza $a$, $b$, $c$ è data dalla formula}
    \begin{equation}
        S = \sqrt{p(p-a)(p-b)(p-c)},
    \end{equation}
    \pol{gdzie $p = \frac 1 2 (a + b + c)$ oznacza połowę obwodu.}
    \ita{dove $p = \frac 1 2 (a + b + c)$ denota il semiperimetro.}
\end{proposition}

\pol{Istnieje częściowe uogólnienie wzoru Herona dla czworokątów:}
\ita{Esiste una generalizzazione parziale della formula di Erone ai quadrilateri:}

\begin{proposition}[\pol{wzór Brahmagupty}\ita{formula di Brahmagupta}]
\label{brahmagupta_formula}%
\pol{\index{czworokąt!cykliczny}}%
\ita{\index{quadrilatero!ciclico}}%
\pol{\index{wzór!Brahmagupty}}%
\ita{\index{formula!di Brahmagupta}}%
    \pol{Pole czworokąta wypukłego, na którym można opisać okrąg, o bokach długości $a, b, c, d$ wyraża się wzorem}
    \ita{L'area di un quadrilatero convesso ciclico con lati di lunghezza $a$, $b$, $c$, $d$ è data dalla formula}
    \begin{equation}
        S = \sqrt{(p-a)(p-b)(p-c)(p-d)},
    \end{equation}
    \pol{gdzie $p$ oznacza połowę obwodu.}
    \ita{dove $p$ denota il semiperimetro.}
\end{proposition}

\pol{Dowód polega na zastosowaniu wzoru Herona dwa razy do trójkątów podobnych lub ponownym skorzystaniu z dobrodziejstw trygonometrii.}
\ita{La dimostrazione consiste nell'applicare due volte la formula di Erone a opportuni triangoli oppure nel ricorrere nuovamente alla trigonometria.}

\begin{proposition}[\pol{wzór Bretschneidera, 1842}\ita{formula di Bretschneider, 1842}]
\pol{\index{wzór!Bretschneidera}}%
\ita{\index{formula!di Bretschneider}}%
    \pol{Pole czworokąta o bokach długości $a, b, c, d$ i sumie miar dwóch przeciwległych kątów wewnętrznych $2 \varphi$ wyraża się wzorem}
    \ita{L'area di un quadrilatero con lati di lunghezza $a$, $b$, $c$, $d$ e somma di due angoli opposti pari a $2\varphi$ è data dalla formula}
    \begin{equation}
        S = \sqrt{(p-a)(p-b)(p-c)(p-d) - abcd \cos^2  \varphi},
    \end{equation}
    \pol{gdzie $p$ oznacza połowę obwodu.}
    \ita{dove $p$ denota il semiperimetro.}
\end{proposition}

\pol{Inny, równie przydatny wzór znajdzie Julian Coolidge \cite{coolidge_1939}:}
\ita{Julian Coolidge \cite{coolidge_1939} troverà un'altra formula altrettanto utile:}

\index[persons]{Coolidge, Julian}%

\pol{pole czworokąta o bokach długości $a, b, c, d$ i przekątnych długości $e, f$ wyraża się wzorem}
\ita{l'area di un quadrilatero con lati di lunghezza $a$, $b$, $c$, $d$ e diagonali di lunghezza $e$, $f$ è data dalla formula}

\begin{equation}
    S = \sqrt{(p-a)(p-b)(p-c)(p-d) - \frac 1 4 (ac + bd + pq)(ac + bd - pq)},
\end{equation}

\pol{gdzie $p$ oznacza raz jeszcze (i nie ostatni) połowę obwodu.}
\ita{dove $p$ denota ancora una volta (e non per l'ultima) il semiperimetro.}
