\pol{\subsection{Nierówność izoperymetryczna}}
\ita{\subsection{Disuguaglianza isoperimetrica}}

\begin{proposition}[\pol{nierówność izoperymetryczna}\ita{disuguaglianza isoperimetrica}]
    \pol{Dany jest trójkąt o obwodzie $2p$ oraz polu $S$.}
    \ita{Dato un triangolo con perimetro $2p$ e area $S$.}
    \pol{Wtedy}
    \ita{Allora}
    \begin{equation}
        S \le \frac{p^2}{3 \sqrt 3},
    \end{equation}
    \pol{zatem wśród trójkątów o ustalonym obwodzie największe pole ma trójkąt równoboczny.}
    \ita{pertanto, tra i triangoli con perimetro fissato, quello equilatero ha l'area massima.}
    \pol{\index{nierówność!izoperymetryczna}}
	\ita{\index{disuguaglianza!isoperimetrica}}
    % TODO: https://www.deltami.edu.pl/2011/04/symetryzacja-steinera/
\end{proposition}

\pol{Guzicki \cite[s. 169, 170]{guzicki_2021} wyprowadza nierówność izoperymetryczną z nierówności między średnimi oraz ze wzoru Herona, a później \cite[s. 362]{guzicki_2021} raz jeszcze, trygonometrycznie.}
\ita{Guzicki \cite[p. 169, 170]{guzicki_2021} ricava la disuguaglianza isoperimetrica dalle disuguaglianze tra medie e dalla formula di Erone, e successivamente \cite[p. 362]{guzicki_2021} di nuovo, trigonometricamente.}

\pol{Bartłomiej Bzdęga opisze w $\Delta_{23}^1$ nierówność $4 \pi S \le L^2$, gdzie $L$ oznacza obwód figury, $S$ jej pole, zaś równość ma miejsce dokładnie dla koła.}
\ita{Bartłomiej Bzdęga descriverà in $\Delta_{23}^1$ la disuguaglianza $4 \pi S \le L^2$, dove $L$ indica il perimetro della figura, $S$ la sua area, e l'uguaglianza si verifica esattamente per il cerchio.}
\pol{(Mamy też nierówność Bonnesena, która orzeka, że $\pi (R-r)^2 \le 4p^2 - 4 \pi L$).}
\ita{(Esiste anche la disuguaglianza di Bonnesen, che stabilisce che $\pi (R-r)^2 \le 4p^2 - 4 \pi L$).}
