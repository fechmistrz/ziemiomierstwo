\pol{\subsection{Prosta Wallace'a-Simsona}}
\ita{\subsection{Retta di Wallace-Simson}}

\begin{proposition}[\pol{twierdzenie Wallace'a}\ita{teorema di Wallace}]
	\pol{Dany jest trójkąt $ABC$ oraz punkt $P$.}
	\ita{Siano dati un triangolo $ABC$ e un punto $P$.}
	\pol{Niech $P_a$, $P_b$, $P_c$ oznacza rzuty punktu $P$ na proste $BC$, $AC$, $AB$.}
	\ita{Siano $P_a$, $P_b$, $P_c$ le proiezioni del punto $P$ sulle rette $BC$, $AC$, $AB$.}
	\pol{Wtedy następujące warunki są równoważne: punkt $P$ leży na okręgu opisanym na $\triangle ABC$; punkty $P_a$, $P_b$, $P_c$ są współliniowe.}
	\ita{Allora le seguenti condizioni sono equivalenti: il punto $P$ appartiene alla circonferenza circoscritta a $\triangle ABC$; i punti $P_a$, $P_b$, $P_c$ sono allineati.}
\end{proposition}

\pol{Kiedy punkt $P$ leży na wspomnianym okręgu, prostą przez punkty $P_a$, $P_b$ oraz $P_c$ nazywa się prostą Simsona; na cześć Roberta Simsona, szkockiego matematyka żyjącego w latach 1687-1768; chociaż pierwszą pracę na jej temat opublikuje Wallace w 1799 roku.}
\ita{Quando il punto $P$ appartiene alla circonferenza suddetta, la retta passante per i punti $P_a$, $P_b$ e $P_c$ è detta retta di Simson, in onore di Robert Simson, matematico scozzese vissuto tra il 1687 e il 1768, sebbene il primo lavoro sull'argomento sia stato pubblicato da Wallace nel 1799.}
\index[persons]{Wallace, ?}

\pol{Być może właściwsze byłoby pisać ,,prosta Wallace'a-Simsona''.}
\ita{Forse sarebbe più corretto parlare di ``retta di Wallace-Simson''.}

\pol{Współliniowość spodków punktu $P$ na boki trójkąta jest twierdzeniem z dowodem u Zetela \cite[s. 55]{zetel_2020}; ćwiczeniem u Audina \cite[s. 104]{audin_2003}, Hartshorne'a \cite[s. 61]{hartshorne2000} i Neugebauera \cite[s. 25]{neugebauer_2018}.}
\ita{L'allineamento dei piedi delle perpendicolari dal punto $P$ ai lati del triangolo è presentato come teorema con dimostrazione da Zetel \cite[p. 55]{zetel_2020}; come esercizio da Audin \cite[p. 104]{audin_2003}, Hartshorne \cite[p. 61]{hartshorne2000} e Neugebauer \cite[p. 25]{neugebauer_2018}.}

\begin{proposition}[\pol{twierdzenie Steinera}\ita{teorema di Steiner}]
	\pol{Dany jest trójkąt $ABC$ oraz punkt $P$ leżący na okręgu opisanym na trójkącie.}
	\ita{Siano dati un triangolo $ABC$ e un punto $P$ appartenente alla sua circonferenza circoscritta.}
	\pol{Wtedy obrazy punktu $P$ w symetrii względem boków trójkąta leżą na jednej prostej, która przechodzi przez ortocentrum trójkąta.}
	\ita{Allora le immagini del punto $P$ rispetto alle simmetrie aventi come assi i lati del triangolo giacciono su una stessa retta, che passa per l'ortocentro del triangolo.}
\end{proposition}

\ita{Dal teorema della retta di Wallace-Simson segue, come in Zetel \cite[p. 57]{zetel_2020}:}

\begin{proposition}[\pol{twierdzenie Salmona}\ita{teorema di Salmon}]
\pol{\index{twierdzenie!Salmona}}%
\ita{\index{teorema!di Salmon}}%
	\pol{Dany jest okrąg oraz trzy jego różne cięciwy $PA$, $PB$, $PC$ takie, że przekrojem okręgów na średnicach $PA$, $PB$ (odpowiednio: $PB$, $PC$ i $PA$, $PC$) są punkty $P$, $M$ (odpowiednio: $P$, $K$ oraz $P$, $L$).}
	\ita{Sono dati un cerchio e tre sue corde distinte $PA$, $PB$, $PC$ tali che l'intersezione dei cerchi aventi per diametri $PA$, $PB$ (rispettivamente $PB$, $PC$ e $PA$, $PC$) sia costituita dai punti $P$, $M$ (rispettivamente $P$, $K$ e $P$, $L$).}
	\pol{Wtedy punkty $K$, $L$, $M$ są współliniowe.}
	\ita{Allora i punti $K$, $L$, $M$ sono allineati.}
\end{proposition}
