\pol{\subsection{Zadanie Fagnano}}
\ita{\subsection{Problema di Fagnano}}

\begin{problem}[\pol{zadanie Fagnano}\ita{problema di Fagnano}]
	\pol{Dany jest trójkąt ostrokątny $ABC$.}
	\ita{Sia dato un triangolo acutangolo $ABC$.}
	\pol{Wpisać w niego trójkąt $DEF$ (czyli wybrać punkt $D$ na boku $BC$, $E$ na $AC$, $F$ na $AB$ tak, by tworzyły trójkąt) o możliwie najmniejszym obwodzie.}
	\ita{Inscrivere in esso un triangolo $DEF$ (cioè scegliere un punto $D$ sul lato $BC$, $E$ su $AC$, $F$ su $AB$ in modo che formino un triangolo) avente perimetro minimo possibile.}
\index{zadanie!Fagnano}%
\ita{\index{problema!di Fagnano}}%
\end{problem}

\pol{Coxeter \cite[s. 36, 37]{coxeter_1967} pokaże tak jak Fejer, że rozwiązaniem zadania jest trójkąt spodkowy (którego wierzchołkami są spodki wysokości danego trójkąta; zwany brzydko ortycznym).}
\ita{Coxeter \cite[pp.~36--37]{coxeter_1967} mostrerà, come Fejer, che la soluzione del problema è il triangolo pedale (i cui vertici sono i piedi delle altezze del triangolo dato; spesso chiamato triangolo ortico).}
\pol{Zetel \cite[s. 97-100]{zetel_2020} przytoczy oprócz dowodu Fejera także dowód Izwolskiego, oparty na twierdzeniu: promienie okręgu opisanego na trójkącie, przechodzące przez wierzchołki trójkąta, są prostopadłe do odpowiednich boków trójkąta spodkowego.}
\ita{Zetel \cite[pp.~97--100]{zetel_2020} presenterà, oltre alla dimostrazione di Fejer, anche quella di Izvol'skij, basata sul teorema secondo cui i raggi della circonferenza circoscritta passanti per i vertici del triangolo sono perpendicolari ai corrispondenti lati del triangolo pedale.}
\pol{Pompe \cite[s. 16-18]{pompe_2022} wykorzysta własności obrotów.}
\ita{Pompe \cite[pp.~16--18]{pompe_2022} utilizzerà le proprietà delle rotazioni.}
\pol{Audin \cite[s. 101]{audin_2003} podaje ten fakt w~formie ćwiczenia.}
\ita{Audin \cite[p.~101]{audin_2003} presenta questo fatto sotto forma di esercizio.} % todo: fagnano czy gemrat?

\pol{Zadanie traci sens dla trójkątów nieostrokątnych: wtedy dwa z trzech wierzchołków trójkąta wpisanego o najmniejszym obwodzie pokrywają się z wierzchołkiem kąta rozwartego.}
\ita{Il problema perde significato per i triangoli non acutangoli: in tal caso due dei tre vertici del triangolo inscritto di perimetro minimo coincidono con il vertice dell'angolo ottuso.}

\begin{corollary}
	\pol{Wysokości trójkąta $ABC$ są dwusiecznymi kątów jego trójkąta spodkowego.}
	\ita{Le altezze del triangolo $ABC$ sono le bisettrici degli angoli del suo triangolo pedale.} % Pompe, Wokół obrotów, wniosek 2.3
\end{corollary}

\pol{Ten wniosek pojawia się u Zetela \cite[s. 89]{zetel_2020} i Pompego \cite[s. 19]{pompe_2022}.}
\ita{Questo corollario compare in Zetel \cite[p.~89]{zetel_2020} e in Pompe \cite[p.~19]{pompe_2022}.}
\pol{Będzie również zadaniem przygotowawczym na I OM, Guzicki \cite[s. 351]{guzicki_2021} rozwiąże je trygonometrycznie.}
\ita{Sarà anche un problema preparatorio per la I Olimpiade Matematica; Guzicki \cite[p.~351]{guzicki_2021} lo risolverà con metodi trigonometrici.}

\begin{proposition}
    \pol{Boki trójkąta spodkowego są antyrównoległe do boków trójkąta danego.}
    \ita{I lati del triangolo pedale sono antiparalleli ai lati del triangolo dato.}
\end{proposition}

(\pol{Zetel} \cite[p.~89]{zetel_2020}\ita{Zetel \cite[p.~89]{zetel_2020}}).

\begin{proposition}
	\pol{Niech wysokości trójkąta ABC przecinają się w punkcie H.}
	\ita{Sia $H$ il punto d'intersezione delle altezze del triangolo $ABC$.}
	\pol{Każdy z czterech punktów A, B, C, H jest ortocentrum trójkąta, którego wierzchołkami są trzy pozostałe punkty.}
	\ita{Ciascuno dei quattro punti $A$, $B$, $C$, $H$ è l'ortocentro del triangolo avente come vertici gli altri tre punti.}
\end{proposition}

(\pol{Zetel} \cite[p.~95]{zetel_2020}\ita{Zetel \cite[p.~95]{zetel_2020}}).
