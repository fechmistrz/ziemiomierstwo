\begin{theorem}[\pol{twierdzenie Ponceleta-Steinera}\ita{teorema di Poncelet-Steiner}]
\label{thm:poncelet_steiner}%
\index{twierdzenie!Ponceleta-Steinera}%
\pol{    Jeśli dana konstrukcja jest wykonalna za pomocą cyrkla i linijki, to jest ona wykonalna za pomocą samej linijki, o ile dany jest na płaszczyźnie pewien okrąg wraz ze środkiem.}
\ita{    Se una costruzione è realizzabile con riga e compasso, allora è realizzabile con la sola riga, purché nel piano sia dato un cerchio insieme al suo centro.}
    \index{twierdzenie!Ponceleta-Steinera}%
\end{theorem}

\pol{Nazwa twierdzenia pochodzi od dwóch panów: Jeana Ponceleta, który postawi w 1822 roku hipotezę oraz Jakoba Steinera, który dostarczy brakujący dowód w 1833 roku.}
\ita{Il nome del teorema deriva da due studiosi: Jean Poncelet, che formulò la congettura nel 1822, e Jakob Steiner, che fornì la dimostrazione mancante nel 1833.}
\index[persons]{Poncelet, Jean}%
\index[persons]{Steiner, Jakob}%
\pol{Zamiast całego okręgu, wystarczy, że mamy do dyspozycji mały jego łuk -- to wzmocnienie, o które pokusi się Francesco Severi w 1904 roku.}
\ita{Al posto dell'intero cerchio, è sufficiente avere a disposizione un suo piccolo arco: questo rafforzamento fu ottenuto da Francesco Severi nel 1904.}
\index[persons]{Severi, Francesco}%

\pol{Sama linijka to za mało, bo nie pozwala na wyciąganie pierwiastków kwadratowych, tak jak opisze to Łukasz Rajkowski w $\Delta_{15}^8$.}

\index[persons]{Rajkowski, Łukasz}%
\pol{Ale zamiast środka okręgu możemy wymagać:}
\ita{Tuttavia, al posto del centro del cerchio possiamo richiedere:}

\end{itemize}
