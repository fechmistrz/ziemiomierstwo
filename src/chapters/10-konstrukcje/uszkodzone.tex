\pol{\section{Uszkodzone przyrządy}}
\ita{\section{Strumenti danneggiati}}

\pol{Problemy od \ref{broken_ruler_compass_hartshorne_start} do \ref{broken_ruler_compass_hartshorne_end} pochodzą z książki Hartshorne'a \cite[s. 25, 26]{hartshorne2000}.}
\ita{I problemi da \ref{broken_ruler_compass_hartshorne_start} a \ref{broken_ruler_compass_hartshorne_end} provengono dal libro di Hartshorne \cite[p. 25, 26]{hartshorne2000}.}

\begin{geoconstruction}[\pol{połamana linijka}\ita{riga spezzata}]
\label{broken_ruler_compass_hartshorne_start}%
\index{linijka!połamana}%
    \pol{Dane są dwa punkty $A$ i $B$ na płaszczyźnie, odległe od siebie o około trzy nible.}
    \ita{Sono dati due punti $A$ e $B$ nel piano, distanti circa tre nibli.}
    \pol{Mając do dyspozycji fragment linijki o długości jednej nibli oraz sprawny cyrkiel, narysować odcinek $AB$.}
    \ita{Disponendo di un frammento di riga lungo una nible e di un compasso funzionante, tracciare il segmento $AB$.}
\end{geoconstruction}
% Hartshorne s. 25
\pol{O problemie pisze Eves \cite[s. 181]{eves1_1972}.}
\ita{Del problema scrive Eves \cite[p. 181]{eves1_1972}.}

\begin{geoconstruction}[\pol{zardzewiały cyrkiel}\ita{compasso arrugginito}]
    \index{cyrkiel!zardzewiały}
    \pol{Dane są dwa punkty $A$ i $B$ na płaszczyźnie, odległe od siebie o około pięć nibli.}
    \ita{Sono dati due punti $A$ e $B$ nel piano, distanti circa cinque nibli.}
    \pol{Mając do dyspozycji zardzewiały cyrkiel, którym można kreślić jedynie okręgi o promieniu dwóch nibli, skonstruować trójkąt równoboczny oparty o bok $AB$.}
    \ita{Disponendo di un compasso arrugginito con cui è possibile tracciare soltanto cerchi di raggio pari a due nibli, costruire un triangolo equilatero avente $AB$ come lato.}
\end{geoconstruction}

\pol{O zardzewiałym cyrklu będzie rozmyślać Abu al-Wafa Buzjani, matematyk perski,}
\ita{Sul compasso arrugginito rifletterà Abu al-Wafa Buzjani, matematico persiano,}
\index[persons]{Buzjani, Abu al-Wafa}%
\pol{a pod koniec piętnastego wieku zastosują go w konstrukcjach Leonardo da Vinci oraz Albrecht Dürer.}
\ita{e alla fine del XV secolo esso verrà impiegato nelle costruzioni di Leonardo da Vinci e Albrecht Dürer.}
\index[persons]{Dürer, Albrech}%
\index[persons]{da Vinci, Leonardo}%
% ŹRÓDŁO: https://en.wikipedia.org/wiki/Poncelet-Steiner_theorem#History

\begin{geoconstruction}
    \pol{Dany jest punkt $A$ leżący na prostej $l$.}
    \ita{È dato un punto $A$ appartenente alla retta $l$.}
    \pol{Skonstruować prostą prostopadłą do $l$ przechodzącą przez $A$ przy użyciu linijki i zardzewiałego cyrkla.}
    \ita{Costruire la retta perpendicolare a $l$ passante per $A$ usando una riga e un compasso arrugginito.}
\end{geoconstruction}
% Hartshorne s. 25

\begin{geoconstruction}
    \pol{Dany jest punkt $A$ leżący ponad cztery nible od prostej $l$.}
    \ita{È dato un punto $A$ a più di quattro nibli dalla retta $l$.}
    \pol{Skonstruować prostą prostopadłą do $l$ przechodzącą przez $A$ przy użyciu linijki i zardzewiałego cyrkla.}
    \ita{Costruire la retta perpendicolare a $l$ passante per $A$ usando una riga e un compasso arrugginito.}
\end{geoconstruction}
% Hartshorne s. 25

\begin{geoconstruction}
    \pol{Dane są trzy niewspółliniowe punkty $A$, $B$ oraz $C$.}
    \ita{Sono dati tre punti non allineati $A$, $B$ e $C$.}
    \pol{Skonstrować punkt $D$ na prostej $AC$ tak, żeby odcinki $AD$ oraz $AB$ były równej długości, przy użyciu linijki i zardzewiałego cyrkla.}
    \ita{Costruire un punto $D$ sulla retta $AC$ in modo che i segmenti $AD$ e $AB$ abbiano la stessa lunghezza, usando una riga e un compasso arrugginito.}
\end{geoconstruction}
% Hartshorne s. 26

\begin{geoconstruction}
    \pol{Dany jest odcinek $AB$ o długości ponad dwóch nibli oraz prosta $l$, która nie przechodzi przez końce odcinka.}
    \ita{Sono dati un segmento $AB$ di lunghezza superiore a due nibli e una retta $l$ che non passa per gli estremi del segmento.}
    \pol{Skonstrować punkt $C$ na prostej $l$ tak, żeby odcinki $AB$ oraz $AC$ były równej długości, przy użyciu linijki i zardzewiałego cyrkla.}
    \ita{Costruire un punto $C$ sulla retta $l$ in modo che i segmenti $AB$ e $AC$ abbiano la stessa lunghezza, usando una riga e un compasso arrugginito.}
\end{geoconstruction}
% Hartshorne s. 26

\begin{proposition}
\label{broken_ruler_compass_hartshorne_end}%
    \pol{Każdą konstrukcję geometryczną, którą można wykonać cyrklem i linijką, można powtórzyć przy użyciu zardzewiałego cyrkla i linijki.}
    \ita{Qualsiasi costruzione geometrica realizzabile con riga e compasso può essere ripetuta usando una riga e un compasso arrugginito.}
\end{proposition}
% Hartshorne s. 26
