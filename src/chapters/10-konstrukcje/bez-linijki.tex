\pol{\section{Konstrukcje bez linijki}}
\ita{\section{Costruzioni senza riga}}

\pol{Coxeter \cite[s. 95]{coxeter_1967} sugeruje skonstruować (samym) cyrklem wierzchołki sześciokąta foremnego, odcinek dwa razy dłuższy (i krótszy) od danego, obraz inwersyjny danego punktu leżącego dowolnie blisko środka i wreszcie podzielić dany odcinek na $n$ równych części.}
\ita{Coxeter \cite[p. 95]{coxeter_1967} suggerisce di costruire con il solo compasso i vertici di un esagono regolare, un segmento lungo il doppio (e la metà) di un segmento dato, l'immagine inversiva di un punto dato situato arbitrariamente vicino al centro e infine di dividere un segmento dato in $n$ parti uguali.}

\pol{Okazuje się, że do żadnej klasycznej konstrukcji nie jest potrzebna linijka.}
\ita{Si scopre che nessuna costruzione classica richiede realmente l'uso della riga.}

\begin{theorem}[\pol{twierdzenie Mohra-Mascheroniego}\ita{teorema di Mohr-Mascheroni}]
\index{twierdzenie!Mohra-Mascheroniego}%
    \pol{Jeśli dana konstrukcja jest wykonalna za pomocą cyrkla i~linijki, to jest ona wykonalna za pomocą samego cyrkla, o ile pominiemy rysowanie linii i ograniczymy się do wyznaczania punktów konstrukcji.}
    \ita{Se una costruzione è realizzabile con riga e compasso, allora è realizzabile con il solo compasso, purché si ometta il tracciamento delle linee e ci si limiti a determinare i punti della costruzione.}
\end{theorem}
% PRZECZYTANO: https://en.wikipedia.org/wiki/Mohr-Mascheroni_theorem

\pol{Dla dowodu wystarczy pokazać, że cyrkel wystarcza do przeprowadzenia pięciu konstrukcji:}
\ita{Per la dimostrazione basta mostrare che il compasso è sufficiente per eseguire le seguenti cinque costruzioni:}
\begin{enumerate}
    \item \pol{poprowadzenia prostej przez dwa punkty,}\ita{tracciare la retta passante per due punti,}
    \item \pol{wykreślenia okręgu o danym środku i promieniu,}\ita{tracciare un cerchio di centro e raggio assegnati,}
    \item \pol{znalezienia punktu przecięcia dwóch nierównoległych prostych,}\ita{trovare il punto di intersezione di due rette non parallele,}
    \item \pol{znalezienia punktu (lub punktów) przecięcia prostej z okręgiem,}\ita{trovare il punto (o i punti) di intersezione tra una retta e un cerchio,}
    \item \pol{znalezienia punktu (lub punktów) przecięcia dwóch okręgów.}\ita{trovare il punto (o i punti) di intersezione di due cerchi.}
\end{enumerate}

\begin{problem}[\pol{zadanie Napoleona}\ita{problema di Napoleone}]
\index{zadanie!Napoleona}
\label{napoleon_problem}
	\pol{Dany jest okrąg oraz jego środek.}
	\ita{Sono dati un cerchio e il suo centro.}
	\pol{Podzielić okrąg na cztery łuki równej miary korzystając z cyrkla, ale nie linijki.}
	\ita{Dividere il cerchio in quattro archi di uguale ampiezza usando il compasso ma non la riga.}
\end{problem}
% PRZECZYTANO: https://en.wikipedia.org/wiki/Napoleon%27s_problem
