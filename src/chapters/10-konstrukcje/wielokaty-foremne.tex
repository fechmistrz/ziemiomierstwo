\pol{\section{Wielokąty foremne}}
\ita{\section{Poligoni regolari}}
\index{wielokąt!foremny}

\begin{problem}
    \pol{Skonstrować trójkąt równoboczny wpisany w okrąg, którego środek nie jest znany.}
    \ita{Costruire un triangolo equilatero inscritto in un cerchio di cui non è noto il centro.}
    \hfill \emph{\pol{(7 kroków)}\ita{(7 passi)}}
\end{problem}

\begin{problem}
    \pol{Skonstrować kwadrat.}
    \ita{Costruire un quadrato.}
    \hfill \emph{\pol{(9 kroków)}\ita{(9 passi)}}
\end{problem}

\pol{Konstrukcja trójkąta równobocznego oraz kwadratu będzie znana dobrze w~starożytności i~nie wiadomo, kto pierwszy na nią wpadnie.}
\ita{La costruzione del triangolo equilatero e del quadrato sarà ben nota nell'antichità e non si sa chi l'abbia scoperta per primo.}
\pol{Pojawią się u Euklidesa (I.1), (I.46).}
\ita{Esse compaiono negli \emph{Elementi} di Euclide (I.1), (I.46).}

\begin{problem}
    \pol{Skonstrować pięciokąt foremny.}
    \ita{Costruire un pentagono regolare.}
\end{problem}

\pol{Pięciokąt foremny jest pierwszym wielokątem foremnym, którego konstrukcja nie jest oczywista.}
\ita{Il pentagono regolare è il primo poligono regolare la cui costruzione non è immediata.}
\pol{Różne pomysły podadzą Euklides (IV.11), Ptolemeusz w~Almageście (łatwiej będzie jednak przeczytać Guzickiego \cite[s. 286-287]{guzicki_2021}).}
\ita{Diverse costruzioni saranno proposte da Euclide (IV.11) e da Tolomeo nell'\emph{Almagesto} (anche se sarà più semplice leggere Guzicki \cite[p. 286-287]{guzicki_2021}).}
\index[persons]{Ptolemeusz}%
\index{Almagest}%
\pol{Nie pominie jej Hartshorne \cite[s. 45-49]{hartshorne2000} w swoim ,,przedłużeniu'' Elementów z komentarzem, że dowód Euklidesa poprawności konstrukcji tworzy standard pięknego dowodu matematycznego.}
\ita{Hartshorne \cite[p. 45-49]{hartshorne2000} la include nella sua sorta di ``continuazione'' degli \emph{Elementi}, osservando che la dimostrazione di Euclide costituisce un modello di elegante dimostrazione matematica.}
\pol{Jeszcze inne eleganckie rozwiązanie (opisane w~$\Delta_{24}^{10}$) znajdzie Yosifusa Hirano, japoński miłośnik matematyki z XIX wieku, a opisze jego przyjaciel Kawakita Chorin w~manuskrypcie ,,\emph{Sanpo Jyojutu Kaigi}'' (rozwiązania Sanpo Jyojutu).}
\ita{Un'altra elegante soluzione (descritta in $\Delta_{24}^{10}$) sarà trovata da Yosifusa Hirano, appassionato giapponese di matematica del XIX secolo, e verrà descritta dal suo amico Kawakita Chorin nel manoscritto \emph{Sanpo Jyojutu Kaigi} (Soluzioni del Sanpo Jyojutu).}
\index[persons]{Hirano, Yosifusa}%
\pol{Konstrukcja Hirano to kolejny powód, by sięgnąć po książkę Fukagawy, Pedoe \cite{fukagawa_1989}.}
\ita{La costruzione di Hirano è un'ulteriore ragione per consultare il libro di Fukagawa e Pedoe \cite{fukagawa_1989}.}

\pol{Jeśli mamy zadany jeden z jego boków, konstrukcja wymaga 11 kroków.}
\ita{Se è assegnato uno dei lati, la costruzione richiede 11 passi.}

\begin{problem}
    \pol{Skonstruować wielokąt foremny o $n$ kątach.}
    \ita{Costruire un poligono regolare con $n$ lati.}
\end{problem}

\pol{Poza wyżej wymienionymi, Euklides potrafi skonstruować sześciokąt foremny (IV.15) albo, po wykreśleniu trójkąta i pięciokąta opisanych wewnątrz tego samego okręgu, piętnastokąt (IV.16).}
\ita{Oltre ai casi già menzionati, Euclide sa costruire un esagono regolare (IV.15) e, dopo aver costruito nello stesso cerchio un triangolo e un pentagono, anche un pentadecagono (IV.16).}
\index[persons]{Euklides}%
\pol{Mając wielokąt o $n$ bokach, bisekcja kąta środkowego pozwala podwoić liczbę boków do $2n$.}
\ita{Dato un poligono di $n$ lati, la bisezione dell'angolo al centro permette di raddoppiare il numero dei lati fino a $2n$.}
\pol{Możemy więc uznać, że Euklides wykreśli wielokąty, które mają}
\ita{Possiamo quindi considerare che Euclide sia in grado di costruire poligoni con}
\begin{equation}
    3, 4, 5, 6, 8, 10, 12, 15, 16, 20, \ldots
\end{equation}
\pol{boków.}
\ita{lati.}
\pol{Lista zostanie rozszerzona dopiero w 1796 roku przez Gaussa: pokaże, że siedemnastokąt foremny jest konstruowalny.}
\ita{L'elenco verrà ampliato soltanto nel 1796 da Gauss, che mostrerà che il poligono regolare di 17 lati è costruibile.}
\index{Gauss, Carl Friedrich}%
\index{wielokąt!siedemnastokąt}%
\pol{Będzie tak zadowolony z tego wyniku, że zażyczy sobie wyryć właśnie tę figurę na swoim grobie.}
\ita{Sarà così soddisfatto di questo risultato da desiderare che tale figura venga incisa sulla sua tomba.}
\pol{Faktycznej konstrukcji dokona Johannes Erchinger w 1800 roku.}
\ita{La costruzione effettiva sarà realizzata da Johannes Erchinger nel 1800.}
\index[persons]{Erchinger, Johannes}%
\pol{Hartstone \cite[s. 250-259]{hartshorne2000} poświęca tej konstrukcji całą sekcję; Guzicki nie szczędzi rachunków \cite[s. 290-299]{guzicki_2021}.}
\ita{Hartshorne \cite[p. 250-259]{hartshorne2000} dedica a questa costruzione un'intera sezione; Guzicki non risparmia i dettagli di calcolo \cite[p. 290-299]{guzicki_2021}.}
\pol{Siedemnastokąt pojawi się też u Evesa \cite[s. 185-191]{eves1_1972} i Coxetera \cite[s. 42, 43]{coxeter_1967}.}
\ita{L'eptadecagono compare anche in Eves \cite[p. 185-191]{eves1_1972} e in Coxeter \cite[p. 42, 43]{coxeter_1967}.}

\pol{W ogólności, mamy:}
\ita{Più in generale, vale il seguente risultato:}
\begin{theorem}[\pol{Gaussa-Wantzla}\ita{di Gauss-Wantzel}]
    \label{gauss_wantzel}
    \pol{Wielokąt foremny o $n$ bokach jest konstruowalny przy pomocy cyrkla i~linijki bez podziałki wtedy i tylko wtedy, kiedy $n$ jest postaci}
    \ita{Un poligono regolare di $n$ lati è costruibile con riga non graduata e compasso se e solo se $n$ è della forma}
    \begin{equation}
        n = 2^r p_1 \cdot \ldots \cdot p_s,
    \end{equation}
    \pol{$r, s \ge 0$, gdzie $p_i$ są różnymi liczbami pierwszymi postaci $2^{2^k} + 1$, na przykład: $3$, $5$, $17$, $257$, $65537$.}
    \ita{$r, s \ge 0$, dove i $p_i$ sono distinti numeri primi della forma $2^{2^k}+1$, per esempio: $3$, $5$, $17$, $257$, $65537$.}
    \index{liczba pierwsza!Fermata}%
\end{theorem}

\pol{Gauss znajdzie warunek wystarczający i napisze bez dowodu w~\emph{Disquisitiones Arithmeticae} (po polsku byłyby to Dociekaniach arytmetyczne, 1801), że jest też konieczny.}
\ita{Gauss troverà la condizione sufficiente e scriverà senza dimostrazione nelle \emph{Disquisitiones Arithmeticae} (1801) che essa è anche necessaria.}
\pol{Brakujący dowód doda Pierre Laurent Wantzel w 1837 roku.}
\ita{La dimostrazione mancante sarà fornita da Pierre Laurent Wantzel nel 1837.}
\index[persons]{Wantzel, Pierre Laurent}%
\pol{Spisany współczesnym językiem znajdzie się u Hartstone'a \cite[s. 258]{hartshorne2000}; wzmiankować go będzie Guzicki \cite[s. 285]{guzicki_2021}.}
\ita{Una presentazione in linguaggio moderno si trova in Hartshorne \cite[p. 258]{hartshorne2000}; Guzicki ne fa menzione in \cite[p. 285]{guzicki_2021}.}

\pol{Twierdzenie Gaussa zachęca do znalezienia przepisu na wielokąty foremne o większej liczbie boków.}
\ita{Il teorema di Gauss incoraggia a cercare costruzioni di poligoni regolari con un numero ancora maggiore di lati.}
\pol{W 1822 roku Magnus Georg von Paucker\footnote{Właściwie Магнус-Георг Андреевич Паукер, urodzon w Simunie, wtedy części Imperium Rosyjskiego, później Łotwy.} \cite{paucker_1822}, a po dziesięciu latach także Friederich Julius Richelot \cite{richelot_1832} znajdą konstrukcję $257$-kąta foremnego.}
\ita{Nel 1822 Magnus Georg von Paucker\footnote{In realtà Магнус-Георг Андреевич Паукер, nato a Simuna, allora parte dell'Impero Russo e oggi in Estonia.} \cite{paucker_1822}, e dieci anni più tardi anche Friederich Julius Richelot \cite{richelot_1832}, troveranno una costruzione del poligono regolare di $257$ lati.}
\index[persons]{Paucker@von Paucker, Magnus Georg}%
\index[persons]{Richelot, Friederich Julius}%
\pol{Ich cierpliwość przyćmiewa konstrukcja $65\,537$-kąta Johanna Hermesa z 1896 roku, na którą poświęci dziesięć lat swojego życia (i dwieście stron rękopisu).}
\ita{La loro pazienza sarà però superata dalla costruzione del poligono regolare di $65\,537$ lati realizzata da Johann Hermes nel 1896, alla quale dedicò dieci anni della sua vita e duecento pagine di manoscritto.}
\index[persons]{Hermes, Johann(es?)}%

\pol{W 2025 roku nie będzie jeszcze wiadomo, ile jest liczb pierwszych Fermata, więc ludzkość będzie w stanie wykreślić wielokąty o nieparzyście wielu bokach dokładnie wtedy, gdy liczba tych boków będzie dzielnikiem $2^{32} - 1 = 4\,294\,967\,295$.}
\ita{Nel 2025 non si saprà ancora quante siano le prime di Fermat, per cui l'umanità sarà in grado di costruire poligoni con un numero dispari di lati esattamente quando tale numero divide $2^{32}-1=4\,294\,967\,295$.}
\pol{Dla oszczędności papieru i tuszu darujemy sobie wypisywanie tych dzielników.}
\ita{Per risparmiare carta e inchiostro eviteremo di elencare tutti questi divisori.}

\pol{Dwie konstrukcje przybliżone pięciokąta foremnego podadzą Leonardo da Vinci (gdzie kąty środkowe mają miarę $71.6350$ oraz $72.5475$ zamiast dokładnie $72$ stopni) oraz Dürer; obydwie można znaleźć u Guzickiego \cite[s. 287-290]{guzicki_2021}.}

\pol{Jeżeli mamy do dyspozycji trysektor kąta, potrafimy zbudować więcej wielokątów foremnych niż sugeruje to twierdzenie \ref{gauss_wantzel}:}
\ita{Se disponiamo di un trisettore d'angolo, possiamo costruire più poligoni regolari di quanti ne consenta il teorema \ref{gauss_wantzel}:}
\index{trysekcja!kąta}
\index{trysektor kąta}

\begin{proposition}
    \pol{Wielokąt foremny o $n$ bokach jest konstruowalny przy pomocy cyrkla, linijki bez podziałki oraz trysektora kąta wtedy i tylko wtedy, kiedy $n$ jest postaci}
    \ita{Un poligono regolare di $n$ lati è costruibile con compasso, riga non graduata e trisettore d'angolo se e solo se $n$ è della forma}
    \begin{equation}
        n = 2^r 3^q p_1 \cdot \ldots \cdot p_s,
    \end{equation}
    \pol{$q, r, s \ge 0$, gdzie $p_i$ są różnymi liczbami pierwszymi Pierponta, postaci $2^t3^u + 1$.}
    \ita{$q, r, s \ge 0$, dove i $p_i$ sono distinti numeri primi di Pierpont della forma $2^t3^u + 1$.}
    \index{liczba pierwsza!Pierponta}
\end{proposition}

\pol{To są dokładnie wielokąty, które można skonstruować używając stożkowych, albo równoważnie origami.}
\ita{Questi sono esattamente i poligoni che si possono costruire usando le coniche oppure, in modo equivalente, l'origami.}
\pol{Lista nowych nieparzystych liczb $n$, który dostarcza ten fakt, to \textbf{7}, $9$, \textbf{13}, \textbf{19}, $21$, $27$, $35$, \textbf{37}, $39$, $45$, $57$, $63$, $65$, \textbf{73}, $81$, $91$, $95$, \textbf{97}, $105$, \textbf{109}, $111$, $117$, $119$, $133$, $135$, $153$, \textbf{163}, $171$, $185$, $189$, \textbf{193}, $195$, $219$, $221$, $243$, $247$, \textbf{257}, $259$, $273$, $285$, $291\ldots$ (pogrubiono liczby pierwsze).}
\ita{L'elenco dei nuovi valori dispari di $n$ forniti da questo risultato è \textbf{7}, $9$, \textbf{13}, \textbf{19}, $21$, $27$, $35$, \textbf{37}, $39$, $45$, $57$, $63$, $65$, \textbf{73}, $81$, $91$, $95$, \textbf{97}, $105$, \textbf{109}, $111$, $117$, $119$, $133$, $135$, $153$, \textbf{163}, $171$, $185$, $189$, \textbf{193}, $195$, $219$, $221$, $243$, $247$, \textbf{257}, $259$, $273$, $285$, $291\ldots$ (i numeri primi sono evidenziati in grassetto).}
