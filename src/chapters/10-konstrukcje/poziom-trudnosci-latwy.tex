\pol{Hartshorne \cite[s. 122]{hartshorne2000} napisze, że Kartezjusz około 1637 pokazał, że:}
\ita{Hartshorne \cite[p. 122]{hartshorne2000} scrive che Cartesio mostrò intorno al 1637 che:}

\begin{proposition}
    \pol{Niech $P_1 = (x_1, y_1)$, $P_2 = (x_2, y_2)$, \ldots, $P_n = (x_n, y_n)$ będą punktami na rzeczywistej płaszczyźnie kartezjańskiej, na której dane są również punkty $(0, 0)$ oraz $(1, 0)$.}
    \ita{Siano $P_1 = (x_1, y_1)$, $P_2 = (x_2, y_2)$, \ldots, $P_n = (x_n, y_n)$ punti del piano cartesiano reale, sul quale sono inoltre assegnati i punti $(0,0)$ e $(1,0)$.}
    \pol{Wtedy następujące warunki są równoważne:}
    \ita{Allora le seguenti condizioni sono equivalenti:}
    \begin{itemize}
        \item \pol{punkt $P = (x, y)$ można wykreślić przy pomocy cyrkla i linijki;}
        \ita{il punto $P=(x,y)$ può essere costruito con riga e compasso;}
        \item \pol{liczby $x, y$ można otrzymać z $a_1, a_2, \ldots, a_n, b_1, b_2, \ldots, b_n$ przy pomocy dodawania, odejmowania, mnożenia, dzielenia lub rozwiązując skończoną liczbę liniowych lub kwadratowych równań, w których wyciągamy pierwiastki kwadratowe jedynie z dodatnich liczb.}
        \ita{i numeri $x,y$ possono essere ottenuti da $a_1, a_2, \ldots, a_n, b_1, b_2, \ldots, b_n$ mediante addizione, sottrazione, moltiplicazione, divisione oppure risolvendo un numero finito di equazioni lineari o quadratiche, nelle quali si estraggono radici quadrate soltanto di numeri positivi.}
    \end{itemize}
\end{proposition}
