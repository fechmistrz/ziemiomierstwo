\pol{\subsection{Problem Cramera-Castillona}}
\ita{\subsection{Problema di Cramer-Castillon}}

\pol{Poniższy problem będzie mieć ciekawą historię.}
\ita{Il seguente problema avrà una storia interessante.}

\index{zadanie!Cramera-Castillona}
\begin{problem}
    \label{problem_cramera_castillona}%
    \pol{Dane jest $n$ punktów oraz okrąg.}
    \ita{Sono dati $n$ punti e un cerchio.}
    \pol{Znaleźć taki wielokąt wpisany w okrąg, że na każdym jego boku (lub jego przedłużeniu) znajduje się dokładnie jeden z danych punktów.}
    \ita{Trovare un poligono inscritto nel cerchio tale che su ciascun lato (o sul suo prolungamento) si trovi esattamente uno dei punti assegnati.}
\end{problem}
