\pol{Konstrukcje od \ref{delta_2024_12_start} do \ref{delta_2024_12_end} opisze Stanisław Majchrzak w $\Delta_{24}^{12}$.}
\ita{Le costruzioni da \ref{delta_2024_12_start} a \ref{delta_2024_12_end} sono descritte da Stanisław Majchrzak in $\Delta_{24}^{12}$.}

\begin{geoconstruction}
    \label{delta_2024_12_start}
    \pol{Dane jest pięć punktów leżących na okręgu $\Gamma$.}
    \ita{Sono dati cinque punti appartenenti a un cerchio $\Gamma$.}
    \pol{Wykreślić styczną do $\Gamma$ w jednym z tych punktów.}
    \ita{Costruire la tangente a $\Gamma$ in uno di questi punti.}
\end{geoconstruction}

\pol{Uzasadnienie poprawności tej konstrukcji wymaga znajomości twierdzenia Pascala, patrz $\Delta_{14}^9$.}
\ita{La dimostrazione della correttezza di questa costruzione richiede la conoscenza del teorema di Pascal; si veda $\Delta_{14}^9$.}

\begin{geoconstruction}
    \pol{Dane jest pięć punktów leżących na okręgu $\Gamma$ oraz prosta $l$ przechodząca przez jeden z nich.}
    \ita{Sono dati cinque punti appartenenti a un cerchio $\Gamma$ e una retta $l$ passante per uno di essi.}
    \pol{Wyznaczyć drugi punkt przecięcia okręgu $\Gamma$ oraz prostej $l$.}
    \ita{Determinare il secondo punto d'intersezione tra il cerchio $\Gamma$ e la retta $l$.}
\end{geoconstruction}

\pol{Rozwiązanie zostanie wydrukowane w $\Delta_{17}^6$.}
\ita{La soluzione sarà pubblicata in $\Delta_{17}^6$.}

\begin{geoconstruction}
    \pol{Dane są dwa okręgi przecinające się w dwóch punktach.}
    \ita{Sono dati due cerchi che si intersecano in due punti.}
    \pol{Wyznaczyć środek jednego z nich.}
    \ita{Determinare il centro di uno di essi.}
\end{geoconstruction}

\pol{W kolejnych konstrukcjach przydatna jest znajomość twierdzenia: biegunowe dowolnego punktu $P$ względem okręgów należących do jednego pęku są współpękowe.}
\ita{Nelle costruzioni successive è utile conoscere il seguente teorema: le polari di un punto arbitrario $P$ rispetto ai cerchi appartenenti a uno stesso fascio sono concorrenti.}

\begin{geoconstruction}
    \pol{Dane są dwa okręgi oraz punkt $A$ na zewnątrz jednego z nich (wewnątrz obydwu).}
    \ita{Sono dati due cerchi e un punto $A$ esterno a uno di essi (e interno a entrambi).}
    \pol{Wykreślić okrąg przechodzący przez $A$, który jest współpękowy z dwoma danymi.}
    \ita{Costruire un cerchio passante per $A$ che appartenga allo stesso fascio dei due cerchi assegnati.}
\end{geoconstruction}

\begin{geoconstruction}
    \label{delta_2024_12_end}
    \pol{Dane są cztery (w trudniejszej wersji: trzy) okręgi takie, że żadne trzy nie są współpękowe.}
    \ita{Sono dati quattro cerchi (nella versione più difficile: tre) tali che nessuna terna appartenga allo stesso fascio.}
    \pol{Wyznaczyć środek przynajmniej jednego z nich.}
    \ita{Determinare il centro di almeno uno di essi.}
\end{geoconstruction}

\pol{Dla rozłącznych okręgów należących do jednego pęku, środka nie da się (!) skonstruować.}
\ita{Per cerchi disgiunti appartenenti a uno stesso fascio, il centro non può (!) essere costruito.}
