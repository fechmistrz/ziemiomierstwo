\pol{\section{Twierdzenie sinusów}}
\ita{\section{Teorema dei seni}}
\pol{\index{twierdzenie!sinusów}}\ita{\index{teorema!dei seni}}%

\begin{proposition}[\pol{twierdzenie sinusów}\ita{teorema dei seni}]
    \pol{Niech trójkąt o bokach długości $a, b, c$ oraz kątach $\alpha$, $\beta$, $\gamma$ naprzeciwko tychże boków (w tej samej kolejności) będzie wpisany w okrąg o promieniu $R$.}
    \ita{Sia un triangolo con lati di lunghezze $a, b, c$ e angoli $\alpha$, $\beta$, $\gamma$ opposti a questi lati (nello stesso ordine), inscritto in un cerchio di raggio $R$.}
    \pol{Wtedy}
    \ita{Allora}
    \label{twierdzenie_sinusow}%
    \begin{equation}
        \frac{a}{\sin \alpha} = \frac{b}{\sin \beta} = \frac{c}{\sin \gamma} = 2R.
    \end{equation}
\end{proposition}

\pol{Ptolemeusz w II wieku będzie wiedzieć o równoważnej formie, że boki trójkąta są proporcjonalne do cięciw podwojonych kątów naprzeciwko.}
\ita{Tolomeo nel II secolo conoscerà una forma equivalente, secondo cui i lati del triangolo sono proporzionali alle corde dei doppi angoli opposti.}
\pol{Brahmagupta w VII wieku wyrazi promień okręgu opisanego jako iloczyn dwóch boków trójkąta podzielony przez podwojoną wysokość, stąd można już wyprowadzić twierdzenie.}
\ita{Brahmagupta nel VII secolo esprimerà il raggio del cerchio circoscritto come il prodotto di due lati del triangolo diviso per il doppio dell'altezza; da qui si può già dedurre il teorema.}
\pol{Sferyczną wersję pierwszy poda któryś z~muzułmańskich matematyków X wieku: Abu-Mahmud al-Khujandi, Abu Nasr Mansur albo Abu al-Wafa.}
\ita{Una versione sferica sarà data per primo da uno dei matematici musulmani del X secolo: Abu-Mahmud al-Khujandi, Abu Nasr Mansur oppure Abu al-Wafa.}

\pol{Coxeter \cite[s. 28, 29]{coxeter_1967} wyprowadza twierdzenie sinusów z symetrii okręgu opisanego na trójkącie.}
\ita{Coxeter \cite[pp. 28, 29]{coxeter_1967} deduce il teorema dei seni dalla simmetria del cerchio circoscritto al triangolo.}

\ita{Bogdańska e Neugebauer \cite[p. 89]{neugebauer_2018} si baseranno sul fatto \ref{maybe_3_28}, che gli angoli inscritti insistenti sullo stesso arco hanno misure uguali.}
