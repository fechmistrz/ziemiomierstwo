\section{Twierdzenie tangensów}
\index{twierdzenie!tangensów}%

\pol{Twierdzenie odkryje Abu al-Wafa Buzjani w X wieku oraz niezależnie Ibn Mu'adh al-Jayyani (arabski matematyk, autor pierwszego traktatu o trygonometrii sferycznej) w XI wieku.}
\ita{Il teorema fu scoperto da Abu al-Wafa Buzjani nel X secolo e indipendentemente da Ibn Mu'adh al-Jayyani (matematico arabo, autore del primo trattato di trigonometria sferica) nell'XI secolo.}
\index[persons]{Buzjani, Abu al-Wafa}%
\index[persons]{Ibn Mu'adh al-Jayyani}%
% The law of tangents for spherical triangles was described in the 13th century by Persian mathematician Nasir al-Din al-Tusi (1201–1274), who also presented the law of sines for plane triangles in his five-volume work Treatise on the Quadrilateral.
\pol{Po polsku często padać będzie nazwa wzór Regiomontana.}
\ita{In polacco viene spesso usata la denominazione formula di Regiomontano.}
\index{wzór Regiomontana}%

\begin{proposition}[\pol{twierdzenie tangensów}\ita{teorema delle tangenti}]
	\pol{W trójkącie o bokach długości $a, b, c$ z kątami $\alpha$, $\beta$ naprzeciwko krawędzi $a$, $b$ zachodzi}
	\ita{In un triangolo con lati di lunghezza $a, b, c$ e angoli $\alpha$, $\beta$ opposti ai lati $a$, $b$ vale}
	\label{twierdzenie_tangensow}%
	\begin{equation}
		\frac{a-b}{a+b} = \frac{\tan \frac 1 2 (\alpha - \beta)}{\tan \frac 1 2 (\alpha + \beta)}.
	\end{equation}
\end{proposition}

\pol{Prosty dowód wykorzystuje twierdzenie sinusów.}
\ita{Una dimostrazione semplice utilizza il teorema dei seni.}
\index{twierdzenie!sinusów}%
\pol{Inny korzysta ze wzorów Mollweidego (to jest faktu \ref{wzor_mollweidego}), które wyprowadza się i tak z twierdzenia sinusów.}
\ita{Un altro usa le formule di Mollweide (cioè il fatto \ref{wzor_mollweidego}), che comunque si ricavano dal teorema dei seni.}
\index{wzór!Mollweidego}%

\pol{Dopóki nie zostanie wynaleziony komputer albo chociaż kalkulator elektroniczny, twierdzenie tangensów będzie mieć przewagę nad twierdzeniem kosinusów, ponieważ nie wymaga sięgania po tablice logarytmiczne do znalezienia pierwiastka}
\ita{Finché non verranno inventati il computer o almeno una calcolatrice elettronica, il teorema delle tangenti avrà un vantaggio rispetto al teorema del coseno, poiché non richiede l’uso di tavole logaritmiche per estrarre la radice}
\index{twierdzenie!kosinusów}%
\begin{equation}
c = \sqrt{a^2 + b^2 - 2 a b \cos \gamma}.
\end{equation}

\pol{Po wynalezieniu rzeczonej maszyny, przewaga nie zniknie.}
\ita{Dopo l’invenzione di tale macchina, il vantaggio non scomparirà.}
\pol{Twierdzenie tangensów pozwala uniknąć błędów numerycznych, kiedy $a \approx b$, zaś kąt $\gamma$ jest bardzo mały.}
\ita{Il teorema delle tangenti permette di evitare errori numerici quando $a \approx b$ e l’angolo $\gamma$ è molto piccolo.}

\pol{Istnieje rzadko stosowane uogólnienie do czworokątów cyklicznych (wpisanych w okrąg):}
\ita{Esiste una generalizzazione poco usata ai quadrilateri ciclici (inscritti in una circonferenza):}

% źródło: https://de.wikipedia.org/wiki/Tangenssatz#Verallgemeinerung_für_Sehnenvierecke
\begin{proposition}
	\pol{W czworokącie $ABCD$ wpisanym w okrąg, o bokach $a = AB$, $b = BC$, $c = CD$, $d = DA$ oraz kątach $\alpha = \angle DAB$, $\beta = \angle ABC$ zachodzi}
	\ita{In un quadrilatero $ABCD$ inscritto in una circonferenza, con lati $a = AB$, $b = BC$, $c = CD$, $d = DA$ e angoli $\alpha = \angle DAB$, $\beta = \angle ABC$ vale}
	\begin{equation}
		\frac{(a+c)(b+d)}{(a-c)(b-d)} = \frac{\tan \frac 1 2 (\alpha + \beta)}{\tan \frac 1 2 (\alpha - \beta)}.
	\end{equation}
\end{proposition}
