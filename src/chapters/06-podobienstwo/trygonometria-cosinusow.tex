\ita{\section{Teorema del coseno}}

\pol{Twierdzenie kosinusów jest bardzo stare.}
\ita{Il teorema del coseno è molto antico.}
\pol{Euklides rozpatrzy je w Elementach osobno dla trójkątów rozwartokątnych (II.12) i ostrokątnych (II.13).}
\ita{Euclide lo tratta negli Elementi separatamente per i triangoli ottusangoli (II.12) e acutangoli (II.13).}

\pol{Perski matematyk Jamshid al-Kashi znajdzie wartość $2\pi$ z~dokładnością do szesnastu cyfr, będzie autorem najdokładniejszych tablic trygonometrycznych swoich czasów i poda równoważną postać wzoru,}
\ita{Il matematico persiano Jamshid al-Kashi calcolerà il valore di $2\pi$ con precisione fino a sedici cifre, sarà autore delle tavole trigonometriche più precise del suo tempo e fornirà una forma equivalente della formula,}
\index[persons]{al-Kashi, Jamshid}
\begin{equation}
	c = \sqrt{(b - a \cos \gamma)^2 + (a \sin \gamma)^2}
\end{equation}
\pol{dla ostrego kąta $\gamma$.}
\ita{per l'angolo acuto $\gamma$.}
\pol{Taka sama metoda rozwiązywania trójkątów pojawi się w Europie w 1464 roku, kiedy Regiomontanus opublikuje \emph{,,De triangulis omnimodis''} (czyli \emph{,,O trójkątach wszelkiego rodzaju''}).}
\ita{Lo stesso metodo per risolvere i triangoli apparirà in Europa nel 1464, quando Regiomontano pubblicherà \emph{De triangulis omnimodis} (cioè \emph{Sui triangoli di ogni tipo}).}
\index[persons]{Regiomontanus}%
\pol{Współczesna forma twierdzenia to zasługa François Viète'a.}
\ita{La forma moderna del teorema è merito di François Viète.}
\index[persons]{Viète, François}%

\begin{proposition}[\pol{twierdzenie kosinusów}\ita{teorema del coseno}]
	\pol{W trójkącie o bokach długości $a, b, c$ z kątem $\gamma$ naprzeciwko krawędzi $c$ zachodzi}
	\ita{In un triangolo con lati di lunghezza $a, b, c$ e angolo $\gamma$ opposto al lato $c$ vale}
	\label{twierdzenie_kosinusow}%
	\begin{equation}
		c^2 = a^2 + b^2 - 2ab \cos \gamma.
	\end{equation}
	% https://en.wikipedia.org/wiki/Law_of_cosines
\end{proposition}

\pol{Znamy różne dowody tego twierdzenia: korzystające z twierdzenia Pitagorasa, trzech wysokości trójkąta, twierdzenia Ptolemeusza, geometrii koła albo twierdzenia sinusów.}
\ita{Conosciamo varie dimostrazioni di questo teorema: basate sul teorema di Pitagora, sulle tre altezze del triangolo, sul teorema di Tolomeo, sulla geometria del cerchio o sul teorema dei seni.}
\pol{We Francji twierdzenie kosinusów do późna będzie nazywane \emph{théorème d'Al-Kashi}, na cześć wspomnianego wyżej Persa.}
\ita{In Francia il teorema del coseno sarà a lungo chiamato \emph{théorème d'Al-Kashi}, in onore del suddetto persiano.}

\pol{Piszą o nim Guzicki \cite[s. 258]{guzicki_2021}, Hartshorne \cite[s. 185]{hartshorne2000} jako ćwiczenie.}
\ita{Ne scrivono Guzicki \cite[s. 258]{guzicki_2021}, Hartshorne \cite[s. 185]{hartshorne2000} come esercizio.}

\begin{corollary}[\pol{twierdzenie Apoloniusza}\ita{teorema di Apollonio}]
	\pol{W trójkącie $ABC$ poprowadzono środkową $AD$.}
	\ita{In un triangolo $ABC$ è stata tracciata la mediana $AD$.}
	\pol{Wtedy suma pól kwadratów zbudowanych na bokach $AB$ i $AC$ jest dwa razy mniejsza od sumy pól kwadratów zbudowanych na środkowej $AD$ i połowie podstawy ($BD$):}
	\ita{Allora la somma delle aree dei quadrati costruiti sui lati $AB$ e $AC$ è la metà della somma delle aree dei quadrati costruiti sulla mediana $AD$ e sulla metà della base ($BD$):}
	\begin{equation}
		|AB|^2 + |AC|^2 = 2|BD|^2 + 2 |AD|^2.
	\end{equation}
	% https://en.wikipedia.org/wiki/Apollonius%27s_theorem
\end{corollary}

\begin{corollary}
	\pol{W trójkącie $\triangle ABC$ o bokach długości $a, b, c$ poprowadzono środkową oraz dwusieczną z~wierzchołka $C$.}
	\ita{In un triangolo $\triangle ABC$ con lati di lunghezza $a, b, c$ è stata tracciata la mediana e la bisettrice dall'angolo $C$.}
	\pol{Długość środkowej wynosi}
	\ita{La lunghezza della mediana è}
	\begin{equation}
		\sqrt{\frac{a^2 + b^2}{2} - \frac{c^2}{4}},
	\end{equation}
	\pol{zaś dwusieczna ma długość}
	\ita{mentre la bisettrice ha lunghezza}
	\begin{equation}
		\frac{\sqrt{ab (a+b+c)(a+b-c)}}{a+b}.
	\end{equation}
\end{corollary}
