\pol{\section{Twierdzenie Menelaosa}}
\ita{\section{Teorema di Menelao}} % Menelausa!!!

\pol{Znamy trzy twierdzenia o współliniowości: o prostej Eulera, o prostej Wallace'a-Simsona, o prostej Auberta.}
\ita{Conosciamo tre teoremi di allineamento: sulla retta di Eulero, sulla retta di Wallace-Simson e sulla retta di Aubert.}

\begin{proposition}[\pol{twierdzenie Menelaosa}\ita{teorema di Menelao}]
	...
	\pol{Wówczas punkty $K, L, M$ są współliniowe wtedy i tylko wtedy, gdy zachodzi}
	\ita{Allora i punti $K$, $L$ e $M$ sono allineati se e solo se vale}
	\begin{equation}
		[AMB] [BKC] [CLA] = -1.
	\end{equation}
	\pol{\index{twierdzenie!Menelaosa}}
	\ita{\index{teorema!di Menelao}}
\end{proposition}

\pol{Brak pewności co do tego, kto będzie pierwszym odkrywcą tego.}
\ita{Non vi è certezza su chi ne sia stato il primo scopritore.}
\pol{Ale najstarsze ocalałe do czasów późniejszych wyjaśnienie pojawia się u Menelaosa ( Σφαιρική) jako lemat, użyteczny w sferycznej wersji twierdzenia.}
\ita{Tuttavia, la più antica esposizione giunta fino a noi compare nella \emph{Sphaerica} (Σφαιρική) di Menelao come lemma utile per la versione sferica del teorema.}

\pol{Piszą o nim Hartshorne \cite[s. 180, 181]{hartshorne2000}, Audin \cite[s. 38]{audin_2003}, Zetel \cite[s. 43]{zetel_2020}.}
\ita{Ne scrivono Hartshorne \cite[pp. 180, 181]{hartshorne2000}, Audin \cite[p. 38]{audin_2003} e Zetel \cite[p. 43]{zetel_2020}.}
\pol{Uogólnieniem twierdzenia Menelaosa będzie twierdzenie Carnota (udowodnione w 1803 roku w \emph{,,Géométrie de position}''), że iloczyn długości odcinków wyznaczonych przez prostą na bokach wielokąta i niemających wspólnych końców jest równy iloczynowi długości pozostałych odcinków;}
\ita{Una generalizzazione del teorema di Menelao è il teorema di Carnot (dimostrato nel 1803 nella \emph{``Géométrie de position''}), secondo cui il prodotto delle lunghezze dei segmenti determinati da una retta sui lati di un poligono e privi di estremi comuni è uguale al prodotto delle lunghezze dei segmenti rimanenti;}
