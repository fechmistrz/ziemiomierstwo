\pol{\index{punkt!Lemoine'a}}
\ita{\index{punto!di Lemoine}}

\pol{Emile Lemoine odczyta w 1873 roku pracę \emph{,,Sur quelques propriétés d'un point remarquable du triangle''}, gdzie po raz pierwszy pojawią się bez dowodów fakty \ref{random_prp_symmedians_proportional_squares} i \ref{punkt_lemoine}.}
\ita{Nel 1873 Émile Lemoine presenterà il lavoro \emph{,,Sur quelques propriétés d'un point remarquable du triangle''}, nel quale compariranno per la prima volta, senza dimostrazione, i fatti \ref{random_prp_symmedians_proportional_squares} e \ref{punkt_lemoine}.}
\pol{Termin symediana zaproponuje Maurice d'Ocagne w 1883 roku, jako zastępstwo \emph{antyrównoległej środkowej} używanego przez Lemoine'a.}
\ita{Il termine \emph{symmediana} sarà proposto da Maurice d'Ocagne nel 1883 come sostituto di \emph{mediana antiparallela}, espressione usata da Lemoine.}
% NouvdUs Annales de Malhematiquet, 3rd series, II. 451 (1883).
\pol{Punkt Lemoine'a pojawia się u Józefa Neuberga (praca \emph{Mémoire sur le tétraèdre} z 1886 roku).}
\ita{Il punto di Lemoine compare nell'opera di Joseph Neuberg (\emph{Mémoire sur le tétraèdre}, 1886).}

\pol{W~1847 roku Ernst Grebe skonstruuje kwadraty na zewnątrz trójkąta i odkryje punkt Lemoine'a przed Lemoinem, dlatego w literaturze padać będzie też nazwa punkt Grebego.}
\ita{Nel 1847 Ernst Grebe costruirà quadrati all'esterno di un triangolo e scoprirà il punto di Lemoine prima dello stesso Lemoine; per questo motivo nella letteratura compare anche il nome di punto di Grebe.}

\begin{definition}[\pol{symediana}\ita{symmediana}]
    \pol{Odbicie symetryczne środkowej trójkąta względem dwusiecznej wychodzącej z~tego samego wierzchołka nazywamy symedianą.}
    \ita{La riflessione della mediana di un triangolo rispetto alla bisettrice uscente dallo stesso vertice si chiama symmediana.}
\end{definition}

\pol{Symediana to prosta izogonalna ze środkową (Zetel \cite[s. 111]{zetel_2020}; Ambroszczyk i Komisarczyk \cite[s. 26]{komisarczyk_2024}; Dutka w $\Delta_{15}^2$).}
\ita{La symmediana è la retta izogonale della mediana (Zetel \cite[p. 111]{zetel_2020}; Ambroszczyk e Komisarczyk \cite[p. 26]{komisarczyk_2024}; Dutka in $\Delta_{15}^2$).}
% https://www.deltami.edu.pl/2015/02/krotka-opowiesc-o-symedianie/
\pol{Symediana jest miejscem geometrycznym punktów, których odległości od dwóch boków trójkąta są proporcjonalne do tych boków.}
\ita{La symmediana è il luogo geometrico dei punti le cui distanze da due lati del triangolo sono proporzionali a tali lati.}
\pol{Ale dużo ważniejszą jej własnością jest:}
\ita{Ma una proprietà molto più importante è la seguente:}

\begin{proposition}
    \label{random_prp_symmedians_proportional_squares}
    \pol{Symediana poprowadzona z wierzchołka $C$ trójkąta $ABC$ dzieli wewnętrznie przeciwległy bok proporcjonalnie do kwadratów długości boków przyległych:}
    \ita{La symmediana condotta dal vertice $C$ del triangolo $ABC$ divide internamente il lato opposto in proporzione ai quadrati delle lunghezze dei lati adiacenti:}
    \begin{equation}
        \frac{AF}{BF} = \frac{b^2}{a^2}.
    \end{equation}
\end{proposition}

\pol{I odwrotnie, punkt leżący na boku $AB$ i spełniający powyższą proporcję wyznacza symedianę z~wierzchołka $C$.}
\ita{Viceversa, un punto appartenente al lato $AB$ che soddisfa la proporzione precedente determina la symmediana uscente dal vertice $C$.}
\pol{Zetel \cite[s. 111, 112]{zetel_2020} uważa to za szczególny przypadek twierdzenie Steinera o~prostych izogonalnych.}
\ita{Zetel \cite[p. 111, 112]{zetel_2020} considera questo un caso particolare del teorema di Steiner sulle rette izogonali.}
\pol{(Chyba nie ma tego twierdzenia w tej książce).}

\begin{proposition}
    \pol{Symediana wychodząca z wierzchołka $A$ trójkąta $ABC$ ma długość}
    \ita{La symmediana uscente dal vertice $A$ del triangolo $ABC$ ha lunghezza}
    \begin{equation}
        \frac{bc}{b^2 + c^2} \cdot \sqrt {2b^2 + 2c^2 - a^2}.
    \end{equation}
\end{proposition}

\pol{Zetel \cite[s. 119]{zetel_2020} pisze, że można dostać tę długość z twierdzenia Stewarta.}
\ita{Zetel \cite[p. 119]{zetel_2020} scrive che questa lunghezza può essere ottenuta mediante il teorema di Stewart.}

\begin{proposition}[\pol{punkt Lemoine'a}\ita{punto di Lemoine}]
    \label{punkt_lemoine}%
    \pol{W ustalonym trójkącie $ABC$, trzy symediany przecinają się w jednym punkcie $L$, zwanym punktem Lemoine'a.}
    \ita{In un triangolo fissato $ABC$, le tre symmediane si incontrano in un unico punto $L$, detto punto di Lemoine.}
\end{proposition}

\pol{Zetel \cite[s. 114]{zetel_2020} sugeruje udowodnić \ref{punkt_lemoine} na podstawie twierdzenia odwrotnego do twierdzenia Cevy.}
\ita{Zetel \cite[p. 114]{zetel_2020} suggerisce di dimostrare \ref{punkt_lemoine} mediante il teorema inverso di Ceva.}
\pol{Ross Honsberger nazwie istnienie punktu Lemoine'a ,,jednym z klejnotów koronnych współczesnej geometrii''.}
\ita{Ross Honsberger definirà l'esistenza del punto di Lemoine come \emph{uno dei gioielli della corona della geometria moderna}.}

\begin{proposition}
    \pol{Punkt Lemoine'a minimalizuje sumę kwadratów odległości od boków trójkąta.}
    \ita{Il punto di Lemoine minimizza la somma dei quadrati delle distanze dai lati del triangolo.}
\end{proposition}

\begin{proposition}
    \pol{Punkt Lemoine'a $L$ trójkąta $ABC$ jest środkiem ciężkości trójkąta, którego wierzchołki to rzuty punktu $L$ na boki $AB$, $BC$, $AC$.}
    \ita{Il punto di Lemoine $L$ del triangolo $ABC$ è il baricentro del triangolo i cui vertici sono le proiezioni di $L$ sui lati $AB$, $BC$, $AC$.}
\end{proposition}

\begin{proposition}[\pol{twierdzenie Schlömilcha}\ita{teorema di Schlömilch}]
    \pol{Trzy proste łączące środki boków trójkąta ze środkami odpowiednich wysokości są współpękowe, przechodzą przez punkt Lemoine'a.}
    \ita{Le tre rette che congiungono i punti medi dei lati di un triangolo con i punti medi delle rispettive altezze sono concorrenti e passano per il punto di Lemoine.}
\end{proposition}

\begin{definition}[\pol{antyrównoległa}\ita{antiparallela}]
    \pol{Antyrównoległa do boku $AB$ w trójkącie $ABC$ to prosta, która tnie bok naprzeciw wierzchołka $A$ pod takim samym kątem jak ten przy wierzchołku $A$ oraz bok naprzeciw wierzchołka $B$ pod takim samym kątem jak ten przy wierzchołku $B$.}
    \ita{Un'antiparallela al lato $AB$ nel triangolo $ABC$ è una retta che interseca il lato opposto al vertice $A$ con lo stesso angolo presente in $A$ e il lato opposto al vertice $B$ con lo stesso angolo presente in $B$.}
\end{definition}

\begin{proposition}
    \pol{Trzy antyrównoległe do boków trójkąta przechodzące przez punkt Lemoine'a wyznaczają sześć nowych punktów na brzegu trójkąta. Leżą one na pierwszym okręgu Lemoine'a.}
    \ita{Le tre antiparallele ai lati del triangolo passanti per il punto di Lemoine determinano sei nuovi punti sul contorno del triangolo. Essi giacciono sul primo cerchio di Lemoine.}
\end{proposition}

\begin{proposition}
    \pol{Trzy równoległe do boków trójkąta przechodzące przez punkt Lemoine'a wyznaczają sześć nowych punktów na brzegu trójkąta. Leżą one na drugim okręgu Lemoine'a.}
    \ita{Le tre rette parallele ai lati del triangolo passanti per il punto di Lemoine determinano sei nuovi punti sul contorno del triangolo. Essi giacciono sul secondo cerchio di Lemoine.}
\end{proposition}

\pol{Istnieje jeszcze trzeci okrąg Lemoine'a, ale jego konstrukcja nie jest już taka prosta.}
\ita{Esiste anche un terzo cerchio di Lemoine, ma la sua costruzione non è altrettanto semplice.}
