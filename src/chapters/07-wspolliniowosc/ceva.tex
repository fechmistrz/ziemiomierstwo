\pol{\section{Twierdzenie Cevy}}
\ita{\section{Teorema di Ceva}}

\pol{Ważnym kryterium współpękowości trzech czewian jest:}
\ita{Un importante criterio di concorrenza di tre ceviane è il seguente:}

\begin{theorem}[Cevy]
\label{twierdzenie_cevy}%
    \pol{Dany jest trójkąt $ABC$ i trzy różne od wierzchołków punkty $K \in BC$, $L \in CA$, $M \in AB$.}
    \ita{Siano dati un triangolo $ABC$ e tre punti distinti dai vertici: $K \in BC$, $L \in CA$, $M \in AB$.}
    \pol{Wówczas czewiany $AK$, $BL$, $CM$ są współpękowe wtedy i tylko wtedy, gdy}
    \ita{Allora le ceviane $AK$, $BL$, $CM$ sono concorrenti se e solo se}
    \begin{equation}
        [AMB] [BKC] [CLA] = 1.
    \end{equation}
    \pol{\index{twierdzenie!Cevy}}
    \ita{\index{teorema!di Ceva}}
\end{theorem}

\pol{Wnioskiem z twierdzenia Cevy jest wynik znany jako lemat Steinbarta:}
\ita{Una conseguenza del teorema di Ceva è il risultato noto come lemma di Steinbart:}

\begin{lemma}[Steinbarta]
    \pol{W trójkąt $ABC$ wpisano okrąg, styczny do boków $BC$, $AC$ i $AB$ kolejno w punktach $D$, $E$, $F$.}
    \ita{In un triangolo $ABC$ è inscritta una circonferenza tangente ai lati $BC$, $AC$ e $AB$ rispettivamente nei punti $D$, $E$, $F$.}
    \pol{Punkty $X$, $Y$, $Z$ leżą odpowiednio na krótszych łukach $EF$, $FD$ i $DE$.}
    \ita{I punti $X$, $Y$, $Z$ appartengono rispettivamente agli archi minori $EF$, $FD$ e $DE$.}
    \pol{Następujące warunki są równoważne:}
    \ita{Le seguenti condizioni sono equivalenti:}
    \begin{itemize}
        \item \pol{proste $XD$, $YE$, $ZF$ są współpękowe,}
              \ita{le rette $XD$, $YE$, $ZF$ sono concorrenti,}
        \item \pol{proste $AX$, $BY$, $CZ$ są współpękowe.}
              \ita{le rette $AX$, $BY$, $CZ$ sono concorrenti.}
    \end{itemize}
\end{lemma}

\pol{Prostą, która przechodzi przez wierzchołek trójkąta i przeciwległy bok nazywa się czewianą, nazwy tej używa się też wobec odcinka trójkąta leżącego na tej prostej.}
\ita{Una retta che passa per un vertice di un triangolo e il lato opposto è detta ceviana; lo stesso termine viene usato anche per il segmento del triangolo contenuto in tale retta.}

\begin{proposition}[\pol{wzór Steinera-Routha}\ita{formula di Steiner--Routh}]
    \pol{Rozpatrzmy trójkąt $ABC$ z czewianami $AD$, $BE$, $CF$ tak, że}
    \ita{Consideriamo un triangolo $ABC$ con ceviane $AD$, $BE$, $CF$ tali che}
    \begin{equation}
        \frac{BD}{DC} = \lambda_1, \quad
        \frac{CE}{EA} = \lambda_2, \quad
        \frac{AF}{FB} = \lambda_3.
    \end{equation}
    \pol{Oznaczmy przekrój $BE$ i $CF$ przez $G_1$; $AD$ i $CF$ przez $G_2$, zaś $AD$ i $BE$ przez $G_3$.}
    \ita{Indichiamo con $G_1$ l'intersezione di $BE$ e $CF$, con $G_2$ quella di $AD$ e $CF$, e con $G_3$ quella di $AD$ e $BE$.}
    \pol{Wtedy}
    \ita{Allora}
    \begin{equation}
        \frac{S_{G_1G_2G_3}}{S_{ABC}} = \frac{(\lambda_1 \lambda_2 \lambda_3 - 1)^2}{
            (\lambda_1 \lambda_2 + \lambda_1 + 1)
            (\lambda_2 \lambda_3 + \lambda_2 + 1)
            (\lambda_3 \lambda_1 + \lambda_3 + 1)
        }
    \end{equation}
    \pol{oraz}
    \ita{e}
    \begin{equation}
        \frac{S_{DEF}}{S_{ABC}} = \frac{\lambda_1 \lambda_2 \lambda_3 + 1}{
            (\lambda_1 + 1)
            (\lambda_2 + 1)
            (\lambda_3 + 1)
        }.
    \end{equation}
\end{proposition}

\pol{Stanowi to uogólnienie twierdzeń Cevy ($\lambda_1 \lambda_2 \lambda_3 = 1$) i Menelausa ($\lambda_1 \lambda_2 \lambda_3 = -1$).}
\ita{Si tratta di una generalizzazione dei teoremi di Ceva ($\lambda_1 \lambda_2 \lambda_3 = 1$) e di Menelao ($\lambda_1 \lambda_2 \lambda_3 = -1$).}
\pol{Napiszą o nim w Delcie ($\Delta_{11}^6$, $\Delta_{12}^3$, $\Delta_{16}^8$)}
\ita{Se ne parlerà su Delta ($\Delta_{11}^6$, $\Delta_{12}^3$, $\Delta_{16}^8$).}

\begin{example}[\pol{trójkąt, który zaskoczył Feynmana}\ita{il triangolo che sorprese Feynman}]
    \pol{Dla $\lambda_1 = \lambda_2 = \lambda_3 = 2$, stosunek pól trójkątów wynosi $1 : 7$.}
    \ita{Per $\lambda_1 = \lambda_2 = \lambda_3 = 2$, il rapporto tra le aree dei triangoli è $1 : 7$.}
\end{example}
