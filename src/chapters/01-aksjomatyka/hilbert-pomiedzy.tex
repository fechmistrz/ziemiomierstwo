\pol{\subsection{Aksjomaty leżenia pomiędzy}}
\ita{\subsection{Assiomi di ordinamento}}

\pol{\index{aksjomat!leżenia pomiędzy}}
\ita{\index{assioma!di ordinamento}}

\pol{O relacji między aksjomatami B1-B4 i uporządkowaniem ciała, nad którym pracujemy, rozwodzi się Hartshorne \cite[s. 135-140]{hartshorne2000}}
\ita{Hartshorne \cite[pp.~135--140]{hartshorne2000} discute dettagliatamente la relazione tra gli assiomi B1--B4 e l'ordinamento del campo su cui si lavora.}

\begin{axiom}[\pol{leżenia pomiędzy, B1}\ita{di ordinamento, B1}]
    \pol{Jeśli punkt $B$ leży między punktami $A$ i $C$, to punkty $A$, $B$, $C$ są różnymi punktami tej samej prostej oraz punkt $B$ leży także między punktami $C$ i $A$.}
    \ita{Se il punto $B$ si trova tra i punti $A$ e $C$, allora $A$, $B$ e $C$ sono punti distinti della stessa retta e inoltre $B$ si trova anche tra $C$ e $A$.}
\end{axiom}

\begin{axiom}[\pol{leżenia pomiędzy, B2}\ita{di ordinamento, B2}]
    \pol{Dla każdej pary punktów $A$ i $B$ istnieje punkt $C$ taki, że punkt $B$ leży między punktami $A$ i $C$.}
    \ita{Per ogni coppia di punti $A$ e $B$ esiste un punto $C$ tale che $B$ si trovi tra $A$ e $C$.}
\end{axiom}

\begin{axiom}[\pol{leżenia pomiędzy, B3}\ita{di ordinamento, B3}]
    \pol{Spośród trzech punktów leżących na prostej, dokładnie jeden leży pomiędzy pozostałymi dwoma.}
    \ita{Fra tre punti allineati, esattamente uno si trova tra gli altri due.}
\end{axiom}

\begin{axiom}[\pol{leżenia pomiędzy, B4}\ita{di ordinamento, B4}]
    \pol{Niech $A$, $B$ i $C$ będą trzema niewspółliniowymi punktami, zaś $l$ prostą, która nie przechodzi przez żaden z nich.}
    \ita{Siano $A$, $B$ e $C$ tre punti non allineati e sia $l$ una retta che non passa per nessuno di essi.}

    \pol{Jeśli prosta $l$ przechodzi przez punkt między punktami $A$ i $B$, to przechodzi też przez punkt między punktami $A$ i $C$ albo $B$ i $C$, ale nie przez obydwa.}
    \ita{Se la retta $l$ passa per un punto compreso tra $A$ e $B$, allora passa anche per un punto compreso tra $A$ e $C$ oppure tra $B$ e $C$, ma non per entrambi.}
\end{axiom}

\pol{Powyższy aksjomat nazywany jest też aksjomatem Pascha, ponieważ Moritz Pasch \cite{pasch_1882} przyłapał dopiero w 1882 roku geometrów całego świata na tym, że korzystali z takiej przesłanki.}
\ita{L'assioma precedente è chiamato anche assioma di Pasch, poiché soltanto nel 1882 Moritz Pasch \cite{pasch_1882} fece notare che i geometri di tutto il mondo stavano già utilizzando implicitamente un'ipotesi di questo tipo.}

\pol{\index{aksjomat!Pascha}}
\ita{\index{assioma!di Pasch}}

\index[persons]{Pasch, Moritz}%

\pol{Aksjomatu Pascha (B4) nie wolno mylić z aksjomatem Veblena-Younga geometrii rzutowej\footnote{Jeśli prosta przecina dwa boki trójkąta, to przecina także jego trzeci bok. \index{aksjomat!Veblena-Younga}}!}
\ita{L'assioma di Pasch (B4) non va confuso con l'assioma di Veblen--Young della geometria proiettiva\footnote{Se una retta interseca due lati di un triangolo, allora interseca anche il terzo lato. \index{assioma!di Veblen--Young}}!}

\begin{proposition}
    \pol{Z aksojmatów I1 do I3, B1 do B4 wynika, że każda prosta ma nieskończenie wiele punktów.}
    \ita{Dagli assiomi I1--I3 e B1--B4 segue che ogni retta contiene infiniti punti.}
\end{proposition}

\begin{definition}[\pol{odcinek}\ita{segmento}]
    \pol{Niech $A$, $B$ będą punktami.}
    \ita{Siano $A$ e $B$ due punti.}

    \pol{Zbiór złożony z punktów $A$, $B$ oraz punktów, które leżą między nimi, nazywamy odcinkiem i oznaczamy:}
    \ita{L'insieme formato dai punti $A$, $B$ e da tutti i punti che si trovano tra essi si chiama segmento e si indica con:}

\pol{\index{odcinek}}
\ita{\index{segmento}}

    \begin{equation}
        \overline{AB}.
    \end{equation}
\end{definition}

\pol{Jeżeli nie prowadzi to do nieporozumień, kreskę pomijamy.}
\ita{Quando non vi è rischio di ambiguità, la barra viene omessa.}

\begin{definition}[\pol{trójkąt}\ita{triangolo}]
    \pol{Niech $A$, $B$, $C$ będą punktami.}
    \ita{Siano $A$, $B$ e $C$ tre punti.}

    \pol{Sumę odcinków $AB$, $BC$, $AC$ nazywamy trójkątem, wspomniane odcinki -- jego bokami, zaś punkty $A$, $B$ i $C$ -- wierzchołkami.}
    \ita{L'unione dei segmenti $AB$, $BC$ e $AC$ si chiama triangolo; tali segmenti sono i suoi lati e i punti $A$, $B$ e $C$ i suoi vertici.}

\pol{\index{trójkąt}}
\ita{\index{triangolo}}
\end{definition} % Hartshorne 74

\pol{Trójkąty zazwyczaj oznaczamy tak: $\triangle ABC$ i nie odróżniamy go od trójkąta z wnętrzem.}
\ita{Di solito indichiamo un triangolo con $\triangle ABC$ e non distinguiamo il bordo del triangolo dalla figura comprensiva del suo interno.}

\pol{Mamy podział na}
\ita{Si distinguono}
\begin{itemize}
    \item \pol{trójkąty ostrokątne (z trzema kątami ostrymi),}
          \ita{triangoli acutangoli (con tre angoli acuti),}
    \item \pol{prostokątne (z jednym kątem prostym),}
          \ita{rettangoli (con un angolo retto),}
    \item \pol{rozwartokątne (z jednym kątem większym niż prosty),}
          \ita{ottusangoli (con un angolo ottuso),}
\end{itemize}

\pol{ale też na:}
\ita{e inoltre}

\begin{itemize}
\item \pol{równoboczne (z trzema bokami, które są przystające),}
      \ita{equilateri (con tre lati congruenti),}
\item \pol{równoramienne (z dwoma przystającymi bokami zwanymi ramionami, wtedy trzeci bok nazywa się podstawą) albo}
      \ita{isosceli (con due lati congruenti detti lati obliqui; il terzo lato è chiamato base) oppure}
\item \pol{różnoboczne, z angielskiego \emph{scalene}.}
      \ita{scaleni.}
\end{itemize}

\begin{proposition}
    \pol{Niech $l$ będzie prostą.}
    \ita{Sia $l$ una retta.}

    \pol{Wtedy zbiór punktów, które nie leżą na prostej $l$ można rozbić na dwa niepuste zbiory $S_1$, $S_2$ takie, że: dwa punkty, które nie leżą na prostej $l$, należą do tego samego zbioru ($S_1$ lub $S_2$) wtedy i tylko wtedy, gdy odcinek $AB$ nie przecina prostej $l$.}
    \ita{Allora l'insieme dei punti che non appartengono alla retta $l$ può essere suddiviso in due insiemi non vuoti $S_1$, $S_2$ tali che due punti esterni a $l$ appartengano allo stesso insieme ($S_1$ oppure $S_2$) se e solo se il segmento $AB$ non interseca la retta $l$.}
\end{proposition} % Hartshorne 74

\pol{Zbiory $S_1$, $S_2$ nazywamy stronami prostej $l$.}
\ita{Gli insiemi $S_1$ e $S_2$ sono detti semipiani determinati dalla retta $l$.}

\pol{\index{prosta!dwie strony prostej}}
\ita{\index{retta!due lati di una retta}}

\pol{Podobnie punkt wyznacza na prostej dwa zbiory, które leżą po różnych stronach tego punktu.}
\ita{Analogamente, un punto determina sulla retta due insiemi situati da parti opposte rispetto a quel punto.}

\begin{definition}[\pol{półprosta}\ita{semiretta}]
    \pol{Niech $A$, $B$ będą punktami.}
    \ita{Siano $A$ e $B$ due punti.}

    \pol{Zbiór złożony z punktów $A$, $B$ oraz punktów, które leżą po tej samej stronie punktu $A$ na prostej $AB$ co punkt $B$, nazywamy półprostą nie ma sensownego oznaczenia}
    \ita{L'insieme formato dai punti $A$, $B$ e da tutti i punti che si trovano sulla retta $AB$ dalla stessa parte di $A$ in cui si trova $B$ è chiamato semiretta; non esiste una notazione universalmente soddisfacente}

\pol{\index{półprosta}}
\ita{\index{semiretta}}

    % \begin{equation}
        % \overrightarrow{AB}. % AB^{\rightarrow} ?
    % \end{equation}
\end{definition} % Hartshorne 77

\begin{definition}[\pol{kąt}\ita{angolo}]
    \pol{Sumę dwóch półprostych $AB$, $AC$, które nie leżą na jednej prostej, nazywamy kątem, zaś punkt $A$ wierzchołkiem tego kąta.}
    \ita{L'unione di due semirette $AB$ e $AC$ che non giacciono sulla stessa retta si chiama angolo; il punto $A$ è il vertice dell'angolo.}

    \pol{Wnętrze kąta $\angle BACS$ składa się z tych punktów $D$ takich, że $D$ i $C$ leżą po tej samej stronie prostej $AB$ oraz $D$ i $B$ leżą po tej samej stronie prostej $AC$.}
    \ita{L'interno dell'angolo $\angle BACS$ è costituito dai punti $D$ tali che $D$ e $C$ si trovino dalla stessa parte della retta $AB$ e che $D$ e $B$ si trovino dalla stessa parte della retta $AC$.}

    \pol{\index{kąt}}
    \ita{\index{angolo}}
\end{definition} % Hartshorne 77

\pol{W myśl tej definicji, nie ma kąta zerowego, wklęsłego ani półpełnego.}
\ita{Secondo questa definizione non esistono angoli nulli, concavi né piatti.}

\pol{\index{kąt!zerowy}}
\ita{\index{angolo!nullo}}

\pol{\index{kąt!półpełny}}
\ita{\index{angolo!piatto}}

\pol{\index{kąt!wklęsły}}
\ita{\index{angolo!concavo}}

\pol{Wnętrze trójkąta $ABC$ to część wspólna wnętrz kątów $\angle ABC$, $\angle BCA$, $\angle CAB$; jest wypukłe (jeśli dwa punkty końce odcinka należą do wnętrza, to również każdy inny punkt tego odcinka znajduje się tam) i niepuste.}
\ita{L'interno del triangolo $ABC$ è l'intersezione degli interni degli angoli $\angle ABC$, $\angle BCA$ e $\angle CAB$; esso è convesso (se gli estremi di un segmento appartengono all'interno, allora vi appartiene anche ogni punto del segmento) e non vuoto.}
