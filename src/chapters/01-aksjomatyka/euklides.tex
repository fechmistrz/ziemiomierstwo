\ita{\section{Assiomi di Euclide}}
\pol{\index{aksjomat!Euklidesa}}%
\ita{\index{assioma!di Euclide}}%

\todofoot{https://kpbc.umk.pl/dlibra/publication/37/edition/66/content}

\pol{Opowiemy krótko o rozwoju geometrii w okresie klasycznym starożytnej Grecji, pomijając przy tym liczne szczegóły, ponieważ światu przekaże je lepiej Proklos zwany Diadochem, operając się na wcześniejszej pracy Eudemosa z Rodos, obejmującej okres do 355 roku przed Chrystusem.}
\ita{Racconteremo brevemente lo sviluppo della geometria nella Grecia classica, tralasciando molti dettagli, poiché Proclo detto Diadoco lo descriverà meglio, basandosi sull'opera precedente di Eudemo di Rodi, che copriva il periodo fino al 355 a.C.}
\index[persons]{Proklos zwany Diadochem}%
\index[persons]{Eudemos z Rodos}%
\pol{Później przytoczymy definicje i twierdzenia zgromadzone przez Euklidesa z Aleksandrii.}
\ita{Successivamente presenteremo le definizioni e i teoremi raccolti da Euclide di Alessandria.}
\index[persons]{Euklides z Aleksandrii}%
\pol{(Imię Euklidesa znamy z tylko jednego źródła: komentarza Proklosa! Pozostali nazywali go autorem Στοιχεῖα, znaczy \emph{Stoicheia}, znaczy \emph{Elementów}).}
\ita{(Conosciamo il nome di Euclide da una sola fonte: il commento di Proclo! Gli altri lo chiamavano semplicemente autore degli Στοιχεῖα, cioè degli \emph{Elementi}).}
\pol{Interesować nas będzie nie wszystkie trzynaście ksiąg, a:}
\ita{Non ci interesseranno tutti e tredici i libri, ma:}
\begin{itemize}
    \item \pol{I~(o~trójkątach),}\ita{I (sui triangoli),}
    \item \pol{II (o~prostokątach),}\ita{II (sui rettangoli),}
    \item \pol{III (o~okręgach),}\ita{III (sulle circonferenze),}
    \item \pol{IV (o~wielokątach),}\ita{IV (sui poligoni),}
    \item \pol{V (o~proporcjach),}\ita{V (sulle proporzioni),}
    \item \pol{VI (o~podobieństwie), ale także, choć w~mniejszym stopniu,}\ita{VI (sulla similitudine), ma anche, sebbene in misura minore,}
    \item \pol{XI i~XIII (o~wielościanach).}\ita{XI e XIII (sui poliedri).}
\end{itemize}

\pol{Według Eudemosa/Proklosa, początków geometrii greckiej należy szukać w dokonaniach Talesa z Miletu, a więc mniej więcej w VI wieku p.n.e.}
\ita{Secondo Eudemo e Proclo, le origini della geometria greca vanno ricercate nelle opere di Talete di Mileto, cioè all'incirca nel VI secolo a.C.}
\pol{Tales zacznie stosować metody dedukcyjne.}
\ita{Talete iniziò a utilizzare metodi deduttivi.}
\index[persons]{Tales z Miletu}%
\pol{Potem będzie na pewno pobierający (a być może u Talesa) nauki Pitagoras i jego szkoła, której uczniowie zaczną tworzyć łańcuchy różnych geometrycznych faktów oraz wniosków z nich.}
\ita{Successivamente vi saranno certamente Pitagora e la sua scuola, i cui allievi inizieranno a costruire catene di fatti geometrici e delle loro conseguenze.}
\index[persons]{Pitagoras}%
\pol{Hipokrates z Chios postara się uporządkować twierdzenia i częściowo uda mu się to.}
\ita{Ippocrate di Chio cercherà di organizzare i teoremi e vi riuscirà in parte.}
\index[persons]{Hipokrates z Chios}%
\pol{Podobne próby podejmą Teudios z Magnezji oraz Leon (hehe); ale nie wiadomo za dużo ani o nich, ani o ich dziełach poza tym, że miały tytuł \emph{Elementy}.}
\ita{Tentativi analoghi saranno compiuti da Teudio di Magnesia e da Leone (hehe); tuttavia sappiamo ben poco sia di loro sia delle loro opere, se non che portavano il titolo di \emph{Elementi}.}
\index[persons]{Teudios z Magnezji}%
\index[persons]{Leon, uczeń Neokleidesa}%

\pol{Proces przekształcania empirycznej geometrii Egipcjan i Babilończyków w systematyczną naukę, której fakty ustala się nie przez obserwacje, ale na drodze rozumowania dedukcyjnego, będzie mieć swoje ukoronowanie około 300 roku p.n.e.}
\ita{Il processo di trasformazione della geometria empirica degli Egizi e dei Babilonesi in una scienza sistematica, nella quale i fatti vengono stabiliti non tramite osservazioni ma attraverso il ragionamento deduttivo, raggiungerà il suo culmine intorno al 300 a.C.}
\pol{Wtedy wielki sukces odniesie Euklides ze swoim epokowym dziełem -- także \emph{Elementami}, spójnym ciągiem 465 twierdzeń dotyczących geometrii płaskiej, przestrzennej, teorii liczb oraz częściowo algebry.}
\ita{Fu allora che Euclide ottenne un grande successo con la sua opera epocale, anch'essa intitolata \emph{Elementi}, una raccolta coerente di 465 teoremi riguardanti la geometria piana, la geometria dello spazio, la teoria dei numeri e, in parte, l'algebra.}
\index[persons]{Euklides z Aleksandrii}%
\pol{Grek formułuje tam pewien minimalny zbiór aksjomatów (czyli zdań przyjętych bez dowodu), z którego sukcesywnie wyprowadza co raz bardziej skomplikowane twierdzenia.}
\ita{Il matematico greco vi formula un insieme minimo di assiomi (cioè enunciati accettati senza dimostrazione), dal quale deduce progressivamente teoremi sempre più complessi.}
\pol{Pogląd, że tak powinno uprawiać się naukę, utrzyma się przez tysiące lat, jeśli nie wieczność; jego książka zaś ukształtuje wykład geometrii aż do początków XIX wieku.}
\ita{L'idea che la scienza debba essere praticata in questo modo sopravvivrà per migliaia di anni, se non per l'eternità; il suo libro modellerà invece l'insegnamento della geometria fino agli inizi del XIX secolo.}
\pol{Imponujące!}
\ita{Impressionante!}
