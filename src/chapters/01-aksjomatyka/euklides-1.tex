\pol{\subsubsection{Postulaty}}
\ita{\subsubsection{Postulati}}
\begin{enumerate}
    \item [1.1] \pol{Przez każde dwa punkty przechodzi prosta.}\ita{Per ogni coppia di punti passa una retta.}
    \item [1.2] \pol{Każdy odcinek można przedłużyć do prostej.}\ita{Ogni segmento può essere prolungato in una retta.}
    \item [1.3] \pol{Z każdego środka można wykreślić okrąg o dowolnym promieniu.}\ita{Con qualsiasi centro si può tracciare una circonferenza di raggio arbitrario.}
    \item [1.4] \pol{Wszystkie kąty proste są sobie równe.}\ita{Tutti gli angoli retti sono uguali tra loro.}
    \item [1.5] \pol{Jeżeli prosta przecinająca dwie proste tworzy z nimi kąty jednostronnie wewnętrzne o sumie mniejszej niż $\pi$, to te dwie proste przecinają się po tej stronie.}\ita{Se una retta che interseca due rette forma con esse angoli interni dallo stesso lato la cui somma è minore di $\pi$, allora le due rette si incontrano da quel lato.}
\end{enumerate}

\pol{Jak łatwo zauważyć, sformułowanie ostatniego postulatu używa więcej słów niż pozostałe razem wzięte; wbrew przekonaniu, że postulaty miały wyrażać treści oczywiste i proste.}
\ita{Come si può facilmente osservare, l'enunciato dell'ultimo postulato utilizza più parole di tutti gli altri messi insieme, contrariamente all'idea che i postulati dovessero esprimere fatti semplici ed evidenti.}
\pol{Piąty postulat wydawał się bardziej skomplikowany, więc nasuwał podejrzenie, że wynika z poprzednich czterech.}
\ita{Il quinto postulato sembrava più complicato e suggeriva quindi il sospetto che fosse una conseguenza dei quattro precedenti.}
\pol{Zauważył to już Proklos zwany Diadochem (410-485):}
\ita{Lo osservò già Proclo detto Diadoco (410--485):}
\index[persons]{Proklos zwany Diadochem}%
\pol{\emph{,,Nie jest możliwe, aby uczony tej miary co Euklides godził się na obecność tak długiego postulatu w aksjomatyce -- obecność postulatu wzięła się z pospiesznego kończenia przez niego Elementów, tak aby zdążyć przed nadejściem słusznie oczekiwanej rychłej śmierci; my zatem -- czcząc jego pamięć -- powinniśmy ten postulat usunąć lub co najmniej znacznie uprościć.''}}
\ita{\emph{«Non è possibile che uno studioso della statura di Euclide accettasse la presenza di un postulato tanto lungo in un sistema assiomatico: esso vi comparve soltanto perché stava terminando in fretta gli Elementi prima dell'attesa e imminente morte; noi dunque, onorandone la memoria, dovremmo eliminare questo postulato o almeno semplificarlo notevolmente.»}}

\pol{Wiele osób będzie bezskutecznie stawiać czoło wyzwaniu postawionemu przez Proklosa.}
\ita{Molti tenteranno invano di raccogliere la sfida lanciata da Proclo.}
\pol{Powód niepowodzeń będzie prozaiczny: piąty postulat jest niezależny od pozostałych!}
\ita{La ragione dei loro insuccessi sarà semplice: il quinto postulato è indipendente dagli altri!}

\pol{\subsubsection{Twierdzenia}}
\ita{\subsubsection{Proposizioni}}

    \item [1.1] \pol{Konstrukcja trójkąta równobocznego.}\ita{Costruzione di un triangolo equilatero.}
    \item [1.2] \pol{Konstrukcja odcinka (w innym miejscu).}\ita{Costruzione di un segmento (in un'altra posizione).}
    \item [1.3] \pol{Konstrukcja różnicy odcinków.}\ita{Costruzione della differenza di due segmenti.}
    \item [1.4] \pol{Cecha przystawania bok-kąt-bok.}\ita{Criterio di congruenza lato-angolo-lato.}
    \item [1.5] \pol{Kąty przy podstawie trójkąta równoramiennego są równe.}\ita{Gli angoli alla base di un triangolo isoscele sono uguali.}
    \item [1.6] \pol{Twierdzenie odwrotne: boki trójkąta leżące naprzeciw równych kątów są równe.}\ita{Teorema inverso: i lati opposti ad angoli uguali in un triangolo sono uguali.}
    \item [1.7] \pol{Boki trójkąta jednoznacznie go wyznaczają.}\ita{I lati di un triangolo lo determinano univocamente.}
    \item [1.8] \pol{Cecha przystawania bok-bok-bok.}\ita{Criterio di congruenza lato-lato-lato.}
    \item [1.9] \pol{Konstrukcja połowy kąta.}\ita{Costruzione della bisettrice di un angolo.}
    \item [1.10] \pol{Konstrukcja połowy odcinka.}\ita{Costruzione del punto medio di un segmento.}
    \item [1.11] \pol{Konstrukcja prostopadłej przez punkt na prostej.}\ita{Costruzione della perpendicolare per un punto appartenente alla retta.}
    \item [1.12] \pol{Konstrukcja prostopadłej przez punkt poza prostą.}\ita{Costruzione della perpendicolare per un punto esterno alla retta.}
    \item [1.13] \pol{Kąty przyległe mają łączną miarę $\pi$.}\ita{Gli angoli adiacenti hanno somma pari a $\pi$.}
    \item [1.14] \pol{Twierdzenie odwrotne: kąty o wspólnym ramieniu (i różnych drugich ramionach), których łączna miara to $\pi$, są przyległe.}\ita{Teorema inverso: due angoli con un lato in comune (e gli altri due distinti), la cui somma è $\pi$, sono adiacenti.}
    \item [1.15] \pol{Kąty wierzchołkowe mają równe miary.}\ita{Gli angoli opposti al vertice hanno la stessa ampiezza.}
    \item [1.16] \pol{W trójkącie o kątach $\alpha, \beta, \gamma$, kąt przyległy do $\alpha$ jest większy od $\beta$ i większy od $\gamma$.}\ita{In un triangolo con angoli $\alpha$, $\beta$, $\gamma$, l'angolo esterno adiacente ad $\alpha$ è maggiore di $\beta$ e di $\gamma$.}
    \item [1.17] \pol{Suma dwóch kątów trójkąta jest mniejsza od $\pi$.}\ita{La somma di due angoli di un triangolo è minore di $\pi$.}
    \item [1.18] \pol{Większy bok trójkąta leży naprzeciw większego kąta.}\ita{Al lato maggiore di un triangolo si oppone l'angolo maggiore.}
    \item [1.19] \pol{Większy kąt trójkąta leży naprzeciw większego boku.}\ita{All'angolo maggiore di un triangolo si oppone il lato maggiore.}
    \item [1.20] \pol{Nierówność trójkąta.}\ita{Disuguaglianza triangolare.}
    \item [1.21] \pol{W trójkącie $ABC$ z punktem $D$ wewnątrz, $BD + CD < AB + AC$.}\ita{In un triangolo $ABC$, se $D$ è un punto interno, allora $BD + CD < AB + AC$.}
    \item [1.22] \pol{Twierdzenie odwrotne do nierówności trójkąta: jeśli suma dwóch krótszych odcinków jest dłużsa niż trzeci, to można z nich zbudować trójkąt.}\ita{Teorema inverso della disuguaglianza triangolare: se la somma di due segmenti è maggiore del terzo, essi possono formare un triangolo.}
    \item [1.23] \pol{Konstrukcja kąta (w innym miejscu).}\ita{Costruzione di un angolo (in un'altra posizione).}
    \item [1.24] \pol{Wariacja 1.18: jeśli $AB = DE$, $AC = DF$, $\angle BAC > \angle EDF$, to $BC > EF$.}\ita{Variazione di 1.18: se $AB = DE$, $AC = DF$, $\angle BAC > \angle EDF$, allora $BC > EF$.}
    \item [1.25] \pol{Wariacja 1.19: jeśli $AB = DE$, $AC = DF$, $BC > EF$, to $\angle BAC > \angle EDF$.}\ita{Variazione di 1.19: se $AB = DE$, $AC = DF$, $BC > EF$, allora $\angle BAC > \angle EDF$.}
    \item [1.26] \pol{Cecha przystawania kąt-bok-kąt.}\ita{Criterio di congruenza angolo-lato-angolo.}
    \item [1.27] \pol{Dwie proste przecięte trzecią pod tym samym kątem (naprzemianległym) są równoległe.}\ita{Due rette tagliate da una terza che formano angoli alterni interni uguali sono parallele.}
    \item [1.28] \pol{Dwie proste przecięte trzecią pod tym samym kątem (odpowiadającym) są równoległe.}\ita{Due rette tagliate da una terza che formano angoli corrispondenti uguali sono parallele.}
    \item [1.29] \pol{Kąty naprzemianległe są równe.}\ita{Gli angoli alterni interni sono uguali.}
    \item [1.30] \pol{Równoległość prostych jest przechodnia.}\ita{Il parallelismo delle rette è transitivo.}
    \item [1.31] \pol{Konstrukcja prostej równoległej przez punkt.}\ita{Costruzione di una parallela passante per un punto dato.}
    \item [1.32] \pol{Kąt zewnętrzny przy wierzchołku trójkąta jest sumą pozostałych dwóch kątów wewnętrznych.}\ita{L'angolo esterno di un triangolo è uguale alla somma degli altri due angoli interni.}
    \item [1.33] \pol{Jeśli odcinki $AB$, $CD$ są równe i równoległe, to odcinki $AC$ i $BD$ też są równe i równoległe.}\ita{Se i segmenti $AB$ e $CD$ sono uguali e paralleli, allora anche $AC$ e $BD$ sono uguali e paralleli.}
    \item [1.34] \pol{Przekątna równoległoboku dzieli go na dwa przystające trójkąty.}\ita{La diagonale di un parallelogramma lo divide in due triangoli congruenti.}
    \item [1.35] \pol{Dwa równoległoboki, których drugie podstawy leżą na tej samej prostej mają równe pola.}\ita{Due parallelogrammi aventi la stessa base e compresi tra le stesse parallele hanno la stessa area.}
    \item [1.36] \pol{Dwa równoległoboki, które mają równe podstawy i równe wysokości mają równe pola.}\ita{Due parallelogrammi con basi uguali e altezze uguali hanno la stessa area.}
    \item [1.37] \pol{Dwa trójkąty, które mają tę samą podstawę i równe wysokości mają równe pola.}\ita{Due triangoli con la stessa base e la stessa altezza hanno la stessa area.}
    \item [1.38] \pol{Dwa trójkąty, które mają równe podstawy i równe wysokości mają równe pola.}\ita{Due triangoli con basi uguali e altezze uguali hanno la stessa area.}
    \item [1.39] \pol{Jeśli trójkąty $ABC$, $ABD$ mają równe pola, a $C, D$ leżą po jednej stronie $AB$, to $AB \parallel CD$.}\ita{Se i triangoli $ABC$ e $ABD$ hanno la stessa area e $C$, $D$ si trovano dalla stessa parte di $AB$, allora $AB \parallel CD$.}
    \item [1.40] \pol{(dziwne)}\ita{(strano)}
    \item [1.41] \pol{Równoległobok ma dwa razy większe pole niż trójkąt o tej samej podstawie i równej wysokości.}\ita{Un parallelogramma ha area doppia rispetto a un triangolo con la stessa base e la stessa altezza.}
    \item [1.42] \pol{Konstrukcja równoległoboku o polu jak dany trójkąt i danym kącie wewnętrznym.}\ita{Costruzione di un parallelogramma con area uguale a quella di un triangolo dato e con un dato angolo interno.}
    \item [1.43] \pol{W równoległoboku $ABCD$ wybrano punkt $P$ na przekątnej $AC$ i poprowadzono przez niego proste równoległe do boków, przecinające boki $AB$, $BC$, $CD$, $AD$ w punktach $H$, $F$, $G$, $I$. Wtedy równoległoboki $HBFP$ i $IPGD$ mają równe pola. (To jest znane po angielsku jako \emph{theorem of the gnomon}; Euklides wprowadzi termin gnomon w drugiej definicji drugiej księgi).}
    \ita{In un parallelogramma $ABCD$ si scelga un punto $P$ sulla diagonale $AC$ e si traccino per esso rette parallele ai lati, che incontrano $AB$, $BC$, $CD$, $AD$ rispettivamente nei punti $H$, $F$, $G$, $I$. Allora i parallelogrammi $HBFP$ e $IPGD$ hanno la stessa area. (Questo risultato è noto in inglese come \emph{theorem of the gnomon}; Euclide introduce il termine gnomone nella seconda definizione del secondo libro).}
    \item [1.44] \pol{Konstrukcja równoległoboku o danej podstawie, danym kącie wewnętrznym i polu równym polu danego trójkąta.}\ita{Costruzione di un parallelogramma con base assegnata, angolo interno assegnato e area uguale a quella di un triangolo dato.}
    \item [1.45] \pol{Konstrukcja równoległoboku o danej podstawie, danym kącie wewnętrznym i polu równym polu danego czworokąta.}\ita{Costruzione di un parallelogramma con base assegnata, angolo interno assegnato e area uguale a quella di un quadrilatero dato.}
