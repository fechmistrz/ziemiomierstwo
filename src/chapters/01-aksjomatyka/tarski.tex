\pol{\section{Aksjomaty Tarskiego}}
\ita{\section{Assiomi di Tarski}}
\pol{\index{aksjomat!Tarskiego}}%
\ita{\index{assioma!di Tarski}}%

\pol{Ludzie będą starać się manipulować pojęciami pierwotnymi, by dostać w jakimś sensie prostszą aksjomatykę.}
\ita{I matematici cercheranno di modificare i concetti primitivi per ottenere, in un certo senso, un'assiomatica più semplice.}
\pol{Jedną z popularniejszych alternatyw do pomysłów Hilberta zaproponuje Alfred Tarski około 1926 roku (a później ulepszą ją: Wanda Szmielew, Wolfram Schwabhäuser i inni).}
\ita{Una delle alternative più popolari alle idee di Hilbert sarà proposta da Alfred Tarski intorno al 1926 (e successivamente migliorata da Wanda Szmielew, Wolfram Schwabhäuser e altri).}
\pol{Opisujemy jego pomysły tutaj.}
\ita{Descriviamo qui le sue idee.}

\pol{O aksjomatach Tarskiego napisze Marek Kordos w artykule ,,Kompletne, beznadziejne manowce''.}
\ita{Sugli assiomi di Tarski scriverà Marek Kordos nell'articolo ,,Kompletne, beznadziejne manowce''.}

\pol{W aksjomatyce Tarskiego mamy zmienne (punkty) i dwie relacje $\beta$, $\equiv$.}
\ita{Nell'assiomatica di Tarski abbiamo variabili (punti) e due relazioni $\beta$, $\equiv$.}
\pol{Relacja $\beta (x, y, z)$ mówi, że trzy punkty leżą na jednej prostej w kolejności takiej, jak w napisie: $y$ między $x$ oraz $z$; zaś relacja $\equiv$ orzeka, że dwa odcinki są przystające.}
\ita{La relazione $\beta (x, y, z)$ afferma che tre punti giacciono sulla stessa retta nell'ordine indicato dalla scrittura: $y$ è tra $x$ e $z$; la relazione $\equiv$ invece afferma che due segmenti sono congruenti.}
\begin{enumerate}
    \item \pol{jeśli $\beta (x, y, x)$, to $x = y$;}
          \ita{se $\beta (x, y, x)$, allora $x = y$;}
    \item \pol{jeśli $\beta (x, y, u)$ oraz $\beta (y, z, u)$,  to $\beta (x, z, u)$;}
          \ita{se $\beta (x, y, u)$ e $\beta (y, z, u)$, allora $\beta (x, z, u)$;}
    \item \pol{jeśli $\beta (x, y, z)$, $\beta (x, y, u)$ oraz $x \neq y$, to $\beta (x, z, u)$ lub $\beta (x, u, z)$;}
          \ita{se $\beta (x, y, z)$, $\beta (x, y, u)$ e $x \neq y$, allora $\beta (x, z, u)$ oppure $\beta (x, u, z)$;}
    \item \pol{$xy \equiv yx$;}
          \ita{$xy \equiv yx$;}
    \item \pol{jeśli $xy \equiv zu$ oraz $xy \equiv vw$, to $zu \equiv vw$;}
          \ita{se $xy \equiv zu$ e $xy \equiv vw$, allora $zu \equiv vw$;}
    \item \pol{jeśli $xy \equiv zz$, to $x = y$;}
          \ita{se $xy \equiv zz$, allora $x = y$;}
    \item \pol{aksjomat Pascha: jeśli $\beta (x, t, u)$, $\beta (y, u, z)$, to istnieje taki $v$, że $\beta (x, v, y)$, $\beta(z, t, v)$'';}
          \ita{assioma di Pasch: se $\beta (x, t, u)$, $\beta (y, u, z)$, allora esiste un punto $v$ tale che $\beta (x, v, y)$, $\beta(z, t, v)$'';}
          \pol{\index{aksjomat Pascha}}%
          \ita{\index{assioma di Pasch}}%
    \item \pol{V aksjomat Euklidesa\footnote{Ten aksjomat można zastąpić przez ,,na każdym trójkącie można opisać okrąg'' (mniej zmiennych) albo ,,linia środkowa w trójkącie jest dwukrotnie krótsza od podstawy'' (brak kwantyfikatorów).} (dla każdego kąta i punktu wewnątrz istnieje odcinek przez ten punkt, który przecina ramiona kąta): jeśli $\beta(x, u, t)$, $\beta (y, u, z)$, $x \neq y$, to istnieją takie $v, w$, że $\beta(x, z, v)$, $\beta(x, y, w)$ oraz $\beta (v, t, w)$;}
          \ita{V postulato di Euclide\footnote{Questo assioma può essere sostituito da ,,su ogni triangolo si può costruire un cerchio circoscritto'' (meno variabili) oppure da ,,il segmento medio di un triangolo è lungo la metà della base'' (senza quantificatori).} (per ogni angolo e ogni punto interno esiste un segmento passante per quel punto che interseca i lati dell'angolo): se $\beta(x, u, t)$, $\beta (y, u, z)$, $x \neq y$, allora esistono $v, w$ tali che $\beta(x, z, v)$, $\beta(x, y, w)$ e $\beta (v, t, w)$;}
          \pol{\index{V postulat}}
          \ita{\index{V postulato}}
    \item \pol{aksjomat pięciu odcinków (cecha przystawania bkb): jeśli $xy \equiv x'y'$, $yz \equiv y'z'$, $xu \equiv x'u'$, $yu \equiv y'u'$, $\beta (x, y, z)$, $\beta(x', y', z')$ i $x \neq y$, to $zu \equiv z'u'$;}
          \ita{assioma dei cinque segmenti (criterio di congruenza lato-angolo-lato): se $xy \equiv x'y'$, $yz \equiv y'z'$, $xu \equiv x'u'$, $yu \equiv y'u'$, $\beta (x, y, z)$, $\beta(x', y', z')$ e $x \neq y$, allora $zu \equiv z'u'$;}
          \pol{\index{cecha przystawania bkb}}%
          \ita{\index{criterio di congruenza!lato-angolo-lato}}%
    \item \pol{odkładanie odcinków: istnieje $z$ taki, że $\beta (x, y, z)$ i $yz \equiv uw$;}
          \ita{trasporto dei segmenti: esiste un punto $z$ tale che $\beta (x, y, z)$ e $yz \equiv uw$;}
    \item \pol{istnieją trzy niewspółliniowe punkty\footnote{Bez tego aksjomatu teoria ma model na prostej rzeczywistej.} $x, y, z$: $\neg \beta(x, y, z)$, $\neg \beta(y, z, x)$, $\neg \beta(z, x, y, )$;}
          \ita{esistono tre punti non allineati\footnote{Senza questo assioma la teoria ammette un modello sulla retta reale.} $x, y, z$: $\neg \beta(x, y, z)$, $\neg \beta(y, z, x)$, $\neg \beta(z, x, y, )$;}
    \item \pol{trzy punkty równoodległe od końców odcinka tworzą prostą\footnote{Bez tego aksjomatu teoria ma model w trzech i więcej wymiarach.}: jeśli $xu \equiv xv$, $yu \equiv yv$, $zu \equiv zv$ i $u \neq v$, to $\beta(x, y, z)$, $\beta(y, z, x)$ lub $\beta(z, x, y)$;}
          \ita{tre punti equidistanti dagli estremi di un segmento giacciono su una retta\footnote{Senza questo assioma la teoria ammette modelli in tre o più dimensioni.}: se $xu \equiv xv$, $yu \equiv yv$, $zu \equiv zv$ e $u \neq v$, allora $\beta(x, y, z)$, $\beta(y, z, x)$ oppure $\beta(z, x, y)$;}
    \item \pol{schemat aksjomatu ciągłości.}
          \ita{schema dell'assioma di continuità.}
\end{enumerate}

\pol{Układ aksjomatów Tarskiego pozwoli mu pokazać, że teoria geometrii euklidesowej (pierwszego rzędu) jest niesprzeczna,}
\ita{Il sistema degli assiomi di Tarski gli permetterà di dimostrare che la teoria della geometria euclidea (del primo ordine) è coerente,}
\pol{zupełna i}
\ita{completa e}
\pol{rozstrzygalna:}
\ita{decidibile:}
\pol{każde zdanie w jej języku jest albo dowodliwe, albo obalalne na podstawie aksjomatów, a ponadto istnieje algorytm, który dla dowolnie danego zdania rozstrzyga, czy jest ono dowodliwe, czy nie.}
\ita{ogni enunciato del suo linguaggio è dimostrabile oppure confutabile a partire dagli assiomi, e inoltre esiste un algoritmo che, per ogni enunciato dato, decide se esso sia dimostrabile oppure no.}
