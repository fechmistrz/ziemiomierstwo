W aksjomatyce Tarskiego mamy zmienne (punkty) i dwie relacje $\beta$, $\equiv$. % lang-pl
Relacja $\beta (x, y, z)$ mówi, że trzy punkty leżą na jednej prostej w kolejności takiej, jak w napisie: $y$ między $x$ oraz $z$; zaś relacja $\equiv$ orzeka, że dwa odcinki są przystające. % lang-pl
Nell'assiomatica di Tarski abbiamo variabili (punti) e due relazioni $\beta$, $\equiv$. % lang-it
La relazione $\beta (x, y, z)$ afferma che tre punti giacciono sulla stessa retta nell'ordine indicato dalla scrittura: $y$ è tra $x$ e $z$; la relazione $\equiv$ invece afferma che due segmenti sono congruenti. % lang-it
\begin{enumerate}
      \item jeśli $\beta (x, y, x)$, to $x = y$; % lang-pl
      \item se $\beta (x, y, x)$, allora $x = y$; % lang-it
      
      \item jeśli $\beta (x, y, u)$ oraz $\beta (y, z, u)$,  to $\beta (x, z, u)$; % lang-pl
      \item se $\beta (x, y, u)$ e $\beta (y, z, u)$, allora $\beta (x, z, u)$; % lang-it
      
      \item jeśli $\beta (x, y, z)$, $\beta (x, y, u)$ oraz $x \neq y$, to $\beta (x, z, u)$ lub $\beta (x, u, z)$; % lang-pl
      \item se $\beta (x, y, z)$, $\beta (x, y, u)$ e $x \neq y$, allora $\beta (x, z, u)$ oppure $\beta (x, u, z)$; % lang-it
      
      \item $xy \equiv yx$; % lang-pl
      \item $xy \equiv yx$; % lang-it

      \item jeśli $xy \equiv zu$ oraz $xy \equiv vw$, to $zu \equiv vw$; % lang-pl
      \item se $xy \equiv zu$ e $xy \equiv vw$, allora $zu \equiv vw$; % lang-it

      \item jeśli $xy \equiv zz$, to $x = y$; % lang-pl
      \item se $xy \equiv zz$, allora $x = y$; % lang-it

      \item aksjomat Pascha: jeśli $\beta (x, t, u)$, $\beta (y, u, z)$, to istnieje taki $v$, że $\beta (x, v, y)$, $\beta(z, t, v)$''; % lang-pl
      \item assioma di Pasch: se $\beta (x, t, u)$, $\beta (y, u, z)$, allora esiste un punto $v$ tale che $\beta (x, v, y)$, $\beta(z, t, v)$''; % lang-it
\index{aksjomat Pascha}% % lang-pl
\index{assioma di Pasch}% % lang-it

      \item V aksjomat Euklidesa\footnote{Ten aksjomat można zastąpić przez ,,na każdym trójkącie można opisać okrąg'' (mniej zmiennych) albo ,,linia środkowa w trójkącie jest dwukrotnie krótsza od podstawy'' (brak kwantyfikatorów).} (dla każdego kąta i punktu wewnątrz istnieje odcinek przez ten punkt, który przecina ramiona kąta): jeśli $\beta(x, u, t)$, $\beta (y, u, z)$, $x \neq y$, to istnieją takie $v, w$, że $\beta(x, z, v)$, $\beta(x, y, w)$ oraz $\beta (v, t, w)$;% lang-pl
      \item V postulato di Euclide\footnote{Questo assioma può essere sostituito da ,,su ogni triangolo si può costruire un cerchio circoscritto'' (meno variabili) oppure da ,,il segmento medio di un triangolo è lungo la metà della base'' (senza quantificatori).} (per ogni angolo e ogni punto interno esiste un segmento passante per quel punto che interseca i lati dell'angolo): se $\beta(x, u, t)$, $\beta (y, u, z)$, $x \neq y$, allora esistono $v, w$ tali che $\beta(x, z, v)$, $\beta(x, y, w)$ e $\beta (v, t, w)$; % lang-it
\index{V postulat}% % lang-pl
\index{V postulato}% % lang-it

      \item aksjomat pięciu odcinków (cecha przystawania bkb): jeśli $xy \equiv x'y'$, $yz \equiv y'z'$, $xu \equiv x'u'$, $yu \equiv y'u'$, $\beta (x, y, z)$, $\beta(x', y', z')$ i $x \neq y$, to $zu \equiv z'u'$; % lang-pl
      \item assioma dei cinque segmenti (criterio di congruenza lato-angolo-lato): se $xy \equiv x'y'$, $yz \equiv y'z'$, $xu \equiv x'u'$, $yu \equiv y'u'$, $\beta (x, y, z)$, $\beta(x', y', z')$ e $x \neq y$, allora $zu \equiv z'u'$; % lang-it
\index{cecha przystawania bkb}% % lang-pl
\index{criterio di congruenza!lato-angolo-lato}%

      \item odkładanie odcinków: istnieje $z$ taki, że $\beta (x, y, z)$ i $yz \equiv uw$; % lang-pl
      \item trasporto dei segmenti: esiste un punto $z$ tale che $\beta (x, y, z)$ e $yz \equiv uw$; % lang-it

      \item istnieją trzy niewspółliniowe punkty\footnote{Bez tego aksjomatu teoria ma model na prostej rzeczywistej.} $x, y, z$: $\neg \beta(x, y, z)$, $\neg \beta(y, z, x)$, $\neg \beta(z, x, y, )$; % lang-pl
      \item esistono tre punti non allineati\footnote{Senza questo assioma la teoria ammette un modello sulla retta reale.} $x, y, z$: $\neg \beta(x, y, z)$, $\neg \beta(y, z, x)$, $\neg \beta(z, x, y, )$; % lang-it

      \item trzy punkty równoodległe od końców odcinka tworzą prostą\footnote{Bez tego aksjomatu teoria ma model w trzech i więcej wymiarach.}: jeśli $xu \equiv xv$, $yu \equiv yv$, $zu \equiv zv$ i $u \neq v$, to $\beta(x, y, z)$, $\beta(y, z, x)$ lub $\beta(z, x, y)$; % lang-pl
      \item tre punti equidistanti dagli estremi di un segmento giacciono su una retta\footnote{Senza questo assioma la teoria ammette modelli in tre o più dimensioni.}: se $xu \equiv xv$, $yu \equiv yv$, $zu \equiv zv$ e $u \neq v$, allora $\beta(x, y, z)$, $\beta(y, z, x)$ oppure $\beta(z, x, y)$; % lang-it

      \item schemat aksjomatu ciągłości. % lang-pl
      \item schema dell'assioma di continuità. % lang-it
\end{enumerate}
