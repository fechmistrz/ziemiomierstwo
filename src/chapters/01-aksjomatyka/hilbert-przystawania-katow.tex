\pol{\subsection{Aksjomaty przystawania kątów}}
\ita{\subsection{Assiomi di congruenza degli angoli}}
\pol{\index{aksjomat!przystawania kątów}}\ita{\index{assioma!di congruenza degli angoli}}%

\pol{Dokładnie jak dla odcinków, wprowadzamy niezdefiniowaną relację przystawania kątów, oznaczaną znowu przez $\cong$, ponieważ nie prowadzi to do nieporozumień.}
\ita{Esattamente come per i segmenti, introduciamo una relazione indefinita di congruenza tra angoli, indicata ancora con $\cong$, poiché ciò non genera ambiguità.}

\begin{axiom}[\pol{przystawania, C4}\ita{di congruenza, C4}]
    \pol{Dla każdego kąta $\angle BAC$ i każdej $DF$ półprostej istnieje dokładnie jedna półprosta $DE$ po ustalonej stronie prostej $DF$ taka, że $\angle BAC \cong \angle EDF$.}
    \ita{Per ogni angolo $\angle BAC$ e per ogni semiretta $DF$ esiste esattamente una semiretta $DE$ situata da un lato prefissato della retta $DF$ tale che $\angle BAC \cong \angle EDF$.}
\end{axiom}

\pol{Możemy traktować ten aksjomat jako odpowiednik cyrkla: pozwala przenosić kąty tak, jak Euklides (I.23).}
\ita{Possiamo considerare questo assioma come l'equivalente del compasso: permette di trasportare gli angoli come fa Euclide in (I.23).}
\pol{\index{cyrkiel}}\ita{\index{compasso}}

\begin{axiom}[\pol{przystawania, C5}\ita{di congruenza, C5}]
    \pol{Niech $\alpha, \beta, \gamma$ będą kątami.}
    \ita{Siano $\alpha$, $\beta$, $\gamma$ angoli.}
    \pol{Jeśli $\alpha \cong \beta$ oraz $\alpha \cong \gamma$, to $\beta \cong \gamma$.}
    \ita{Se $\alpha \cong \beta$ e $\alpha \cong \gamma$, allora $\beta \cong \gamma$.}
    \pol{Każdy kąt przystaje do siebie: $\alpha \cong \alpha$.}
    \ita{Ogni angolo è congruente a sé stesso: $\alpha \cong \alpha$.}
\end{axiom}

\pol{Mamy wreszcie cechę przystawania bok-kąt-bok, z której można wyprowadzić dodawanie kątów.}
\ita{Disponiamo infine del criterio di congruenza lato-angolo-lato, dal quale si può derivare l'addizione degli angoli.}
\pol{\index{cecha przystawania!bok-kąt-bok}}\ita{\index{criterio di congruenza!lato-angolo-lato}}%
\pol{Inne cechy przystawania to bok-bok-bok i kąt-bok-kąt.}
\ita{Altri criteri di congruenza sono lato-lato-lato e angolo-lato-angolo.}
\pol{Piszą o nich Hartshorne \cite[s. 99]{hartshorne2000}, Neugebauer \cite[s. 8, 11]{neugebauer_2018}, Pompe \cite[s. 25]{pompe_2022}.}
\ita{Ne scrivono Hartshorne \cite[p. 99]{hartshorne2000}, Neugebauer \cite[p. 8, 11]{neugebauer_2018} e Pompe \cite[p. 25]{pompe_2022}.}

\begin{axiom}[\pol{przystawania, C6}\ita{di congruenza, C6}]
    \pol{Niech $ABC$ i $DEF$ będą dwoma trójkątami takimi, że $AB \cong DE$, $AC \cong DF$ i $\angle BAC \cong \angle EDF$.}
    \ita{Siano $ABC$ e $DEF$ due triangoli tali che $AB \cong DE$, $AC \cong DF$ e $\angle BAC \cong \angle EDF$.}
    \pol{Wtedy pozostałe boki i kąty są przystające, a razem z nimi całe trójkąty: $BC \cong EF$, $\angle ABC \cong \angle DEF$ i $\angle ACB \cong \angle DFE$.}
    \ita{Allora anche i restanti lati e angoli sono congruenti, e con essi i triangoli interi: $BC \cong EF$, $\angle ABC \cong \angle DEF$ e $\angle ACB \cong \angle DFE$.}
\end{axiom}

\pol{Euklides uda, że dowodzi aksjomatu C6 metodą ,,nakładania'' figur, ale Hilbert znajdzie model będący świadkiem, że nie wynika on z poprzednich aksjomatów; patrz \cite[paragraf 11]{hilbert_1988}, jak wiemy od Greenberga \cite[s. 200]{greenberg_2010}.}
\ita{Euclide fingerà di dimostrare l'assioma C6 mediante il metodo della ``sovrapposizione'' delle figure, ma Hilbert costruirà un modello che mostra come esso non segua dagli assiomi precedenti; si veda \cite[paragrafo 11]{hilbert_1988}, come sappiamo da Greenberg \cite[p. 200]{greenberg_2010}.}
\pol{(W obronie Euklidesa należy dodać, że metoda nakładania pojawia się dwa razy: w (I.4) oraz (I.8); gdyby była słuszna, postulat 4, że wszystkie kąty proste są równe, nie byłby potrzebny!).}
\ita{(In difesa di Euclide va aggiunto che il metodo della sovrapposizione compare due volte, in (I.4) e (I.8); se fosse valido, il postulato 4, secondo cui tutti gli angoli retti sono uguali, non sarebbe necessario!).}
\pol{\index{kąt!prosty}}\ita{\index{angolo!retto}}%
\pol{Patrz też do Hartshorne'a \cite[s. 148-158]{hartshorne2000}.}
\ita{Si veda anche Hartshorne \cite[p. 148-158]{hartshorne2000}.}

\pol{Jeżeli chodzi o dodawanie kątów, musimy być ostrożni.}
\ita{Per quanto riguarda l'addizione degli angoli, occorre essere prudenti.}
\pol{Suma dwóch kątów może okazać się prostą (lub ,,dwoma kątami prostymi'' jak pisze Euklides) albo mieć miarę większą niż $\pi$, wtedy składniki sumy nie znajdują się we wnętrzu otrzymanego kąta.}
\ita{La somma di due angoli può risultare un angolo piatto (o ``due angoli retti'', come scrive Euclide) oppure avere ampiezza maggiore di $\pi$; in tal caso gli addendi non si trovano all'interno dell'angolo ottenuto.}
\pol{Kąty można także odejmować.}
\ita{Gli angoli possono anche essere sottratti.}

\begin{definition}[\pol{kąt przyległy}\ita{angolo adiacente}]
    \pol{Niech $\angle BAC$ będzie kątem, zaś $D$ punktem na prostej $AC$ po przeciwnej stronie $A$ niż $C$.}
    \ita{Sia $\angle BAC$ un angolo e sia $D$ un punto sulla retta $AC$, dal lato opposto di $A$ rispetto a $C$.}
    \pol{Wtedy kąty $\angle BAC$ i $\angle BAD$ nazywamy przyległymi.}
    \ita{Allora gli angoli $\angle BAC$ e $\angle BAD$ si dicono adiacenti.}
    \pol{\index{kąt!przyległy}}\ita{\index{angolo!adiacente}}
\end{definition}

\begin{proposition}
\pol{\index{kąt!wierzchołkowy}}\ita{\index{angolo!opposto al vertice}}%
    \pol{Kąty wierzchołkowe (para kątów przy dwóch przecinających się prostych, które nie są przyległe) są przystające.}
    \ita{Gli angoli opposti al vertice (la coppia di angoli formata da due rette incidenti che non sono adiacenti) sono congruenti.}
\end{proposition}

\pol{Piszą o tym Bogdańska, Neugebauer \cite[s. 6]{neugebauer_2018}.}
\ita{Ne scrivono Bogdańska e Neugebauer \cite[p. 6]{neugebauer_2018}.}
\pol{Pompe \cite[s. 5]{pompe_2022} pisze, że kąty naprzemianległe są równej miary wtedy i tylko wtedy, gdy dwie stosowne proste są równoległe.}
\ita{Pompe \cite[p. 5]{pompe_2022} scrive che gli angoli alterni interni hanno la stessa ampiezza se e solo se le due rette considerate sono parallele.}

\pol{Istnieje odpowiednik relacji między odcinkami ,,bycia mniejszym'' dla kątów:}
\ita{Esiste anche per gli angoli un analogo della relazione ``essere minore'' definita per i segmenti:}

\begin{definition}
    \pol{Niech $\angle BAC$ i $\angle EDF$ będą kątami.}
    \ita{Siano $\angle BAC$ e $\angle EDF$ due angoli.}
    \pol{Jeśli istnieje półprosta $DG$ wewnątrz kąta $EDF$ taka, że $\angle BAC \cong \angle GDF$, to powiemy, że kąt $\angle BAC$ jest mniejszy niż kąt $\angle EDF$ (a ten drugi jest większy niż pierwszy).}
    \ita{Se esiste una semiretta $DG$ interna all'angolo $EDF$ tale che $\angle BAC \cong \angle GDF$, diremo che l'angolo $\angle BAC$ è minore dell'angolo $\angle EDF$ (e quest'ultimo maggiore del primo).}
\end{definition}

\begin{definition}
    \pol{Kąt, który przystaje do kąta przyległego, nazywamy prostym.}
    \ita{Un angolo congruente al proprio angolo adiacente si chiama retto.}
    \pol{Dwie proste, które przecinają się i wyznaczają tam cztery kąty proste, będziemy nazywać prostopadłymi.}
    \ita{Due rette che si intersecano formando quattro angoli retti si dicono perpendicolari.}
\end{definition}

\pol{Nie jest ważne, który kąt przyległy wybierzemy, ponieważ kąty wierzchołkowe są przystające.}
\ita{Non importa quale angolo adiacente si scelga, poiché gli angoli opposti al vertice sono congruenti.}
