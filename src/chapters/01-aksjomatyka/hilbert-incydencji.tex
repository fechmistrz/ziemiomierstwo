\pol{\subsection{Aksjomaty incydencji}}
\ita{\subsection{Assiomi di incidenza}}
\pol{\index{aksjomat!incydencji}}
\ita{\index{assioma!di incidenza}}%

\begin{axiom}[\pol{incydencji}\ita{di incidenza}, I1]
    \pol{Dla każdej pary punktów $A$ oraz $B$ istnieje dokładnie jedna prosta $l$, która przechodzi przez te punkty.}
    \ita{Per ogni coppia di punti $A$ e $B$ esiste un'unica retta $l$ che passa per tali punti.}
\end{axiom}

\pol{Zdania ,,prosta $l$ przechodzi przez punkt $A$'' oraz ,,punkt $A$ leży na prostej $l$'' mają dokładnie to samo znaczenie i będziemy je stosować zamiennie.}
\ita{Le espressioni ``la retta $l$ passa per il punto $A$'' e ``il punto $A$ appartiene alla retta $l$'' hanno esattamente lo stesso significato e le useremo come sinonimi.}

\begin{axiom}[\pol{incydencji}\ita{di incidenza}, I2]
    \pol{Na każdej prostej leżą co najmniej dwa punkty.}
    \ita{Su ogni retta giacciono almeno due punti.}
\end{axiom}

\begin{axiom}[\pol{incydencji}\ita{di incidenza}, I3]
    \pol{Istnieją co najmniej trzy punkty, które nie są współliniowe.}
    \ita{Esistono almeno tre punti che non sono allineati.}
\end{axiom}

\pol{,,Być współliniowym'' to synonim leżenia na jednej prostej.}
\ita{``Essere allineati'' significa giacere sulla stessa retta.}
\pol{\index{współliniowy}}
\ita{\index{allineato}}%
\pol{Analogicznie mówimy, że wiele prostych, które przechodzą przez jeden punkt, jest współpękowych.}
\ita{Analogamente, diciamo che più rette che passano per uno stesso punto sono concorrenti.}

\begin{proposition}
    \pol{Dwie różne proste mogą mieć co najwyżej jeden punkt wspólny.}
    \ita{Due rette distinte possono avere al più un punto in comune.}
\end{proposition}

\begin{definition}
    \pol{Dwie różne proste, które nie są współpękowe (nie mają punktów wspólnych), nazywamy równoległymi.}
    \ita{Due rette distinte che non sono concorrenti (non hanno punti in comune) si dicono parallele.}
    \pol{Każda prosta jest też równoległa do siebie.}
    \ita{Ogni retta è inoltre parallela a sé stessa.}
\end{definition}
\pol{\index{prosta!równoległa}}
\ita{\index{retta!parallela}}%

\pol{Symbolicznie: $k \parallel l$.}
\ita{Simbolicamente: $k \parallel l$.}
\pol{Trzeba uważać, bo dla niektórych dwie proste równoległe są współpękowe, bo przechodzą przez punkt w nieskończoności, jak zwykło się mawiać w geometrii rzutowej.}
\ita{Bisogna fare attenzione, perché per alcuni due rette parallele sono concorrenti, poiché passano per un punto all'infinito, come si usa dire nella geometria proiettiva.}

\pol{Na dwóch prostych, które nie są równoległe ($k \nparallel l$), leży dokładnie jeden punkt.}
\ita{Su due rette che non sono parallele ($k \nparallel l$) giace esattamente un punto.}
\pol{Mówimy, że proste się przecinają, a wspomniany punkt jest ich punktem przecięcia.}
\ita{Diciamo che le rette si intersecano e che il punto menzionato è il loro punto di intersezione.}
\pol{Oznaczamy go $k \cdot l$.}
\ita{Lo indichiamo con $k \cdot l$.}

\ita{Hartshorne \cite[p. 38]{hartshorne2000} presenta il seguente assioma:}

\begin{axiom}[\pol{Playfaira}\ita{di Playfair}, P]
    \pol{Dla każdej prostej $l$ oraz punktu $A$, istnieje co najwyżej jedna prosta, która przechodzi przez $A$ i jest równoległa do $l$.}
    \ita{Per ogni retta $l$ e punto $A$, esiste al più una retta che passa per $A$ ed è parallela a $l$.}
\end{axiom}

    \ita{Consideriamo una geometria in cui il piano è costituito da cinque punti $A$, $B$, $C$, $D$, $E$ e contiene dieci rette distinte, ciascuna passante per esattamente due punti distinti.}
    \pol{Wtedy proste $AB$ i $AC$ mają punkt wspólny $A$, chociaż obydwie są równoległe do prostej $DE$.}

\pol{\subsubsection{Wstęp do geometrii rzutowej}}
\ita{\subsubsection{Introduzione alla geometria proiettiva}}

\begin{proposition}
    \pol{Płaszczyzna rzutowa to taki zbiór punktów oraz prostych (podzbiorów zbioru punktów), że:}
    \ita{Un piano proiettivo è un insieme di punti e rette (sottoinsiemi dell'insieme dei punti) tale che:}
    \pol{\index{płaszczyzna!rzutowa}}
    \ita{\index{piano!proiettivo}}
    \begin{itemize}
        \item \pol{przez dwa różne punkty przechodzi dokładnie jedna prosta,}\ita{per due punti distinti passa una e una sola retta,}
        \item \pol{każde dwie proste mają punkt wspólny,}\ita{ogni due rette hanno un punto in comune,}
        \item \pol{każda prosta ma co najmniej trzy punkty i}\ita{ogni retta contiene almeno tre punti e}
        \item \pol{nie wszystkie punkty są współliniowe.}\ita{non tutti i punti sono allineati.}
    \end{itemize}
    \pol{(Każdy z wymienionych aksjomatów jest niezależny od pozostałych, ze wszystkich razem wynikają aksjomaty incydencji).}
    \ita{(Ciascuno degli assiomi elencati è indipendente dagli altri; tutti insieme implicano gli assiomi di incidenza).}
    \pol{Każda płaszczyzna rzutowa ma co najmniej 7 punktów, dokładnie jedna płaszczyzna rzutowa ma dokładnie 7 punktów.}
    \ita{Ogni piano proiettivo ha almeno 7 punti ed esiste un unico piano proiettivo con esattamente 7 punti.}
    \pol{Jeśli istnieje prosta, która ma $n+1$ punktów, to płaszczyzna ma $n^2 + n + 1$ punktów.}
    \ita{Se esiste una retta che contiene $n+1$ punti, allora il piano contiene $n^2+n+1$ punti.}
\end{proposition}

\pol{\subsubsection{Wstęp do geometrii afinicznej}}
\ita{\subsubsection{Introduzione alla geometria affine}}

\begin{proposition}
    \pol{Płaszczyzna afiniczna to taki zbiór punktów i prostych, które spełniają:}
    \ita{Un piano affine è un insieme di punti e rette che soddisfa:}
    \pol{\index{płaszczyzna!afiniczna}}
    \ita{\index{piano!affine}}
    \begin{itemize}
        \item \pol{aksjomaty incydencji oraz}
              \ita{gli assiomi di incidenza e}
        \item \pol{mocniejszą wersję aksjomatu Playfaira: dla każdej prostej $l$ i punktu $A$, dokładnie jedna prosta przechodzi przez punkt $A$ i jest równoległa do $l$.}
              \ita{una versione più forte dell'assioma di Playfair: per ogni retta $l$ e punto $A$, passa per $A$ una e una sola retta parallela a $l$.}
    \end{itemize}
    \pol{Każda prosta na płaszczyźnie afinicznej ma tyle samo punktów.}
    \ita{Ogni retta di un piano affine contiene lo stesso numero di punti.}
    \pol{Jeśli pewna prosta ma $n$ punktów, to płaszczyzna ma dokładnie $n^2$ punktów.}
    \ita{Se una retta contiene $n$ punti, allora il piano contiene esattamente $n^2$ punti.}
    \pol{Istnieją płaszczyzny afiniczne o $4$, $9$, $16$ i $25$ punktach, ale nie istnieje taka, która miałaby $36$ punktów.}
    \ita{Esistono piani affini con $4$, $9$, $16$ e $25$ punti, ma non ne esiste alcuno con $36$ punti.}
\end{proposition} % Hartshorne 71, 72
