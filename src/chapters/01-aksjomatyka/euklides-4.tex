\pol{\subsection{Księga IV}}
\ita{\subsection{Libro IV}}
\pol{\subsubsection{Definicje}}
\ita{\subsubsection{Definizioni}}
\begin{enumerate}
	\item [4.1] \pol{Dwie figury prostokreślne są wpisane jedna w drugą, jeśli wierzchołki pierwszej leżą na bokach drugiej, a boki drugiej przechodzą przez wierzchołki pierwszej.}
	\ita{Due figure rettilinee sono inscritte l'una nell'altra se i vertici della prima giacciono sui lati della seconda e i lati della seconda passano per i vertici della prima.}
	\item [4.2] \pol{Podobnie definiuje się figurę, która jest opisana na innej figurze.}
	\ita{In modo analogo si definisce una figura circoscritta a un'altra figura.}
	\item [4.3] \pol{Figura prostokreślna jest wpisana w okrąg, kiedy wszystkie jej wierzchołki leżą na tym okręgu.}
	\ita{Una figura rettilinea è inscritta in un cerchio quando tutti i suoi vertici appartengono a tale cerchio.}
	\item [4.4] \pol{Figura prostokreślna jest opisana na okręgu, kiedy każdy jej bok jest styczny do tego okręgu.}
	\ita{Una figura rettilinea è circoscritta a un cerchio quando ciascuno dei suoi lati è tangente a tale cerchio.}
	\item [4.5] \pol{Podobnie definiuje się, że okrąg jest wpisany w figurę prostokreślną.}
	\ita{Analogamente si definisce un cerchio inscritto in una figura rettilinea.}
	\item [4.6] \pol{Koło opisuje się na figurze prostokreślnej, kiedy okrąg (jego brzeg) przechodzi przez wszystkie wierzchołki figury.}
	\ita{Un cerchio è circoscritto a una figura rettilinea quando la sua circonferenza passa per tutti i vertici della figura.}
	\item [4.7] \pol{Linia prosta (właściwie: odcinek) kreśli się w kole, gdy jej końca leżą na brzegu tego koła.}
	\ita{Una linea retta (più precisamente: un segmento) è tracciata in un cerchio quando i suoi estremi giacciono sulla circonferenza.}
\end{enumerate}

\pol{\subsubsection{Twierdzenia}}
\ita{\subsubsection{Proposizioni}}
\begin{enumerate}
	\item [4.1] \pol{Wpisać odcinek krótszy od średnicy w dany okrąg.}
	\ita{Inscrivere in un dato cerchio un segmento più corto del diametro.}
	\item [4.2] \pol{Wpisać trójkąt podobny do danego w okrąg.}
	\ita{Inscrivere in un cerchio un triangolo simile a un triangolo dato.}
	\item [4.3] \pol{Opisać trójkąt podobny do danego na okręgu.}
	\ita{Circoscrivere a un cerchio un triangolo simile a un triangolo dato.}
	\item [4.4] \pol{Wpisać okrąg w dany trójkąt.}
	\ita{Inscrivere un cerchio in un dato triangolo.}
	\item [4.5] \pol{Opisać okrąg na danym kwadracie.} \pol{\index{okrąg!opisany}} \ita{\index{cerchio!circoscritto}}
	\ita{Circoscrivere un cerchio a un dato quadrato.}
	\item [4.6] \pol{Wpisać kwadrat w dany okrąg.}
	\ita{Inscrivere un quadrato in un dato cerchio.}
	\item [4.7] \pol{Opisać kwadrat na danym okręgu.}
	\ita{Circoscrivere un quadrato a un dato cerchio.}
	\item [4.8] \pol{Wpisać okrąg w dany kwadrat.}
	\ita{Inscrivere un cerchio in un dato quadrato.}
	\item [4.9] \pol{Opisać okrąg na danym kwadracie.}
	\ita{Circoscrivere un cerchio a un dato quadrato.}
	\item [4.10] \pol{Wykreślić trójkąt równoramienny, którego kąt przy podstawie jest podwojeniem kąta przy wierzchołku (o kątach $\pi/5$, $2\pi/5$, $2\pi/5$; taki trójkąt nazywa się czasami złotym).} \pol{\index{trójkąt!złoty}}
	\ita{Costruire un triangolo isoscele nel quale l'angolo alla base è il doppio dell'angolo al vertice (con angoli $\pi/5$, $2\pi/5$, $2\pi/5$; tale triangolo è talvolta chiamato triangolo aureo).} \ita{\index{triangolo!aureo}}
	\item [4.11] \pol{Wpisać pięciokąt foremny\footnote{Euklides mówi tu o pięciokącie równobocznym i jednocześnie równokątnym.} w dany okrąg.}
	\ita{Inscrivere un pentagono regolare\footnote{Euclide parla qui di un pentagono al tempo stesso equilatero ed equiangolo.} in un dato cerchio.}
	\item [4.12] \pol{Opisać pięciokąt foremny na danym okręgu.} \pol{\index{pięciokąt|see{wielokąt}}}
	\pol{\index{wielokąt!pięciokąt}}
	\ita{Circoscrivere un pentagono regolare a un dato cerchio.} \ita{\index{pentagono|see{poligono}}}
	\ita{\index{poligono!pentagono}}
	\item [4.13] \pol{Wpisać okrąg w dany pięciokąt równoboczny i równokątny.}
	\ita{Inscrivere un cerchio in un dato pentagono equilatero ed equiangolo.}
	\item [4.14] \pol{Opisać okrąg na danym pięciokącie równobocznym i równokątnym.}
	\ita{Circoscrivere un cerchio a un dato pentagono equilatero ed equiangolo.}
	\item [4.15] \pol{Wpisać sześciokąt równoboczny i równokątny w dany okrąg.}
	\ita{Inscrivere un esagono equilatero ed equiangolo in un dato cerchio.}
	\pol{\index{sześciokąt|see{wielokąt}}}
	\pol{\index{wielokąt!sześciokąt}}
	\ita{\index{esagono|see{poligono}}}
	\ita{\index{poligono!esagono}}
	\item [4.16] \pol{Wpisać piętnastokąt równoboczny i równokątny w dany okrąg.}
	\ita{Inscrivere un pentadecagono equilatero ed equiangolo in un dato cerchio.}
	\pol{\index{piętnastokąt|see{wielokąt}}}
	\pol{\index{wielokąt!piętnastokąt}}
	\ita{\index{pentadecagono|see{poligono}}}
	\ita{\index{poligono!pentadecagono}}
\end{enumerate}
