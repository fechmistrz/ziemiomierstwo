\begin{proposition}[\pol{twierdzenie Miquela}\ita{teorema di Miquel}]
\index{twierdzenie!Miquela}%
\pol{Niech $ABC$ będzie trójkątem, na bokach którego ($AB$, $BC$, $AC$) wybrano punkty $C'$, $A'$, $B'$. }%
\ita{Sia $ABC$ un triangolo, sui cui lati ($AB$, $BC$, $AC$) siano scelti i punti $C'$, $A'$, $B'$. }%
\pol{Wtedy okręgi opisane na trójkątach $AB'C'$, $A'BC'$, $A'B'C$ przecinają się w~jednym punkcie. }%
\ita{Allora i cerchi circoscritti ai triangoli $AB'C'$, $A'BC'$, $A'B'C$ si intersecano in un unico punto. }%
\end{proposition}

\pol{Sformułowanie mówi tylko o trójkątach, ale dowód wykorzystuje własności czworokątów cyklicznych. }%
\ita{L'enunciato riguarda solo triangoli, ma la dimostrazione utilizza proprietà dei quadrilateri ciclici. }%
\pol{Hartshorne \cite[s. 61]{hartshorne2000} jako ćwiczenie napisze, że tym razem punkt Miquela zostanie nazwany na cześć osoby, która go odkryje w 1838 roku. }%
\ita{Hartshorne \cite[p. 61]{hartshorne2000} scrive come esercizio che, questa volta, il punto di Miquel sarà chiamato in onore della persona che lo scoprirà nel 1838. }%

\begin{proposition}[\pol{twierdzenie Miquela o czterech okręgach}\ita{teorema di Miquel sui quattro cerchi}]
\label{miquel_to_tez_cztery_okregi}%
\pol{Okręgi opisane na czterech trójkątach wyznaczonych przez czworobok zupełny przechodzą przez jeden punkt. }%
\ita{I cerchi circoscritti ai quattro triangoli determinati da un quadrilatero completo passano per uno stesso punto. }%
\end{proposition}

\pol{Zostanie to ogłoszone krótko przez Jakoba Steinera w wydaniu 1827/1828 \emph{,,Gergonne's Annales de Mathématiques''}, dowiedzione przez Miquela. }%
\ita{Questo sarà annunciato brevemente da Jakob Steiner nel volume 1827/1828 \emph{``Gergonne's Annales de Mathématiques''}, e dimostrato da Miquel. }%

\begin{proposition}[\pol{twierdzenie Miquela o pięciokącie}\ita{teorema di Miquel sul pentagono}]
\pol{Niech $ABCDE$ będzie pięciokątem wypukłym, którego przedłużenia boków przecinają się w punktach $F$, $G$, $H$, $I$, $K$. }%
\ita{Sia $ABCDE$ un pentagono convesso i cui prolungamenti dei lati si intersecano nei punti $F$, $G$, $H$, $I$, $K$. }%
\pol{Na trójkątach $CFD$, $DGE$, $EHA$, $AIB$, $BKC$ opisano okręgi. }%
\ita{Sui triangoli $CFD$, $DGE$, $EHA$, $AIB$, $BKC$ si costruiscono i cerchi circoscritti. }%
\pol{Wtedy ich nowe punkty przecięcia (różne od $A$, $B$, $C$, $D$, $E$) leżą na jednym okręgu. }%
\ita{Allora i loro nuovi punti di intersezione (diversi da $A$, $B$, $C$, $D$, $E$) giacciono su uno stesso cerchio. }%
\end{proposition}

\begin{proposition}[\pol{twierdzenie Miquela o sześciu okręgach}\ita{teorema di Miquel sui sei cerchi}]
\label{miquel6}%
\pol{Niech $\Gamma$ będzie okręgiem opisanym na czworokącie $ABCD$. }%
\ita{Sia $\Gamma$ il cerchio circoscritto al quadrilatero $ABCD$. }%
\pol{Niech $\Gamma_1$, $\Gamma_2$, $\Gamma_3$, $\Gamma_4$ będą dowolnymi okręgami, które przechodzą przez $AB$, $BC$, $CD$, $DA$. }%
\ita{Siano $\Gamma_1$, $\Gamma_2$, $\Gamma_3$, $\Gamma_4$ cerchi arbitrari che passano rispettivamente per $AB$, $BC$, $CD$, $DA$. }%
\pol{Wtedy ich cztery nowe punkty przecięcia tworzą czworokąt cykliczny. }%
\ita{Allora i loro quattro nuovi punti di intersezione formano un quadrilatero ciclico. }%
\index{czworokąt!cykliczny}%
% https://en.wikipedia.org/wiki/Miquel%27s_theorem#Miquel's_six_circle_theorem
% Ostermann, Alexander; Wanner, Gerhard (2012), Geometry by its History, Springer, ISBN 978-3-642-29162-3 STRONA=352
\end{proposition}

\begin{proposition}
\pol{Jeśli oznaczyć punkt Miquela przez $M$, to kąty $\angle MA'B$, $\angle MB'C$ oraz $\angle MC'A$ mają równe miary. }%
\ita{Se si denota il punto di Miquel con $M$, allora gli angoli $\angle MA'B$, $\angle MB'C$ e $\angle MC'A$ hanno la stessa misura. }%
\end{proposition}

\begin{theorem}[\pol{Droza-Farny'ego}\ita{di Droz-Farny}]
\pol{Dany jest trójkąt $\triangle ABC$ z ortocentrum $H$ oraz dwie prostopadłe proste $l_1, l_2$ przechodzące przez $H$. }%
\ita{È dato un triangolo $\triangle ABC$ con ortocentro $H$ e due rette perpendicolari $l_1, l_2$ passanti per $H$. }%
\pol{Niech $A_k$, $B_k$, $C_k$ będą punktami przecięcia boków $a$, $b$, $c$ trójkąta z prostą $l_k$ dla $k = 1, 2$. }%
\ita{Siano $A_k$, $B_k$, $C_k$ i punti di intersezione dei lati $a$, $b$, $c$ del triangolo con la retta $l_k$ per $k = 1, 2$. }%
\pol{Wtedy środki odcinków $A_1A_2$, $B_1B_2$, $C_1C_2$ są współliniowe. }%
\ita{Allora i punti medi dei segmenti $A_1A_2$, $B_1B_2$, $C_1C_2$ sono allineati. }%
\end{theorem}

\pol{Arnold Droz-Farny będzie szwajcarskim matematykiem, nauczycielem w liceum niedaleko Bazylei. }%
\ita{Arnold Droz-Farny era un matematico svizzero, insegnante in un liceo vicino a Basilea. }%
\pol{Twierdzenie postawi w 1899 roku bez dowodu, my wspominamy o nim tutaj, bo można napisać brakujący dowód tak, by korzystał z alternatywnej definicji punktu Miguela (jako ogniska paraboli wpisanej w czworobok zupełny, patrz \cite[s. 262]{neugebauer_2018}). }%
