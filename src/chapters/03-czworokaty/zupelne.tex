\begin{definition}
\pol{Cztery punkty, z których żadne trzy nie są współliniowe, leżące na sześciu prostych przez sześć par punktów, nazywamy czworokątem zupełnym. }%
\ita{Quattro punti, di cui nessuno tre sono allineati, situati su sei rette determinate dalle sei coppie di punti, si chiamano quadrilatero completo. }%
\end{definition}

\pol{Sześć prostych wyznacza dodatkowe trzy punkty, które nazywamy przekątniowymi. }%
\ita{Le sei rette determinano ulteriori tre punti, che chiamiamo punti diagonali. }%
\pol{\index{punkty przekątniowe}}%
\ita{\index{punti diagonali}}%

\begin{definition}
\pol{Cztery proste, z których żadne trzy nie są współpękowe, przecinające się parami w sześciu różnych punktach, nazywamy czworobokiem zupełnym. }%
\ita{Quattro rette, di cui nessuna tre concorrenti, che si intersecano a coppie in sei punti distinti, si chiamano quadrilatero completo. }%
\end{definition}

\begin{theorem}[Jemieljanowa]
\pol{Punkt Miquela czworoboku zupełnego leży na okręgu dziewięciu punktów trójkąta przekątnego tego czworoboku. }%
\ita{Il punto di Miquel del quadrilatero completo giace sul cerchio dei nove punti del triangolo diagonale di tale quadrilatero. }%
\pol{\index{twierdzenie!Jemieljanowa}}%
\ita{\index{teorema!Jemeljanova}}%
\end{theorem}

\pol{Środki trzech przekątnych czworoboku zupełnego leżą na jednej prostej, zwanej prostą Newtona-Gaussa. }%

\begin{theorem}[Annego]
\pol{\index{twierdzenie!Annego}}%
\ita{\index{teorema!Anne}}%
\pol{Niech $ABCD$ będzie wypukłym czworokątem, ale nie równoległobokiem; z punktami $E$, $F$ na środku przekątnych $BD$, $AC$ oraz punktem $L$ gdziekolwiek wewnątrz. }%
\ita{Sia $ABCD$ un quadrilatero convesso, ma non un parallelogramma; con punti $E$, $F$ al centro delle diagonali $BD$, $AC$ e un punto $L$ qualsiasi all'interno. }%
\pol{Wtedy następujące warunki są równoważne: }%
\ita{Allora le seguenti condizioni sono equivalenti: }%
\begin{itemize}
	\item \pol{$|\triangle_{BCL}| + |\triangle_{DAL}| = |\triangle_{LAB}| + |\triangle_{DLC}|$},
	\ita{$|\triangle_{BCL}| + |\triangle_{DAL}| = |\triangle_{LAB}| + |\triangle_{DLC}|$},
	\item \pol{punkt $L$ leży na prostej $EF$.}
	      \ita{il punto $L$ giace sulla retta $EF$.}
\end{itemize}
\end{theorem}

\begin{theorem}[Gaussa-Bodenmillera]
\pol{\index{twierdzenie!Gaussa-Bodenmillera}}%
\ita{\index{teorema!Gauss-Bodenmiller}}%
\pol{Trzy okręgi, których średnice stanowią przekątne ustalonego czworoboku zupełnego, są współosiowe. }%
\ita{Tre cerchi, i cui diametri sono le diagonali di un quadrilatero completo dato, sono coassiali. }%

\ita{Il loro asse radicale comune passa per gli ortocentri dei quattro triangoli determinati dai lati del quadrilatero. }%

\end{theorem}
