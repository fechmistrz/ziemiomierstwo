\begin{proposition}[\pol{okrąg opisany na czworokącie}\ita{cerchio circoscritto a un quadrilatero}]
\pol{\index{okrąg!opisany}}%
\ita{\index{cerchio!circoscritto}}%
\label{prp_incircle}
\pol{Niech $A$, $B$, $C$, $D$ będą czterema punktami na płaszczyźnie takimi, że $A$ i $B$ leżą po tej samej stronie prostej $CD$. }%
\ita{Siano $A$, $B$, $C$, $D$ quattro punti nel piano tali che $A$ e $B$ si trovino dalla stessa parte della retta $CD$. }%
\pol{Wtedy następujące warunki są równoważne: }%
\ita{Allora le seguenti condizioni sono equivalenti: }%
\begin{itemize}
\item \pol{punkty $A$, $B$, $C$, $D$ leżą na jednym okręgu;}
      \ita{i punti $A$, $B$, $C$, $D$ giacciono su uno stesso cerchio;}
\item \pol{kąty $\angle ACB$ i $\angle ADB$ są sobie równe;}
      \ita{gli angoli $\angle ACB$ e $\angle ADB$ sono uguali;}
\item \pol{suma dwóch przeciwległych kątów czworokąta wypukłego wyznaczonego przez wierzchłki $A$, $B$, $C$, $D$ ma miarę kąta półpełnego.}
      \ita{la somma di due angoli opposti del quadrilatero convesso determinato dai vertici $A$, $B$, $C$, $D$ è un angolo piatto.}
\end{itemize}
\end{proposition}

\begin{proposition}
\pol{Dany jest wypukły czworokąt $ABCD$, którego przekątne przecinają się w punkcie $E$, zaś boki $AD$ i $BC$ (a właściwie ich przedłużenia) przecinają się w punkcie $F$. }%
\ita{È dato un quadrilatero convesso $ABCD$, le cui diagonali si intersecano nel punto $E$, mentre i lati $AD$ e $BC$ (più precisamente i loro prolungamenti) si intersecano nel punto $F$. }%
\pol{Niech $\Omega$ będzie okręgiem, którego średnicą jest odcinek $EF$. }%
\ita{Sia $\Omega$ il cerchio avente come diametro il segmento $EF$. }%
\pol{Wtedy $\Omega$ przecina bok $AD$ w $N$, zaś $BC$ w punkcie $M$, przedłużenie $AC$ w $L$, zaś przedłużenie $BD$ w $K$. }%
\ita{Allora $\Omega$ interseca il lato $AD$ in $N$, il lato $BC$ in $M$, il prolungamento di $AC$ in $L$ e il prolungamento di $BD$ in $K$. }%
\pol{Wtedy twierdzenie Pascala o sześciokątach $EKNFML$ i $EKMFNL$ wpisanych w $\Omega$ mówi, że proste $NK$ i $ML$ przecinają się w punkcie $P$ na boku $AB$, zaś $NL$ i $MK$ w punkcie $Q$ na boku $CD$. }%
\ita{Allora il teorema di Pascal sui sei lati $EKNFML$ ed $EKMFNL$ inscritti in $\Omega$ afferma che le rette $NK$ e $ML$ si intersecano in un punto $P$ sul lato $AB$, mentre $NL$ e $MK$ si intersecano in un punto $Q$ sul lato $CD$. }%
\pol{(Punkty $P$, $Q$ nazywane są czasami punktami Pascala). }%
\ita{(I punti $P$, $Q$ sono talvolta chiamati punti di Pascal). }%

\pol{Wtedy następujące warunki są równoważne: }%
\ita{Allora le seguenti condizioni sono equivalenti: }%
\begin{itemize}
\item \pol{czworokąt $ABCD$ jest cykliczny}
      \ita{il quadrilatero $ABCD$ è ciclico}
\item \pol{punkty $P, Q$ i środek okręgu $\Omega$ są współliniowe}
      \ita{i punti $P$, $Q$ e il centro del cerchio $\Omega$ sono allineati}
\item \pol{punkty $P$, $Q$ są środkami boków $AB$, $CD$.}
      \ita{i punti $P$, $Q$ sono i punti medi dei lati $AB$, $CD$.}
\end{itemize}
\end{proposition}

\begin{proposition}
\pol{Czworokąt o bokach długości $a$, $b$, $c$, $d$ jest wpisany w okrąg o promieniu $R$. }%
\ita{Un quadrilatero con lati di lunghezza $a$, $b$, $c$, $d$ è inscritto in un cerchio di raggio $R$. }%
\pol{Wtedy następujące warunki są równoważne: }%
\ita{Allora le seguenti condizioni sono equivalenti: }%
\begin{itemize}
    \item $a^2 + b^2 + c^2 + d^2 = 8R^2$,
    \item \pol{co najmniej jeden kąt czworokąta jest prosty lub jego przekątne są prostopadłe.}
          \ita{almeno un angolo del quadrilatero è retto oppure le sue diagonali sono perpendicolari.}
\end{itemize}
\end{proposition}

\label{prp_excircle}%
\pol{Dany jest czworokąt wypukły $ABCD$. }%
\ita{È dato un quadrilatero convesso $ABCD$. }%
\pol{Wtedy następujące warunki są równoważne: }%
\ita{Allora le seguenti condizioni sono equivalenti: }%
\begin{itemize}
    \item \pol{w czworokąt $ABCD$ można wpisać okrąg,}
          \ita{si può inscrivere un cerchio nel quadrilatero $ABCD$,}
    \item $AB + CD = AD + BC$.
\end{itemize}
\pol{\index{okrąg!wpisany}}%

\end{proposition}

\begin{proposition}[\pol{twierdzenie o cięciwach}\ita{teorema delle corde secanti}]
\label{prop_intersecting_chords}%
\pol{\index{cięciwa}}%
\ita{\index{corda}}%
\pol{\index{twierdzenie!o cięciwach}}%
\ita{\index{teorema!delle corde secanti}}%
\pol{Niech punkty $A$, $B$ leżą na pewnej prostej przechodzącej przez punkt $S$, zaś punkty $C$, $D$ leżą na innej prostej przez ten punkt. }%
\ita{Siano i punti $A$, $B$ su una retta passante per il punto $S$, mentre i punti $C$, $D$ giacciono su un'altra retta passante per lo stesso punto. }%
\pol{Wtedy cztery punkty $A$, $B$, $C$, $D$ leżą na jednym okręgu wtedy i tylko wtedy, gdy: }%
\ita{Allora i quattro punti $A$, $B$, $C$, $D$ giacciono su uno stesso cerchio se e solo se: }%
\begin{equation}
	|AS| \cdot |BS| = |CS| \cdot |DS|.
\end{equation}
\end{proposition}
