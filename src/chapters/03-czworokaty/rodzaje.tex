\begin{definition}[\pol{czworokąt}\ita{quadrilatero}]
\pol{\index{czworokąt}}%
\ita{\index{quadrilatero}}%
    \pol{Dane są cztery punkty $A$, $B$, $C$, $D$, z których żadne trzy nie są współliniowe, takimi że odcinki $AB$, $BC$, $CD$, $DA$ nie mają części wspólnej poza końcami.}
    \ita{Sono dati quattro punti $A$, $B$, $C$, $D$, tali che nessuno tre di essi siano allineati e tali che i segmenti $AB$, $BC$, $CD$, $DA$ non abbiano parti in comune se non agli estremi.}
    \pol{Wtedy sumę tych odcinków nazywamy czworokątem.}
    \ita{Allora l’unione di questi segmenti si chiama quadrilatero.}    
\end{definition}

\pol{Po angielsku \emph{quadrangle}, z łacińskiego \emph{quadri} określającego liczbę cztery oraz \emph{latus}, czyli boku (a~więc technicznie czworobok, nie czworokąt); istnieje też rzadko używane słowo \emph{tetragon} (przez analogię do \emph{polygon, pentagon}) z greckiego \emph{tetra} (cztery) oraz \emph{gon} (znaczącego tym razem kąt).}
\ita{In inglese \emph{quadrangle}, dal latino \emph{quadri} (quattro) e \emph{latus} (lato) (quindi tecnicamente “quadrilatero”, non “quadrangolo”); esiste anche il termine meno comune \emph{tetragon} (per analogia con \emph{polygon, pentagon}), dal greco \emph{tetra} (quattro) e \emph{gon} (che qui significa angolo).}

\begin{definition}[\pol{trapez}\ita{trapezio}]
\pol{\index{trapez}}%
\ita{\index{trapezio}}%
    \pol{Czworokąt, który posiada jedną lub dwie pary boków równoległych nazywamy trapezem.}
    \ita{Un quadrilatero che possiede una o due coppie di lati paralleli si chiama trapezio.}
\end{definition}

\pol{Po brytyjsku \emph{trapezium}, po amerykańsku \emph{trapezoid} --- widać więc kolizję z terminem na czworokąt ,,w~ogólnym położeniu''.}
\ita{In inglese britannico \emph{trapezium}, in quello americano \emph{trapezoid} — si vede quindi una collisione con il termine per un quadrilatero “in posizione generale”.}

\begin{definition}[\pol{trapez równoramienny}\ita{trapezio isoscele}]
\pol{\index{trapez!równoramienny}}%
\ita{\index{trapezio!isoscele}}%
    \pol{Czworokąt, który posiada parę boków równoległych, którego kąty przy podstawie są równej miary, nazywamy trapezem równoramiennym.}
    \ita{Un quadrilatero che possiede una coppia di lati paralleli e i cui angoli alla base sono uguali si chiama trapezio isoscele.}
\end{definition}

\pol{W szczególności, nie można definiować trapezu równoramiennego jako trapezu, który ma ramiona równej długości!}
\ita{In particolare, non si può definire un trapezio isoscele come un trapezio con i lati obliqui di uguale lunghezza!}
\pol{To czyniłoby wszystkie równoległoboki trapezami równoramiennymi, co nie ma żadnych zalet.}
\ita{Questo renderebbe tutti i parallelogrammi trapezi isosceli, il che non ha alcun vantaggio.}
\pol{\index{równoległobok}}%
\ita{\index{parallelogramma}}%
\pol{Inne, równoważne definicje mówią, że jest to czworokąt z osią symetrii, która połowi parę przeciwległych boków albo trapez, który ma przekątne równej długości.}
\ita{Altre definizioni equivalenti affermano che è un quadrilatero con un asse di simmetria che dimezza una coppia di lati opposti, oppure un trapezio con diagonali di uguale lunghezza.}

\begin{definition}[\pol{równoległobok}\ita{parallelogramma}]
\pol{\index{równoległobok}}%
\ita{\index{parallelogramma}}%
    \pol{Czworokąt, który posiada dwie pary boków równoległych, nazywamy równoległobokiem.}
    \ita{Un quadrilatero che possiede due coppie di lati paralleli si chiama parallelogramma.}
\end{definition}

\pol{Każdy równoległobok (po angielsku \emph{parallelogram}) jest też trapezem.}
\ita{Ogni parallelogramma (in inglese \emph{parallelogram}) è anche un trapezio.}
\pol{Alternatywne definicje głoszą, że ma przeciwległe boki równej długości, albo równe przeciwległe kąty, albo połowiące się wzajemnie przekątne (jak pan Pompe opisał \cite[s. 37]{pompe_2022}).}
\ita{Definizioni alternative affermano che ha lati opposti di uguale lunghezza, oppure angoli opposti uguali, oppure diagonali che si bisecano (come descritto da Pompe \cite[p. 37]{pompe_2022}).}

\begin{definition}[\pol{romb}\ita{rombo}]
\pol{\index{romb}}%
\ita{\index{rombo}}%
    \pol{Czworokąt, który posiada cztery boki tej samej długości, nazywamy rombem.}
    \ita{Un quadrilatero con quattro lati della stessa lunghezza si chiama rombo.}
\end{definition}

\pol{Równoważnie, romby to te czworokąty, których prostopadłe przekątne połowią się i rozcinają romb na cztery przystające trójkąty (prostokątne).}
\ita{Equivalentemente, i rombi sono quei quadrilateri le cui diagonali perpendicolari si bisecano e dividono il rombo in quattro triangoli congruenti (rettangoli).}
\pol{Angielski nazywa je \emph{rhombus}.}
\ita{In inglese si chiamano \emph{rhombus}.}

\begin{definition}[\pol{prostokąt}\ita{rettangolo}]
\pol{\index{prostokąt}}%
\ita{\index{rettangolo}}%
    \pol{Czworokąt, który posiada cztery kąty tej samej miary, nazywamy prostokątem.}
    \ita{Un quadrilatero che ha quattro angoli uguali si chiama rettangolo.}
\end{definition}

\begin{definition}[\pol{kwadrat}\ita{quadrato}]
\pol{\index{kwadrat}}%
\ita{\index{quadrato}}%
    \pol{Czworokąt, który posiada cztery kąty tej samej miary i wszystkie boki tej samej długości, nazywamy kwadratem.}
    \ita{Un quadrilatero con quattro angoli uguali e tutti i lati della stessa lunghezza si chiama quadrato.}
\end{definition}

\pol{Kwadraty (\emph{squares}) to dokładnie te prostokąty (\emph{rectangles}), które są równocześnie rombami.}
\ita{I quadrati (\emph{squares}) sono esattamente quei rettangoli (\emph{rectangles}) che sono anche rombi.}
\pol{Pole kwadratu o boku długości $a$ wynosi $a^2$, jego obwód to $4a$, zaś przekątna ma długość $a \sqrt 2$.}
\ita{L’area di un quadrato di lato $a$ è $a^2$, il suo perimetro è $4a$ e la diagonale ha lunghezza $a \sqrt 2$.}

\begin{definition}[\pol{latawiec}\ita{aquilone}]
\pol{\index{latawiec}}%
\ita{\index{aquilone}}%
\pol{\index{deltoid|see{latawiec}}}%
\ita{\index{deltoide|see{aquilone}}}%
    \pol{Czworokąt, który ma dwie pary sąsiednich boków równej długości, nazywamy latawcem.}
    \ita{Un quadrilatero con due coppie di lati consecutivi di uguale lunghezza si chiama aquilone.}    
\end{definition}

\pol{W Polsce bardziej popularne będzie określenie deltoid.}
\ita{In Polonia è più diffuso il termine deltoide.}
\pol{Latawce mają jedną przekątną, która dzieli je na dwa przystające trójkąty, w szczególności jego przekątne są prostopadłe.}
\ita{Gli aquiloni hanno una diagonale che li divide in due triangoli congruenti; in particolare le loro diagonali sono perpendicolari.}
\pol{Na latawcu, który ma dwa kąty proste naprzeciw siebie można opisać okrąg.}
\ita{Su un aquilone che ha due angoli retti opposti si può costruire un cerchio circoscritto.}
\pol{Romby są szczególnym przypadkiem latawców.}
\ita{I rombi sono un caso particolare di aquiloni.}

\pol{Czworokąty wklęsłe mają jeden kąt, którego miara wynosi między $\pi$ i $2\pi$ radianów.}
\ita{I quadrilateri concavi hanno un angolo la cui misura è compresa tra $\pi$ e $2\pi$ radianti.}
\pol{Wyróżniamy wśród nich rzutki (z angielskiego \emph{dart, arrowhead}), które mają oś symetrii albo równoważnie, dwie pary sąsiednich boków równej długości, tak jak latawce.}
\ita{Tra essi distinguiamo i dardi (in inglese \emph{dart, arrowhead}), che hanno un asse di simmetria oppure, equivalentemente, due coppie di lati consecutivi di uguale lunghezza, come gli aquiloni.}

\pol{W starożytnej Grecji czworokąty podzieli się na rozłączne klasy: kwadraty, podłużne prostokąty (po angielsku \emph{oblong}), niekwadratowe romby, romboidy (równoległoboki, które nie są ani rombami, ani prostokątami) oraz \emph{trapezia} (τραπέζιον znaczy dosłownie stolik), które nie pasowały do wcześniej wymienionych.}
\ita{Nell’antica Grecia i quadrilateri venivano suddivisi in classi disgiunte: quadrati, rettangoli allungati (in inglese \emph{oblong}), rombi non quadrati, romboidi (parallelogrammi che non sono né rombi né rettangoli) e \emph{trapezia} (τραπέζιον significa letteralmente “tavolino”), che non rientravano nelle categorie precedenti.}
\pol{Filozof neoplatoński Proklos zwany Diadochem napisze wpływowy komentarz do prac Euklidesa, gdzie wprowadzi trochę inny podział zgodnie z tym, co obmyśli około 100 lat p.n.e. Posejdonios z Rodos (nauczyciel Cycerona).}
\ita{Il filosofo neoplatonico Proclo, detto il Diadoco, scriverà un influente commento alle opere di Euclide, introducendo una classificazione leggermente diversa secondo quanto ideato circa 100 anni a.C. da Posidonio di Rodi (maestro di Cicerone).}

\pol{Czworokąt będzie równoległobokiem (kwadratem, podłużnym prostokątem, rombem, romboidem) albo nie (z parą boków równoległych, jako trapezium równoramienne lub ukośne albo bez niej -- trapezoidem).}
\ita{Un quadrilatero sarà un parallelogramma (quadrato, rettangolo allungato, rombo, romboide) oppure no (con una coppia di lati paralleli, come trapezio isoscele o obliquo, oppure senza, come trapezoide).}
\pol{Wszystkie języki europejskie odziedziczą klasyfikację czworokątów po Proklosie... poza angielskim, gdzie znaczenie trapezoidu i trapezium odwróci słownik wydany przez Charlesa Huttona w 1795 roku.}
\ita{Tutte le lingue europee erediteranno la classificazione dei quadrilateri da Proclo... tranne l’inglese, dove il significato di trapezoid e trapezium sarà invertito da un dizionario pubblicato da Charles Hutton nel 1795.}
\pol{Porządek przywróci się około 1875 roku, ale amerykańskiego angielskiego już nikt nie uratuje.}
\ita{L’ordine sarà ristabilito intorno al 1875, ma l’inglese americano non sarà più recuperato.}
