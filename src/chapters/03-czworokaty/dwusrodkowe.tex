\pol{\section{Czoworokąty dwuśrodkowe}}
\ita{\section{Quadrilateri bicentrici}}

\begin{definition}
\pol{Czworokąt, który jest jednocześnie wpisany w pewien okrąg i opisany na innym okręgu, nazywamy dwuśrodkowym. }%
\ita{Un quadrilatero che è contemporaneamente inscritto in un cerchio e circoscritto a un altro cerchio si chiama bicentrico. }%
\pol{\index{czworokąt!dwuśrodkowy}}%
\ita{\index{quadrilatero!bicentrico}}%
\end{definition}

\pol{Przykładami takich czworokątów są kwadraty, prostokątne latawce i niektóre równoramienne trapezy. }%
\ita{Esempi di tali quadrilateri sono quadrati, aquiloni rettangolari e alcuni trapezi isosceli. }%
\pol{\index{kwadrat}}%
\ita{\index{quadrato}}%
\pol{\index{latawiec}}%
\ita{\index{aquilone}}%
\pol{\index{trapez!równoramienny}}%
\ita{\index{trapezio!isoscele}}%
\pol{Ich pełną charakteryzację można uzyskać przez połączenie warunków \ref{prp_incircle} oraz \ref{prp_excircle}. }%
\ita{La loro completa caratterizzazione può essere ottenuta combinando le condizioni \ref{prp_incircle} e \ref{prp_excircle}. }%
\pol{Ale mamy też inne opisy. }%
\ita{Ma ci sono anche altre descrizioni. }%

\begin{proposition}
\pol{Niech $ABCD$ będzie czworokątem opisanym na okręgu $\Gamma$, który dotyka go w punktach $W$ (na odcinku $AB$), $X$ (na $BC$), $Y$ (na $CD$), $Z$ (na $AD$). }%
\ita{Sia $ABCD$ un quadrilatero circoscritto al cerchio $\Gamma$, che lo tocca nei punti $W$ (su $AB$), $X$ (su $BC$), $Y$ (su $CD$), $Z$ (su $AD$). }%
\pol{Wtedy następujące warunki są równoważne: }%
\ita{Allora le seguenti condizioni sono equivalenti: }%
\begin{itemize}
\item \pol{na czworokącie $ABCD$ można opisać okrąg,}
      \ita{si può descrivere un cerchio sul quadrilatero $ABCD$,}
\item \pol{odcinki $WY$ i $XZ$ są prostopadłe,}
      \ita{i segmenti $WY$ e $XZ$ sono perpendicolari,}
\item $|AW|/|BW| = |DY|/|CY|$,
\item $|AC|/|BD| = (|AW| + |CY|) / (|BX| + |DZ|)$,
\item \pol{równoległobok Varignona jest prostokątem.}
      \ita{Il parallelogramma di Varignon è un rettangolo.}
\pol{\index{równoległobok!Varignona}}%
\ita{\index{parallelogramma!Varignon}}%
\end{itemize}

\end{proposition}

\pol{Pole powierzchni czworokąta dwuśrodkowego o bokach długości $a, b, c, d$ wynosi $S = \sqrt{abcd}$, jest to prosty wniosek ze wzoru Brahmagupty \ref{brahmagupta_formula}. }%
\ita{L'area di un quadrilatero bicentrico con lati di lunghezza $a, b, c, d$ è $S = \sqrt{abcd}$, un semplice risultato dalla formula di Brahmagupta \ref{brahmagupta_formula}. }%
\pol{\index{wzór!Brahmagupty}}%
\ita{\index{formula!Brahmagupta}}%
\pol{Mamy $4r^2 \le S \le 2R^2$, a nawet }%
\ita{Abbiamo $4r^2 \le S \le 2R^2$, e persino }%
\begin{equation}
	S \le r^2 \left(1 + \sqrt{\left(\frac{2R}{r}\right)^2 + 1} \right),
\end{equation}
\pol{z równością wtedy i tylko wtedy, kiedy czworokąt jest prostokątnym latawcem. }%
\ita{con uguaglianza se e solo se il quadrilatero è un aquilone rettangolare. }%
\pol{Inne nierówności, których źródła nie podamy, to }%
\ita{Altre disuguaglianze, le cui fonti non citiamo, sono }%
\begin{equation}
	2 \sqrt {S} \le p \le r + \sqrt{r^2 + 4R^2},
\end{equation}
\pol{gdzie $p$ to połowa obwodu albo }%
\ita{dove $p$ è metà del perimetro oppure }%
\begin{equation}
	S \le \frac 1 6 \left(ab + ac + ad + bc + bd + cd\right).
\end{equation}
\pol{Nierówność }%
\ita{La disuguaglianza }%
\begin{equation}
	R \ge \sqrt 2 r,
\end{equation}
\pol{z równością tylko dla kwadratu, jest nietrywialna. Dowiedzie jej Fejes Tóth w 1948 roku. }%
\ita{con uguaglianza solo per il quadrato, è non banale. Dimostrata da Fejes Tóth nel 1948. }%
\index[persons]{Tóth, Fejes}%

\begin{theorem}[\pol{Fussa, 1792}\ita{Fuss, 1792}]
\pol{Niech $x$ oznacza odległość między środkami okręgu wpisanego i opisanego na czworokącie dwuśrodkowym. }%
\ita{Sia $x$ la distanza tra i centri del cerchio inscritto e del cerchio circoscritto del quadrilatero bicentrico. }%
\pol{Wtedy }%
\ita{Allora }%
\begin{equation}
	\frac{1}{(R-x)^2} + \frac{1}{(R+x)^2} = \frac{1}{r^2}.
\end{equation}
\pol{\index{twierdzenie!Fussa}}%
\ita{\index{teorema!Fuss}}%
\end{theorem}

\pol{Wyznaczając $x$ z twierdzenia Fussa i rozwiązując nierówność $x^2 \ge 0$ dochodzimy znowu do nierówności Tótha. }%
\ita{Determinando $x$ dal teorema di Fuss e risolvendo la disuguaglianza $x^2 \ge 0$ si arriva nuovamente alla disuguaglianza di Tóth. }%
\pol{Aż dziwne, że nikt tego wcześniej nie zrobił. }%
\ita{È sorprendente che nessuno l'abbia fatto prima. }%
\pol{Nicolaus Fuss był szwajcarskim matematykiem, który spędził większość swego życia w Rosji. }%
\ita{Nicolaus Fuss era un matematico svizzero che trascorse gran parte della sua vita in Russia. }%
\index[persons]{Fuss, Nicolaus}%

\begin{proposition}
\pol{Punkt przecięcia przekątnych, środek okręgu wpisanego i środek okręgu opisanego na czworokącie dwuśrodkowym są współliniowe. }%
\ita{Il punto di intersezione delle diagonali, il centro del cerchio inscritto e il centro del cerchio circoscritto del quadrilatero bicentrico sono collineari. }%
\pol{\index{współliniowy}}%
\ita{\index{collineare}}%
\end{proposition}

\begin{proposition}
\pol{Niech $ABCD$ będzie czworokątem dwuśrodkowym, zaś $O$ środkiem okręgu opisanego. }%
\ita{Sia $ABCD$ un quadrilatero bicentrico e $O$ il centro del cerchio circoscritto. }%
\pol{Wtedy środki okręgów wpisanych w trójkąty $\triangle OAB$, $\triangle OBC$, $\triangle OCD$, $\triangle ODA$ leżą na jednym okręgu. }%
\ita{Allora i centri dei cerchi inscritti nei triangoli $\triangle OAB$, $\triangle OBC$, $\triangle OCD$, $\triangle ODA$ giacciono su un unico cerchio. }%
\end{proposition}

\pol{Wreszcie Klamkin pokazał w 1967 roku, że }%
\ita{Infine Klamkin dimostrò nel 1967 che }%
\begin{proposition}
\pol{Niech $p, q$ będą długościami przekątnych czworokąta dwuśrodkowego o bokach długości $a$, $b$, $c$, $d$. }%
\ita{Siano $p, q$ le lunghezze delle diagonali di un quadrilatero bicentrico con lati di lunghezza $a$, $b$, $c$, $d$. }%
\pol{Wtedy }%
\ita{Allora }%
\begin{equation}
	8 pq \le (a + b + c + d)^2
\end{equation}
\end{proposition}

\begin{theorem}[\pol{Ponceleta, małe}\ita{di Poncelet, piccolo}]
\pol{Niech trójkąt $A_0 A_1 A_2$ będzie wpisany w~stożkową $C$ oraz opisany na stożkowej $D$. }%
\ita{Sia il triangolo $A_0 A_1 A_2$ inscritto nella conica $C$ e circoscritto alla conica $D$. }%
\pol{Wtedy każdy punkt $B_0$ stożkowej $C$ jest wierzchołkiem dokładnie jednego trójkąta $B_0 B_1 B_2$ wpisanego w~stożkową $C$ oraz opisanego na stożkowej $D$. }%
\ita{Allora ogni punto $B_0$ della conica $C$ è il vertice di un unico triangolo $B_0 B_1 B_2$ inscritto in $C$ e circoscritto a $D$. }%
\pol{\index{twierdzenie!Poneceleta, małe i duże}}%
\ita{\index{teorema!Poncelet, piccolo e grande}}%
\end{theorem}

\pol{Jest też wielkie twierdzenie Ponceleta, udowodnione przez, jak niezbyt trudno się domyślić, Victora Ponceleta \cite[s. 311-317]{poncelet_1865} (wg Bogdańskiej, Neugebauera w 1813 roku, wg angielskiej Wikipedii w~1822 roku). }%
\ita{Esiste anche il grande teorema di Poncelet, dimostrato, come non è difficile intuire, da Victor Poncelet \cite[s. 311-317]{poncelet_1865} (secondo Bogdańska, Neugebauer nel 1813, secondo Wikipedia inglese nel 1822). }%
\index[persons]{Poncelet, Victor}%

\begin{theorem}[\pol{Ponceleta, wielkie}\ita{Poncelet, grande}]
\label{big_poncelet}%
\pol{Niech $C$ i $D$ będą dwiema stożkowymi, zaś $A_0, A_1, \ldots, A_{n-1}$ takimi punktami na stożkowej $C$, że proste $A_0A_1$, $A_1A_2$, \ldots, $A_{n-1}A_0$ są styczne do stożkowej $D$. }%
\ita{Siano $C$ e $D$ due coniche, e $A_0, A_1, \ldots, A_{n-1}$ punti su $C$ tali che le rette $A_0A_1$, $A_1A_2$, \ldots, $A_{n-1}A_0$ siano tangenti a $D$. }%
\pol{Wtedy dla każdego punktu $B_0$ na stożkowej $C$ istnieją różne punkty $B_1, \ldots, B_{n-1}$, też na stożkowej $C$, że proste $B_0B_1$, $B_1B_2$, \ldots, $B_{n-1}B_0$ są styczne do stożkowej $D$. }%
\ita{Allora per ogni punto $B_0$ su $C$ esistono punti distinti $B_1, \ldots, B_{n-1}$, anch'essi su $C$, tali che le rette $B_0B_1$, $B_1B_2$, \ldots, $B_{n-1}B_0$ siano tangenti a $D$. }%
\end{theorem}

\ita{La dimostrazione si può trovare ad esempio in Akopyan, Zaslavsky \cite[p. 93, 61, 67, 115, 124]{akopyan_2007}. }%
