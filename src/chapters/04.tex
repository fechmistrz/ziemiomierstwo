\begin{definition}[wielokąt]
\pol{\index{wielokąt!pięciokąt}}%
\ita{\index{poligono!pentagono}}%
\pol{Łamaną zamkniętą (sumę odcinków łączących kolejne punkty, aż do początkowego) bez samoprzecięć razem z jej wnętrzem nazywamy wielokątem.}
\ita{Una spezzata chiusa (somma di segmenti che uniscono punti consecutivi fino a tornare al punto iniziale), senza autointersezioni, insieme al suo interno si chiama poligono.}

\pol{Jarosław Górnicki opisze w $\Delta_{20}^{12}$ twierdzenie Edwarda Kasnera z 1903 roku:}
\ita{Jarosław Górnicki descrive in $\Delta_{20}^{12}$ un teorema di Edward Kasner del 1903:}
\index[persons]{Kasner, Edward}%

\begin{proposition}
\pol{Dane są parami różne punkty $S_1, S_2, \ldots, S_n$ ($n \ge 3$), które są środkami kolejnych boków pewnego wielokąta.
Jeśli $n = 2k + 1$ jest nieparzyste, to istnieje dokładnie jeden taki wielokąt.
W przeciwnym przypadku ($n = 2k$ jest parzyste) taki wielokąt istnieje wtedy i tylko wtedy, gdy zbiory punktów $S_1, S_3, \ldots$ o~indeksach nieparzystych oraz $S_2, S_4, \ldots$ o~indeksach parzystych mają ten sam środek ciężkości; wtedy każdy punkt płaszczyzny może być wierzchołkiem $W_1$ szukanego wielokąta.}
\ita{Siano dati punti distinti a due a due $S_1, S_2, \ldots, S_n$ ($n \ge 3$), che sono i punti medi dei lati consecutivi di un poligono.
Se $n = 2k + 1$ è dispari, allora esiste esattamente un tale poligono.
Nel caso contrario ($n = 2k$ è pari), tale poligono esiste se e solo se gli insiemi di punti $S_1, S_3, \ldots$ con indici dispari e $S_2, S_4, \ldots$ con indici pari hanno lo stesso baricentro; in tal caso ogni punto del piano può essere un vertice $W_1$ del poligono cercato.}
\end{proposition}

\pol{Twierdzenie Varignona \ref{theorem_varignon} mówi, że środki boków czworokąta wyznaczają równoległobok, którego przekątne połowią się -- a więc spełnia warunek o dwóch środkach ciężkości.}
\ita{Il teorema di Varignon \ref{theorem_varignon} afferma che i punti medi dei lati di un quadrilatero determinano un parallelogramma le cui diagonali si bisecano -- quindi soddisfa la condizione sui due baricentri.}

\begin{proposition}
\pol{Niech $n > 4$ będzie liczbą boków $n$-kąta foremnego.
Wtedy największą liczbą przekątnych, które przechodzą przez jeden punkt wewnątrz wielokąta różny od jego środka jest:}
\ita{Sia $n > 4$ il numero di lati di un $n$-gono regolare.
Allora il massimo numero di diagonali che passano per uno stesso punto interno al poligono, diverso dal suo centro, è:}
\begin{itemize}
\item \pol{$2$, jeżeli $n$ jest nieparzyste,}
      \ita{$2$, se $n$ è dispari,}
\item \pol{$3$, jeżeli $n$ jest parzyste, ale niepodzielne przez $6$,}
      \ita{$3$, se $n$ è pari ma non divisibile per $6$,}
\item \pol{$5$, jeżeli $n$ jest podzielne przez $6$, ale niepodzielne przez $30$,}
      \ita{$5$, se $n$ è divisibile per $6$ ma non per $30$,}
\item \pol{$7$, jeżeli $n$ jest podzielne przez $30$;}
      \ita{$7$, se $n$ è divisibile per $30$;}
\end{itemize}
\pol{z dwoma wyjątkami: $2$ dla sześciokąta i $4$ dla dwunastokąta (zamiast $5$ jak mówi trzeci warunek).}
\ita{con due eccezioni: $2$ per l'esagono e $4$ per il dodecagono (invece di $5$ come nel terzo caso).}
\end{proposition}
