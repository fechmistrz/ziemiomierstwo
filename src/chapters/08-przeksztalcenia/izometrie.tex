\pol{\section{Izometrie}}
\ita{\section{Isometrie}}

\todofoot{\pol{Izometrie Coxetera, s. 29-36, 47 -- Hjelmslev}\ita{Isometrie in Coxeter, pp. 29--36, 47 -- Hjelmslev}}

\begin{definition}
    \pol{Odwzorowanie płaszczyzny w siebie, które zachowuje długość każdego odcinka, nazywamy izometrią płaszczyzny albo krócej izometrią.}
    \ita{Un'applicazione del piano in sé che conserva la lunghezza di ogni segmento si chiama isometria del piano, o più brevemente isometria.}
\end{definition}

\pol{Izometrie to odwzorowania, które zachowują odległości.}
\ita{Le isometrie sono trasformazioni che conservano le distanze.}
\pol{Izometrie można składać, każda jest bijekcją i funkcja odwrotna do niej też jest izometrią.}
\ita{Le isometrie possono essere composte, ciascuna è una biiezione e anche la sua funzione inversa è un'isometria.}
\pol{Patrz do Coxetera \cite[s. 45-52, 56-63]{coxeter_1967}.}
\ita{Si veda Coxeter \cite[pp.~45--52, 56--63]{coxeter_1967}.}

\begin{proposition}
\label{delta_1997_9_start}%
    \pol{Izometria, która ma punkt stały, przenosi wszystkie okręgi wokół tego punktu na siebie.}
    \ita{Un'isometria che possiede un punto fisso manda in sé tutte le circonferenze aventi quel punto come centro.}
\end{proposition}

\begin{proposition}
    \pol{Izometria, która ma dwa punkty stałe, ma ich nieskończenie wiele (wszystkie punkty prostej przechodzącej przez dwa punkty).}
    \ita{Un'isometria che possiede due punti fissi ne possiede infinitamente molti (tutti i punti della retta passante per essi).}
\end{proposition}

\begin{proposition}
    \pol{Izometria, która ma trzy niewspółliniowe punkty stałe, jest odwzorowaniem tożsamościowym i~wszystkie punkty są jej punktami stałymi.}
    \ita{Un'isometria che possiede tre punti fissi non allineati è l'identità e tutti i punti sono suoi punti fissi.}
\end{proposition}

\begin{proposition}
\label{delta_1997_9_end}%
    \pol{Izometria, której wartości w trzech niewspółliniowych punktach są znane, jest jednoznacznie wyznaczona przez nie.}
    \ita{Un'isometria i cui valori in tre punti non allineati sono noti è univocamente determinata da essi.}
\end{proposition}

\pol{Cztery powyższe fakty to początek przepisu, jaki Marek Kordos poda w $\Delta_{97}^9$ na wynik związany z paradoksalnym rozkładem kuli.}
\ita{I quattro fatti precedenti costituiscono l'inizio del procedimento che Marek Kordos presenterà in $\Delta_{97}^9$ per un risultato legato alla decomposizione paradossale della sfera.}

\pol{Mamy też:}
\ita{Abbiamo inoltre:}

\begin{theorem}[Hjelmsleva]
    \pol{Jeżeli wszystkie punkty $P$ prostej przekształcają się izometrycznie na wszystkie punkty $P'$ innej prostej, to wszystkie środki odcinków $PP'$ są różne i współliniowe, albo wszystkie sprowadzają się do jednego punktu.}
    \ita{Se tutti i punti $P$ di una retta vengono trasformati isometricamente in tutti i punti $P'$ di un'altra retta, allora i punti medi dei segmenti $PP'$ sono tutti distinti e allineati, oppure coincidono tutti in un unico punto.}
\end{theorem}

\pol{Eves \cite[s. 311]{eves1_1972} wspomni to twierdzenie już po zdefiniowaniu geometrii Łobaczewskiego.}
\ita{Eves \cite[p.~311]{eves1_1972} menziona questo teorema subito dopo aver definito la geometria di Lobachevskij.}
