\pol{\subsection{Klasyfikacja izometrii}}
\ita{\subsection{Classificazione delle isometrie}}

\begin{theorem}[\pol{Chasles'a}\ita{di Chasles}, 1830]
\pol{\index{twierdzenie!Chasles'a}}%
\ita{\index{teorema!di Chasles}}%
\label{theorem_chasles}
    \pol{Każda izometria płaszczyzny jest złożeniem co najwyżej trzech symetrii osiowych.}
    \ita{Ogni isometria del piano è composta al massimo da tre simmetrie assiali.}
\end{theorem}

\begin{definition}[symetria z poślizgiem]
    \pol{Niech $p$ będzie prostą, zaś $w$ wektorem równoległym do niej.}
    \ita{Sia $p$ una retta e $w$ un vettore parallelo ad essa.}
    \pol{Złożenie $T_w \circ S_p$ nazywamy symetrią osiową z poślizgiem.}
    \ita{La composizione $T_w \circ S_p$ si chiama simmetria assiale traslante.}
\end{definition}

\pol{Wektor $w$ nie musi być równoległy do prostej $p$, ale jeśli założymy, że tak jest, to nie musimy pamiętać kolejności składania, bo mamy równość $T_w \circ S_p = S_p \circ T_w$.}
\ita{Il vettore $w$ non deve essere parallelo alla retta $p$, ma se assumiamo che lo sia, non dobbiamo ricordare l'ordine delle composizioni, poiché $T_w \circ S_p = S_p \circ T_w$.}

\begin{corollary}
    \pol{Każda izometria płaszczyzny jest przesunięciem, obrotem albo symetrią z poślizgiem, przy czym orientację zachowują dokładnie pierwsze dwa typy izometrii.}
    \ita{Ogni isometria del piano è una traslazione, una rotazione o una simmetria traslante, e solo i primi due tipi conservano l'orientamento.}
\end{corollary}

\pol{Klasyfikacja izometrii płaszczyzny eliminuje paradoks Banacha-Tarskiego z płaszczyzny, bo nie ma podgrupy wolnej w grupie jej izometrii.}
\ita{La classificazione delle isometrie del piano elimina il paradosso di Banach-Tarski dal piano, poiché non esiste un sottogruppo libero nel gruppo delle sue isometrie.}
\pol{Oto szczegóły.}
\ita{Ecco i dettagli.}
\pol{Jeśli $f$ jest izometrią, to $f^2$ jest przesunięciem albo obrotem.}
\ita{Se $f$ è un'isometria, allora $f^2$ è una traslazione o una rotazione.}
\pol{Gdy $g$ też jest izometrią, to $f^2g^2f^{-2}g^{-2}$ jest zawsze przesunięciem; dla dowodu tego należy rozpatrzeć cztery przypadki, z których najtrudniejszy to para obrót-obrót.}
\ita{Se $g$ è anch'esso un'isometria, allora $f^2 g^2 f^{-2} g^{-2}$ è sempre una traslazione; per dimostrarlo bisogna considerare quattro casi, il più difficile è la coppia rotazione-rotazione.}
\pol{Pomocne może być zauważenie, że jeśli mamy dwa obroty $f$, $g$ o różnych środkach i ich złożenie jest obrotem, to $fg$ i $gf$ są obrotami o ten sam kąt, ale różnych środkach.}
\ita{Può essere utile notare che se abbiamo due rotazioni $f$, $g$ con centri diversi e la loro composizione è una rotazione, allora $fg$ e $gf$ sono rotazioni dello stesso angolo ma con centri diversi.}

\pol{Istnieje podobna klasyfikacja izometrii przestrzeni $\mathbb R^3$, jak u Coxetera \cite[s. 113-122]{coxeter_1967}, Evesa \cite[s. 148]{eves1_1972}:}
\ita{Esiste una classificazione simile delle isometrie dello spazio $\mathbb R^3$, come in Coxeter \cite[s. 113-122]{coxeter_1967}, Eves \cite[s. 148]{eves1_1972}:}

\begin{proposition}
    \pol{Każda izometria przestrzeni $\mathbb R^3$
    zachowuje orientację (i jest odwzorowaniem tożsamościowym; przesunięciem; obrotem wokół prostej albo ruchem śrubowym, czyli obrotem wokół prostej z przesunięciem)
    albo nie (i jest odbiciem względem płaszczyzny; odbiciem z poślizgiem; obrotem niewłaściwym, czyli obrotem wokół prostej z odbiciem w płaszczyźnie prostopadłej do niej).}
    \ita{Ogni isometria dello spazio $\mathbb R^3$
    conserva l'orientamento (ed è l'identità, una traslazione, una rotazione attorno a una retta o un moto elicoidale, cioè rotazione attorno a una retta con traslazione)
    oppure no (ed è una riflessione rispetto a un piano, una simmetria traslante o una rotazione impropria, cioè rotazione attorno a una retta con riflessione in un piano perpendicolare ad essa).}
\end{proposition}
