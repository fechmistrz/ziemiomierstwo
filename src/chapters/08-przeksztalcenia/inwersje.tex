\pol{\section{Inwersja względem okręgu}}
\ita{\section{Inversione rispetto a una circonferenza}}

\pol{Inwersję odkryje wiele osób niezależnie od siebie, na przykład Jakob Steiner (1824), Lambert Adolphe Jacques Quetelet (1825), Giusto Bellavitis (1836), John William Stubbs i John Kells Ingram (1842?) albo lord Kelvin (właściwie William Thomson 1845).}
\ita{L'inversione sarà scoperta indipendentemente da molte persone, per esempio Jakob Steiner (1824), Lambert Adolphe Jacques Quetelet (1825), Giusto Bellavitis (1836), John William Stubbs e John Kells Ingram (1842?) oppure lord Kelvin (in realtà William Thomson, 1845).}

    \ita{Sia data una circonferenza $\Gamma$ di centro $O$ e raggio $r$ e un punto $P$, distinto da $O$:}

    \pol{Wtedy jedyny punkt $P'$ leżący na półprostej $OP$ taki, że $|OP| \cdot |OP'| = r^2$ nazywamy inwersją punktu $P$ względem okręgu $\Gamma$.}
    \ita{Allora l'unico punto $P'$ appartenente alla semiretta $OP$ tale che $|OP| \cdot |OP'| = r^2$ si chiama inversione del punto $P$ rispetto alla circonferenza $\Gamma$.}
\end{definition}

\pol{Czasami do płaszczyzny dokłada się $\infty$, punkt w nieskończoności: przyjmujemy, że inwersja odwzorowuje $\infty$ na punkt $O$, zaś punkt $O$ na $\infty$; czyni to ją automorfizmem płaszczyzny, który przenosi wnętrze okręgu na jego zewnętrze.}
\ita{Talvolta si aggiunge al piano il punto $\infty$, il punto all'infinito: si assume che l'inversione mandi $\infty$ nel punto $O$ e il punto $O$ in $\infty$; ciò la rende un automorfismo del piano che trasforma l'interno della circonferenza nel suo esterno.}

\pol{O inwersji piszą Coxeter \cite[s. 93-101, 107-112]{coxeter_1967}.}
\ita{Coxeter scrive dell'inversione in \cite[pag. 93-101, 107-112]{coxeter_1967}.}
\pol{Cyrkiel i linijka wystarczają do znalezienia inwersji dowolnego punktu.}
\ita{Compasso e riga sono sufficienti per trovare l'inversione di un punto arbitrario.}

\pol{Bez straty ogólności załóżmy, że $P$ leży na zewnątrz okręgu.}
\ita{Senza perdita di generalità supponiamo che $P$ si trovi all'esterno della circonferenza.}

\pol{Wtedy jego obraz znajduje się na przecięciu prostej $OP$ oraz cięciwy, która łączy punkty styczności stycznych do okręgu poprowadzonych z punktu $P$.}
\ita{Allora la sua immagine si trova all'intersezione della retta $OP$ e della corda che unisce i punti di tangenza delle tangenti alla circonferenza tracciate dal punto $P$.}

\ita{Coxeter \cite[pag. 98]{coxeter_1967} menziona due dispositivi meccanici che tracciano l'immagine inversiva di una traiettoria: l'inversore di Peaucellier, più precisamente di Peaucellier-Lipkin, e poco dopo anche il meccanismo di Hart\footnote{In inglese rispettivamente Peaucellier's cell e Hart's linkage.} destinato allo stesso scopo.}

\begin{proposition}
    \pol{Inwersje są kątowierne: każdy kąt ma taką samą miarę jak jego obraz względem inwersji.}
    \ita{Le inversioni sono conformi: ogni angolo ha la stessa ampiezza della sua immagine tramite inversione.}
\end{proposition}

\begin{proposition}
    \pol{Inwersje przenoszą okręgi ortogonalne na okręgi ortogonalne, włączając proste jako przypadki szczególne.}
    \ita{Le inversioni trasformano circonferenze ortogonali in circonferenze ortogonali, includendo le rette come casi particolari.}
\end{proposition}

\pol{To, że inwersje przenoszą proste oraz okręgi na proste oraz okręgi będzie wiedzieć Robert Simson w 1749 roku, kiedy odtworzy zagubioną pracę ,,Plane Loci'' Apoloniusza na podstawie komentarza Pappusa.}
\ita{Robert Simson saprà già nel 1749 che le inversioni trasformano rette e circonferenze in rette e circonferenze, quando ricostruirà l'opera perduta ``Plane Loci'' di Apollonio sulla base del commento di Pappo.}

\pol{Wiele trudnych problemów geometrycznych staje się prostsze po zastosowaniu inwersji.}
\ita{Molti problemi geometrici difficili diventano più semplici dopo l'applicazione dell'inversione.}

\pol{Na przykład środek okręgu przed i po odwróceniu są współliniowe ze środkiem inwersji, skąd wynika, że prosta Eulera trójkąta między punktami styczności okręgów dopisanych przechodzi przez pewne ważne punkty.}
\ita{Per esempio, il centro di una circonferenza prima e dopo l'inversione è allineato con il centro dell'inversione; ne consegue che la retta di Eulero del triangolo formato dai punti di tangenza delle circonferenze exinscritte passa per alcuni punti notevoli.}

\pol{Proste prostopadłe do $PP'$, które przechodzą przez jeden z punktów, są biegunowymi drugiego końca odcinka (bieguna).}
\ita{Le rette perpendicolari a $PP'$ che passano per uno dei punti sono le polari dell'altro estremo del segmento (il polo).}

\begin{proposition}
    \pol{Każde przekształcenie płaszczyzny inwersyjnej zachowujące okręgi jest albo podobieństwem, albo złożeniem inwersji i izometrii.}
    \ita{Ogni trasformazione del piano inversivo che preserva le circonferenze è o una similitudine oppure una composizione di un'inversione e un'isometria.}
\end{proposition}

\pol{Stąd wynika, że każde takie przekształcenie jest złożeniem co najwyżej czterech inwersji.}
\ita{Ne segue che ogni tale trasformazione è composizione di al più quattro inversioni.}
\pol{Coxeter \cite[s. 108]{coxeter_1967} pisze o trzech rodzajach przekształceń: eliptycznych, parabolicznych i hiperbolicznych, co sprowadza się w szczególnym przypadku do obrotów, translacji i dylatacji.}
\ita{Coxeter \cite[pag. 108]{coxeter_1967} scrive di tre tipi di trasformazioni: ellittiche, paraboliche e iperboliche, che in casi particolari corrispondono a rotazioni, traslazioni e dilatazioni.}

\begin{proposition}
    \pol{Złożenie czterech inwersji, które nie jest złożeniem dwóch inwersji, nazywamy przekształceniem loksodromicznym.}
    \ita{Una composizione di quattro inversioni che non sia una composizione di due inversioni si chiama trasformazione lossodromica.}
\end{proposition}

\pol{Nazwa ma greckie korzenie: loksodroma to linia przecinająca wszystkie południki pod tym samym kątem (λοξός ukośny, δρόμος linia).}
\ita{Il nome ha origini greche: la lossodroma è una linea che interseca tutti i meridiani con lo stesso angolo (λοξός obliquo, δρόμος linea).}
