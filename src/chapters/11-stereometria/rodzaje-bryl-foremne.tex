\pol{\subsection{Wielościany foremne}}
\ita{\subsection{Poliedri regolari}} % regular

\begin{definition}
    \pol{Wielościan foremny to taki wielościan, którego grupa izometrii działa przechodnio na zbiorze flag.}
    \ita{Un poliedro regolare è un poliedro il cui gruppo di isometrie agisce transitivamente sull'insieme delle bandiere.}
\end{definition}

\pol{Niektóre osoby, przede wszystkim te nieznające definicji flagi, będą używać innych definicji, z~czego najpowszechniejsza będzie głosiła, że wielościan foremny ma mieć identyczne foremne ściany oraz identyczne naroża.}
\ita{Alcune persone, soprattutto quelle che non conoscono la definizione di bandiera, utilizzeranno altre definizioni; la più diffusa affermerà che un poliedro regolare deve avere facce regolari congruenti e vertici tutti uguali.}

\begin{proposition}
    \pol{Istnieje czternaście wielościanów foremnych:}
    \ita{Esistono quattordici poliedri regolari:}
    \begin{itemize}
        \item \pol{pięć platońskich,}\ita{cinque platonici,}
        \item \pol{cztery gwieździste (Keplera-Poinsota),}\ita{quattro stellati (di Keplero-Poinsot),}
        \item \pol{pięć złożeń.}\ita{cinque composti.}
    \end{itemize}
\end{proposition}
\pol{Niektórzy nie uznają złożeń za wielościany foremne.}
\ita{Alcuni non considerano i poliedri composti come poliedri regolari.}
