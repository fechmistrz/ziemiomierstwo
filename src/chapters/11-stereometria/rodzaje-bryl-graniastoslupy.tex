\pol{\subsection{Graniastosłupy i ogólniej, pryzmatoidy}}
\ita{\subsection{Prismi e, più in generale, prismoidi}}
\pol{Pryzmatoid to bryła, której wierzchołki leżą na dwóch równoległych płaszczyznach.}
\ita{Un prismoide è un solido i cui vertici giacciono su due piani paralleli.}

\begin{example}[\pol{ostrosłup}\ita{piramide}]
    \pol{Każdy ostrosłup jest pryzmatoidem: na jednej płaszczyźnie leży jego podstawa, na drugiej wierzchołek.}
    \ita{Ogni piramide è un prismoide: su un piano giace la sua base, sull'altro il vertice.}
\end{example}

\begin{example}[\pol{klin}\ita{cuneo}]
    \pol{Klin, wielościan, którego podstawa jest prostokątem, zaś ściany boczne to dwa trójkąty równoramienne i dwa trapezy, też jest pryzmatoidem.}
    \ita{Anche il cuneo, un poliedro la cui base è un rettangolo e le cui facce laterali sono due triangoli isosceli e due trapezi, è un prismoide.}
\end{example}

\pol{Nie wiadomo, czemu ten kształt ma w ogóle swoją nazwę.}

\begin{example}[\pol{graniastosłup}\ita{prisma}]
    \pol{Graniastosłup, gdzie płaszczyzny są przystającymi wielokątami, zaś ściany boczne to równoległoboki.}
    \ita{Un prisma, in cui le due basi sono poligoni congruenti e le facce laterali sono parallelogrammi.}
\end{example}

\begin{proposition}
    \pol{Objętość pryzmatoidu wyraża wzór}
    \ita{Il volume di un prismoide è dato dalla formula}
    \begin{equation}
        V = \frac 1 6 h (S_l + 4S_m + 4S_h),
    \end{equation}
    \pol{gdzie $h$ oznacza wysokość (odległość między płaszczyznami podstaw), $S_l$ pole podstawy dolnej, $S_h$ pole podstawy górnej, zaś $S_m$ to pole przekroju poprzecznego płaszczyzną, która leży w połowie między podstawami.}
    \ita{dove $h$ è l'altezza (la distanza tra i piani delle basi), $S_l$ è l'area della base inferiore, $S_h$ l'area della base superiore e $S_m$ l'area della sezione ottenuta con un piano situato a metà strada tra le basi.} % eves s. 40
\end{proposition}

\begin{proof}
    \pol{Pole przekroju poprzecznego pryzmatoidu stanowi funkcję kwadratową wysokości.}
    \ita{L'area della sezione trasversale di un prismoide è una funzione quadratica dell'altezza.}
    \pol{Wzór wynika z całkowania metodą Simsona, która daje dokładny wynik dla wielomianów stopnia co najwyżej trzeciego.}
    \ita{La formula deriva dall'integrazione mediante il metodo di Simpson, che fornisce un risultato esatto per i polinomi di grado al più tre.}
\end{proof}
