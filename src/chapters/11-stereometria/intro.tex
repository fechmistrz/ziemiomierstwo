\pol{\section{Preludium}}
\ita{\section{Preludio}}

\pol{Panuje nieziemski bałagan, jeśli chodzi o terminologię dla wielościanów lub, ogólniej, wielotopów. By uniknąć ryzyka zgubienia się, podamy jedynie podstawowe definicje, a nie twierdzenia.}
\ita{Regna una confusione indescrivibile nella terminologia relativa ai poliedri o, più in generale, ai politopi. Per evitare di perderci, presenteremo soltanto le definizioni fondamentali, senza soffermarci sui teoremi.}

\pol{Z każdym wielościanem związana jest grupa jego izometrii: takich przekształceń (odbić, obrotów) bryły na siebie, która zachowuje odległości.}
\ita{A ogni poliedro è associato il suo gruppo di isometrie: le trasformazioni del poliedro in sé stesso (riflessioni, rotazioni) che preservano le distanze.}

\pol{Tylko odwzorowanie tożsamościowe przenosi flagę na siebie.}
\ita{Solo l'identità manda una bandiera in sé stessa.}

\begin{definition}[\pol{symetrie wielościanów}\ita{simmetrie dei poliedri}]
    \pol{Wielościan nazywamy izohedralnym (izotoksalnym, izogonalnym), jeśli jego grupa izometrii działa tranzytywnie na zbiorze ścian (krawędzi, wierzchołków).}
    \ita{Un poliedro si dice isoedrale (isotossale, isogonale) se il suo gruppo di isometrie agisce transitivamente sull'insieme delle facce (degli spigoli, dei vertici).}
\end{definition}

\begin{example}
    \pol{Wielościany platońskie są izogonalne, izotoksalne i izohedralne jednocześnie.}
    \ita{I poliedri platonici sono contemporaneamente isogonali, isotossali e isoedrali.}
\end{example}

\pol{Edmund Hess, Max Brückner, a później także Branko Grünbaum będą badać wielościany szlachetne, czyli takie, które są izohedralne oraz izogonalne.}
\ita{Edmund Hess, Max Brückner e più tardi anche Branko Grünbaum studieranno i poliedri nobili, cioè quelli che sono sia isoedrali sia isogonali.}
\pol{Wiadomo, że wielościany platońskie i Keplera-Poinsota są szlachetne; poza nimi także dwusfenoidy i ogólniej sfenoidi.}
\ita{Si sa che i poliedri platonici e quelli di Keplero-Poinsot sono nobili; inoltre lo sono i disfenoidi e più in generale gli sfenoidi.}
% https://en.wikipedia.org/wiki/Noble_polyhedron

\begin{example}
    \pol{Wielościany Catalana, podwójne ostrosłupy i latawcościany są izohedralne. Bryły dualne do nich, czyli wielościany archimedesowe, graniastosłupy i antygraniastosłupy, są izogonalne.}
    \ita{I poliedri catalani, le bipiramidi e i deltoedri sono isoedrali. I loro duali, cioè i poliedri archimedei, i prismi e gli antiprismi, sono isogonali.}
\end{example}

\pol{Wielościan dualny do izohedralnego jest zawsze izogonalny.}
\ita{Il duale di un poliedro isoedrale è sempre isogonale.}

\pol{Wielościany foremne mają identyczne foremne ściany i identyczne naroża.}
\ita{I poliedri regolari hanno facce regolari congruenti e vertici tutti uguali.}

\pol{Wielościany archimedesowe są wypukłe, mają foremne ściany (dwóch lub trzech rodzajów) i są izogonalne, ale nie izohedralne.}
\ita{I poliedri archimedei sono convessi, hanno facce regolari (di due o tre tipi) e sono isogonali, ma non isoedrali.}

\pol{Wielościany Catalana są dualne do archimedesowych.}
\ita{I poliedri catalani sono duali dei poliedri archimedei.}

\pol{Wielościany jednorodne mają foremne ściany oraz są wierzchołkowo przechodnie.}
\ita{I poliedri uniformi hanno facce regolari e sono transitivi sui vertici.}

\pol{Wielościan Johnsona(-Zalgallera) to wypukły wielościan, którego ściany są wielokątami foremnymi.}
\ita{Un poliedro di Johnson (o di Johnson-Zalgaller) è un poliedro convesso le cui facce sono poligoni regolari.}

\pol{Przez analogię do wielokątów wypukłych mamy:}
\ita{Per analogia con i poligoni convessi, abbiamo:}

\begin{definition}[\pol{wielościan wypukły}\ita{poliedro convesso}]
    \pol{Obszar skończony przestrzeni będący przekrojem półprzestrzeni.}
    \ita{Una regione limitata dello spazio ottenuta come intersezione di semispazi.}
    \pol{Część wycięta z płaszczyzny przez pozostałe jest wielokątem, nazywamy go ścianą.}
    \ita{La parte ritagliata su una delle superfici piane dalle altre è un poligono, che chiamiamo faccia.}
    \pol{Wspólny bok dwóch ścian to krawędź, wspólny koniec dwóch krawędzi to wierzchołek.}
    \ita{Il lato comune di due facce è uno spigolo, mentre l'estremo comune di due spigoli è un vertice.}
\end{definition}

\pol{Najbardziej znanymi są ostrosłupy i graniastosłupy.}
\ita{I più noti sono le piramidi e i prismi.}
\pol{Oprócz nich opowiemy także o podwójnych ostrosłupach, latawcościanach oraz różnych mniej lub bardziej foremnych bryłach.}
\ita{Oltre a essi parleremo anche di bipiramidi, deltoedri e di vari solidi più o meno regolari.}
\pol{Natomiast nie podamy definicji wielościanu (niewypukłego), ponieważ ma powszechnie akceptowanej tegoż.}
\ita{Non daremo invece una definizione di poliedro (non convesso), poiché non esiste una definizione universalmente accettata.}

\begin{definition}[\pol{flaga}\ita{bandiera}]
    \pol{Zbiór złożony ze ściany, krawędzi oraz wierzchołka takich, że każde leży na poprzednim, nazywamy flagą.}
    \ita{Una terna costituita da una faccia, uno spigolo e un vertice, tali che ciascun elemento appartenga al precedente, si chiama bandiera.}
\end{definition} % https://en.wikipedia.org/wiki/Flag_(geometry)
