\pol{\subsection{Czworościany}}
\ita{\subsection{Tetraedri}} % tetrahedron

\pol{W każdym czworościanie istnieje wierzchołek, przy którym trzy kąty płąskie są ostre.}
\ita{In ogni tetraedro esiste un vertice nel quale i tre angoli piani sono acuti.} % https://www.deltami.edu.pl/2013/12/katy-trojscienne/

\begin{proposition}
    \pol{Niech $ABCD$ będzie czworościanem.}
    \ita{Sia $ABCD$ un tetraedro.}
    \pol{Wtedy następujące warunki są równoważne:}
    \ita{Allora le seguenti condizioni sono equivalenti:}
    \begin{enumerate}
        \item \pol{Wszystkie ściany są przystające.}
              \ita{Tutte le facce sono congruenti.}
        \item \pol{Wszystkie ściany są trójkątami ostrokątnymi o takim samym promieniu okręgu opisanego.}
              \ita{Tutte le facce sono triangoli acutangoli con lo stesso raggio della circonferenza circoscritta.}
        \item \pol{Suma kątów płaskich przy każdym wierzchołku wynosi $\pi$.}
              \ita{La somma degli angoli piani in ciascun vertice è uguale a $\pi$.}
        \item \pol{Sumy kątów płaskich przy trzech dowolnych wierzchołkach wynosi $\pi$.}
              \ita{Le somme degli angoli piani in tre vertici arbitrari sono uguali a $\pi$.}
        \item \pol{Siatka czworościanu jest trójkątem ostrokątnym podzielonym na cztery przystające trójkąty.}
              \ita{Lo sviluppo piano del tetraedro è un triangolo acutangolo suddiviso in quattro triangoli congruenti.}
        \item \pol{Kąty $\angle BAC$, $\angle ABD$, $\angle ACD$, $\angle BDC$ są równe.}
              \ita{Gli angoli $\angle BAC$, $\angle ABD$, $\angle ACD$, $\angle BDC$ sono uguali.}
        \item \pol{Przeciwległe krawędzie są równe.}
              \ita{Gli spigoli opposti sono uguali.}
        \item \pol{Trzy odcinki łączące środki przeciwległych krawędzi są parami prostopadłe.}
              \ita{I tre segmenti che uniscono i punti medi degli spigoli opposti sono a due a due perpendicolari.}
        \item \pol{Wszystkie ściany mają równe pola.}
              \ita{Tutte le facce hanno la stessa area.}
        \item \pol{Rzut czworościanu $ABCD$ na dowolną płaszczyznę równoległą do dwóch przeciwległych krawędzi jest prostokątem.}
              \ita{La proiezione del tetraedro $ABCD$ su qualunque piano parallelo a due spigoli opposti è un rettangolo.}
        \item \pol{Każdy odcinek łączący środki przeciwległych krawędzi jest prostopadły do tych krawędzi.}
              \ita{Ogni segmento che unisce i punti medi di due spigoli opposti è perpendicolare a tali spigoli.}
    \end{enumerate}
\end{proposition}

\pol{Dowód wszystkich 11 implikacji znajdziemy w artykulu Pompego, $\Delta_{94}^3$.}
\ita{La dimostrazione di tutte e undici le equivalenze si trova nell'articolo di Pompei, $\Delta_{94}^3$.} % https://www.deltami.edu.pl/1994/03/o-czworoscianie-rownosciennym/

\pol{Czworościan foremny ma trzy osie symetii: to proste przechodzące przez środki przeciwległych krawędzi.}
\ita{Il tetraedro regolare possiede tre assi di simmetria: sono le rette passanti per i punti medi di coppie di spigoli opposti.} % https://www.deltami.edu.pl/1993/04/rozne-roznosci/
