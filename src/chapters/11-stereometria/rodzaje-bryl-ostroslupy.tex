\pol{\subsection{Ostrosłupy}}
\ita{\subsection{Piramidi}}
\pol{\index{ostrosłup}}
\ita{\index{piramide}}

\begin{definition}[\pol{ostrosłup}\ita{piramide}]
    \pol{Wielościan, któreego wszystkie wierzchołki poza jednym leżą na jednej płaszczyźnie, nazywamy ostrosłupem.}
    \ita{Un poliedro i cui vertici, tranne uno, giacciono tutti su uno stesso piano si chiama piramide.}
\end{definition}

\pol{W ostrosłupie, który ma co najmniej pięć wierzchołków, istnieje dokładnie jedna para wierzchołka i~ściany, na której leżą wszystkie pozostałe wierzchołki.}
\ita{In una piramide con almeno cinque vertici esiste esattamente una coppia costituita da un vertice e da una faccia sulla quale giacciono tutti gli altri vertici.}
\pol{Nazywamy je wierzchołkiem i podstawą ostrosłupa.}
\ita{Li chiamiamo vertice e base della piramide.}
\pol{(W ostrosłupie o czterech wierzchołkach, każda ściana może być podstawą).}
\ita{(In una piramide con quattro vertici, ogni faccia può fungere da base).}
\pol{Ostrosłupy są szczególnymi przypadkami pryzmatoidów i mają się do stożków tak, jak graniastosłupy do walców.}
\ita{Le piramidi sono casi particolari di prismoidi e stanno ai coni come i prismi stanno ai cilindri.}

\pol{Ściany boczne ostrosłupów są zawsze trójkątami.}
\ita{Le facce laterali delle piramidi sono sempre triangoli.}
\pol{Wyróżniamy kilka szczególnych przypadków.}
\ita{Distinguiamo alcuni casi particolari.}

\begin{definition}[\pol{ostrosłup $n$-kątny}\ita{piramide a base $n$-gonale}]
    \pol{Ostrosłup, którego podstawa jest $n$-kątem, nazywamy ostrosłupem $n$-kątnym.}
    \ita{Una piramide la cui base è un $n$-gono si chiama piramide a base $n$-gonale.}
\end{definition}

\pol{Ostrosłup trójkątny to inaczej czworościan.}
\ita{Una piramide triangolare è semplicemente un tetraedro.}

\begin{definition}[\pol{ostrosłup prosty}\ita{piramide retta}]
\pol{\index{ostrosłup!prosty}}
\ita{\index{piramide!retta}}%
    \pol{Ostrosłup, którego spodek wysokości pokrywa się ze środkiem okręgu opisanego na podstawie, nazywamy ostrosłupem prostym.}
    \ita{Una piramide il cui piede dell'altezza coincide con il centro della circonferenza circoscritta alla base si chiama piramide retta.}
\end{definition}

\pol{Tak jest na przykład wtedy, kiedy wszystkie krawędzie boczne ostrosłupa albo wszystkie ściany boczne ostrosłupa tworzą z podstawą kąty równej miary.}
\ita{Ciò accade, per esempio, quando tutti gli spigoli laterali oppure tutte le facce laterali formano con la base angoli della stessa ampiezza.}
\pol{Ale pojawią się też konkurencyjne definicje, takie jak: ostrosłup jest prosty, kiedy ma krawędzie boczne równej długości (i pochyły w przeciwnym przypadku).}
\ita{Compariranno però anche definizioni concorrenti, secondo cui una piramide è retta quando ha gli spigoli laterali di uguale lunghezza (e obliqua nel caso contrario).}

\begin{definition}[\pol{ostrosłup prawidłowy}\ita{piramide regolare}]
\pol{\index{ostrosłup!prawidłowy}}
\ita{\index{piramide!regolare}}%
    \pol{Ostrosłup prosty, którego podstawa jest wielokątem foremnym, nazywamy ostrosłupem prawidłowym.}
    \ita{Una piramide retta la cui base è un poligono regolare si chiama piramide regolare.}
\end{definition}

\pol{Ostrosłup prawidłowy czworokątny nazywany jest czasami piramidą.}
\ita{La piramide regolare a base quadrata viene talvolta chiamata semplicemente piramide.}
\pol{\index{piramida}}
\ita{\index{piramide}}%

\begin{definition}[\pol{ostrosłup ścięty}\ita{tronco di piramide}]
\pol{\index{ostrosłup!ścięty}}
\ita{\index{piramide!tronco}}%
    \pol{Część ostrosłupa zawartą między podstawą a pewną płaszczyzną, która przecina ostrosłup, nazywamy ostrosłupem ściętym.}
    \ita{La parte di una piramide compresa tra la base e un piano che la interseca si chiama tronco di piramide.}
\end{definition}

\pol{Jeśli płaszczyzna cięcia jest równoległa do podstawy, to bryłę nazywa się frustrum, z łacińskiego \emph{kęs, kawałek}.}
\ita{Se il piano di taglio è parallelo alla base, il solido viene chiamato \emph{frustum}, dal latino \emph{pezzo, frammento}.}
\pol{\index{frustrum}}
\ita{\index{frustum}}%
\pol{Po polsku nie istnieje takie rozróżnienie: wymaga się, by płaszczyzna tnąca była zawsze równoległa do podstawy.}
\ita{In polacco questa distinzione non esiste: si richiede che il piano di taglio sia sempre parallelo alla base.}
\pol{Eves \cite[s.4]{eves1_1972} napisze, że około 1850 lat przed Chrystusem znany będzie wzór na dokładną objętość frustrum o podstawach długości $a$, $b$ i wysokości $h$:}
\ita{Eves \cite[p. 4]{eves1_1972} scriverà che intorno al 1850 a.C. era già nota la formula per il volume esatto di un frustum con basi di lato $a$, $b$ e altezza $h$:}
\begin{equation}
	V = \frac 1 3 h (a^2 + ab + b^2).
\end{equation}
\pol{Wzór pojawi się bez dowodu w papirusie moskiewskim, nazwanym tak, ponieważ jego pierwszym właścicielem spoza Egiptu będzie Władimir Goleniszczew, rosyjski egiptolog i kolekcjoner sztuki.}
\ita{La formula comparirà senza dimostrazione nel Papiro di Mosca, così chiamato perché il suo primo proprietario al di fuori dell'Egitto sarà Vladimir Golenishchev, egittologo e collezionista d'arte russo.}
\pol{\index{papirus moskiewski}}
\ita{\index{Papiro di Mosca}}%
\pol{\index[persons]{Goleniszczew, Władimir}}
\ita{\index[persons]{Golenishchev, Vladimir}}%
