\pol{\section{Objętość. Trzeci problem Hilberta}}

\pol{Objętość to trójwymiarowy odpowiednik pola powierzchni, wprowadzony nie do końca formalnie przez Euklidesa w (XI.28).}
\ita{Il volume è l'analogo tridimensionale dell'area, introdotto in modo non del tutto formale da Euclide in (XI.28).}
\pol{Pierwszych wyników dowodzi tak, jak ich odpowiedników na płaszczyźnie: (XI.28) mówi, że płaszczyzna przechodząca przez dwa przeciwległe krawędzie równoległościanu rozcina go na dwie części o równych objętościach.}
\ita{Le prime proprietà vengono dimostrate nello stesso modo delle loro controparti piane: (XI.28) afferma che un piano passante per due spigoli opposti di un parallelepipedo lo divide in due parti di uguale volume.}

\pol{O objętości (i trzecim problemie Hilberta) pisze Hartshorne \cite[s. 226--239]{hartshorne2000}.}
\ita{Del volume (e del terzo problema di Hilbert) scrive Hartshorne \cite[p. 226--239]{hartshorne2000}.}

\pol{Euklides pokaże, że ostrosłupy o równych podstawach i wysokościach mają równe objętości (XII.5) i że ich objętość będzie trzy razy mniejsza od objętości graniastosłupa o tej samej podstawie i wysokości:}
\ita{Euclide mostrerà che piramidi con basi e altezze uguali hanno lo stesso volume (XII.5) e che il loro volume è tre volte minore di quello di un prisma avente la stessa base e la stessa altezza:}
\begin{equation}
    V = \frac 1 3 A h.
\end{equation}
\pol{Ale każdy dowód tego wzoru wykorzysta elementy rachunku różniczkowo-całkowego takie jak zasada Cavalieriego albo przejście graniczne (metoda wyczerpywania jak u Euklidesa).}
\ita{Tuttavia ogni dimostrazione di questa formula utilizzerà elementi del calcolo infinitesimale, come il principio di Cavalieri o un procedimento di passaggio al limite (il metodo di esaustione di Euclide).}
\pol{Wzory na pola wielokątów można wyprowadzić elementarnymi metodami, dlatego ludzie będą długo zastanawiać się, co jest nie tak z~trzema wymiarami; patrz \cite[s. 195]{hartshorne2000}.}
\ita{Le formule per le aree dei poligoni possono essere ottenute con metodi elementari; per questo motivo ci si chiederà a lungo che cosa renda così diversi i tre dimensioni; si veda \cite[p. 195]{hartshorne2000}.}

\pol{Christian Gerling, uczeń Gaussa, pokaże mu w liście z 1844 roku, jak rozciąć dowolny czworościan (ostrosłup trójkątny) na 12 części, z których składa się jego lustrzane odbicie.}
\ita{Christian Gerling, allievo di Gauss, gli mostrerà in una lettera del 1844 come suddividere un tetraedro arbitrario in 12 parti che possono essere ricomposte nella sua immagine speculare.}

\pol{Pewne konkretne przykłady czworościanów równoważnych z sześcianem wskaże Micaiah J. M. Hill w~1896 roku:}
\ita{Alcuni esempi concreti di tetraedri equiscomponibili con un cubo saranno forniti da Micaiah J. M. Hill nel 1896:}

\begin{example}
%  Hill, M. J. M. (1895). "Determination of the volumes of certain species of tetrahedra without employment of the method of limits". Proceedings of the London Mathematical Society. 27: 39–53.
    \pol{Dla każdego $0 < \alpha < 2\pi / 3$ niech $v_1, v_2, v_3 \in \mathbb R^3$ będą trzema wektorami jednostkowymi takimi, że kąt między każdymi dwoma ma miarę $\alpha$.}
    \ita{Per ogni $0 < \alpha < 2\pi / 3$, siano $v_1, v_2, v_3 \in \mathbb R^3$ tre vettori unitari tali che l'angolo tra ogni coppia di essi sia uguale a $\alpha$.}
    \pol{Wtedy czworościan}
    \ita{Allora il tetraedro}
    \begin{equation}
        Q(\alpha) = \{c_1 v_1 + c_2 v_2 + c_3 v_3 : 0 \le c_1 \le c_2 \le c_3 \le 1\}
    \end{equation}
    \pol{nazywamy czworościanem Hilla.}
    \ita{si chiama tetraedro di Hill.}
    \pol{Każdy czworościan $Q(\alpha)$ jest równoważny przez rozcinanie z sześcianem.}
    \ita{Ogni tetraedro $Q(\alpha)$ è equiscomponibile con un cubo.}
\end{example}

\pol{Władysław Kretkowski \cite{kretkowski_1880} przygotuje na konkurs matematyczny w 1882 roku zadanie, rozwiązanie którego wyceni na 500 złotych franków.}

\begin{problem}
    \pol{Mając dane dwa czworościeny równej objętości, zresztą najogólniejsze, pociąć, jeżeli się da to wykonać, płaszczyznami jeden z nich na najmniejszą możliwą liczbę kawałków takich, aby przez stosowne tych kawałków zestawienie, można było zbudować czworościen drugi.}
    \ita{Dati due tetraedri qualsiasi di uguale volume, suddividere, se possibile, uno di essi mediante piani nel minor numero possibile di pezzi, in modo che tali pezzi possano essere ricomposti a formare il secondo tetraedro.}

\begin{problem}[\pol{trzeci problem Hilberta}\ita{terzo problema di Hilbert}]
    \label{trzeci_problem_hilberta}
    \pol{Czy mając dane dwa wielościany o równej objętości, można zawsze rozłożyć jeden z nich na skończoną liczbę wielościennych części, a następnie złożyć je w drugi?}
    \ita{Dati due poliedri di uguale volume, è sempre possibile scomporre uno di essi in un numero finito di pezzi poliedrici e ricomporli nel secondo?}
\end{problem}

    \begin{itemize}

    \item \pol{dwa wielościany mają równe objętości oraz równe niezmienniki Dehna.}

\end{proposition}

\begin{theorem}[\pol{Hadwigera}\ita{di Hadwiger}] % Delta 1984 11
    \pol{Niech $\alpha_1$, $\ldots$, $\alpha_p$ będą kątami dwuściennymi wielościanu $P$ leżącymi wzdłuż krawędzi długości $k_1$, $\ldots$, $k_p$, a $\beta_1$, $\ldots$, $\beta_q$ kątami wielościanu $Q$ (wzdłuż krawędzi długości $l_1$, $\ldots$, $l_q$).}
    \ita{Siano $\alpha_1$, $\ldots$, $\alpha_p$ gli angoli diedri del poliedro $P$ lungo spigoli di lunghezza $k_1$, $\ldots$, $k_p$, e $\beta_1$, $\ldots$, $\beta_q$ gli angoli del poliedro $Q$ (lungo spigoli di lunghezza $l_1$, $\ldots$, $l_q$).}
    \pol{Jeśli istnieje taka funkcja $\mathbb Z$-addytywna na zbiorze $M \subseteq \mathbb R$ zawierającym $\pi$, $\alpha_1$, $\ldots$, $\alpha_p$, $\beta_1$, $\ldots$, $\beta_q$, że}
    \ita{Se esiste una funzione $\mathbb Z$-additiva sull'insieme $M \subseteq \mathbb R$ contenente $\pi$, $\alpha_1$, $\ldots$, $\alpha_p$, $\beta_1$, $\ldots$, $\beta_q$, tale che}
    \begin{equation}
        f(P) := f\left(\sum_{i = 1}^p k_i \alpha_i\right) \neq f\left(\sum_{i = 1}^q l_i \beta_i\right) =: f(Q),
    \end{equation}
    \pol{to $P$ i $Q$ nie są równoważne przez rozcinanie.}
    \ita{allora $P$ e $Q$ non sono equiscomponibili.}
\end{theorem}

\begin{example}
    \pol{Sześcian $P_6$ nie jest równoważny przez rozcinanie z czworościanem $P_4$ o tej samej objętości i~kątami dwuściennymi}
    \ita{Il cubo $P_6$ non è equiscomponibile con il tetraedro $P_4$ avente lo stesso volume e angoli diedri}
    \begin{equation}
        \frac \pi 2, \frac \pi 2, \frac \pi 2,
        \arccos \frac{1}{\sqrt 3}, \arccos \frac{1}{\sqrt 3}, \arccos \frac{1}{\sqrt 3}
    \end{equation}

    \pol{Świadkiem tego jest zbiór $M = \{\pi/2, \pi, \alpha\}$ i~funkcja $f$ określona przez $f(\pi/2) = f(\pi) = 0$, $f(\alpha) = 1$ gdzie $\cos \alpha = 1/\sqrt{3}$; zachodzi bowiem $f(P_4) = 3 \sqrt 2 \neq 0 = f(P_6)$.}
    \ita{Una dimostrazione è fornita dall'insieme $M = \{\pi/2, \pi, \alpha\}$ e dalla funzione $f$ definita da $f(\pi/2) = f(\pi) = 0$, $f(\alpha) = 1$, dove $\cos \alpha = 1/\sqrt{3}$; infatti si ha $f(P_4) = 3 \sqrt 2 \neq 0 = f(P_6)$.}
\end{example}

\begin{proposition}[\pol{twierdzenie Cauchy'ego}\ita{teorema di Cauchy}]
    \pol{Każdy wielościan wypukły można rozciąć na czworościany tak, by ich wierzchołkami były jedynie istniejące wierzchołki wielościanu.}
    \ita{Ogni poliedro convesso può essere suddiviso in tetraedri i cui vertici siano soltanto vertici già presenti nel poliedro.}
    \pol{Od rozcięcia można dodatkowo żądać, by pewien wierzchołek należał do wszystkich czworościanów.}
    \ita{Si può inoltre richiedere che un determinato vertice appartenga a tutti i tetraedri della decomposizione.}
\end{proposition}
