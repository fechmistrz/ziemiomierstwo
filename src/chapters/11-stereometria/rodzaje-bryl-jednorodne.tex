\pol{\subsection{Wielościany jednorodne}}
\ita{\subsection{Poliedri uniformi}}

\begin{definition}
    \pol{Wierzchołkowo przechodni wielościan, którego ściany są foremne, nazywamy jednorodnym.}
    \ita{Un poliedro transitivo sui vertici le cui facce sono regolari si chiama uniforme.}
\end{definition}

\pol{Podobnież dzieli się je na foremne (jeśli są ściennie i krawędziowo przechodnie), kwazi-foremne (tylko krawędziowo przechodnie) i semiforemne (jeśli nie są ani takie, ani takie).}
\ita{Essi si dividono inoltre in regolari (se sono transitivi su facce e spigoli), quasiregolari (transitivi solo sugli spigoli) e semiregolari (se non appartengono a nessuna delle due categorie precedenti).}
\pol{Znamy wszystkie wielościany jednorodne:}
\ita{Conosciamo tutti i poliedri uniformi:}
\begin{itemize}
    \item \pol{5 brył platońskich,}\ita{5 solidi platonici,}
    \item \pol{13 wielościanów archimedesowych (2 kwazi-regularne, 11 semi-regularnych),}\ita{13 poliedri archimedei (2 quasiregolari, 11 semiregolari),}
    \item \pol{4 wielościany Keplera-Poinsota,}\ita{4 poliedri di Keplero-Poinsot,}
    \item \pol{53 jednorodne wielościany gwiaździste (14 kwazi, 39 semi).}\ita{53 poliedri stellati uniformi (14 quasiregolari, 39 semiregolari).}
\end{itemize}

\pol{Oprócz tego istnieją dwie nieskończone serie wielościanów jednorodnych: graniastosłupy oraz antygraniastosłupy, tak jak na stronie \pageref{dwie_serie}.}
\ita{Esistono inoltre due serie infinite di poliedri uniformi: i prismi e gli antiprismi, come a pagina \pageref{dwie_serie}.}
