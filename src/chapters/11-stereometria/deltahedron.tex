\pol{Martyn Cundy zaproponuje przez podobieństwo greckiej litery $\Delta$ do trójkąta równobocznego, by rozważać pojęcie takie jak \emph{deltahedron} (nie mylić z \emph{deltohedron}, którego ściany są latawcami):}
\ita{Martyn Cundy proporrà, per la somiglianza della lettera greca $\Delta$ con un triangolo equilatero, di considerare il concetto di \emph{deltahedron} (da non confondere con \emph{deltohedron}, le cui facce sono aquiloni):}
\pol{\index{Cundy, Henry Martyn}}
\ita{\index{Cundy, Henry Martyn}}%

\begin{definition}[\pol{deltościan}\ita{deltaedro}]
    \pol{Wielościan, którego wszystkie ściany są trójkątami równobocznymi, nazywamy deltościanem.}
    \ita{Un poliedro le cui facce sono tutte triangoli equilateri si chiama deltaedro.}
\end{definition}

\begin{proposition}
    \pol{Jest osiem deltościanów wypukłych: trzy bryły platońskie (o czterech, ośmiu albo dwudziestu ścianach) oraz pięć wielościanów Johnsona ($J_{12}$, $J_{13}$, $J_{17}$, $J_{51}$, $J_{84}$).}
    \ita{Esistono otto deltaedri convessi: tre solidi platonici (con quattro, otto o venti facce) e cinque poliedri di Johnson ($J_{12}$, $J_{13}$, $J_{17}$, $J_{51}$, $J_{84}$).}
\end{proposition}
