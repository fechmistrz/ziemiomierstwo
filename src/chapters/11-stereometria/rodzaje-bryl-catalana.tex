\pol{\subsection{Wielościany Catalana}}
\ita{\subsection{Poliedri catalani}}

\begin{definition}[\pol{wielościan dualny}\ita{poliedro duale}]
    \pol{W ustalonym wielościanie usuńmy wierzchołki i krawędzie, zastąpmy każdą ścianę jej środkiem, po czym połączmy te punkty, które należały do sąsiednich ścian.}
    \ita{Dato un poliedro, eliminiamo i vertici e gli spigoli, sostituiamo ogni faccia con il suo centro e poi colleghiamo i punti corrispondenti a facce adiacenti.}
    \pol{Jeśli otrzymany obiekt jest wielościanem, to nazywamy go wielościanem dualnym.}
    \ita{Se l'oggetto ottenuto è un poliedro, lo chiamiamo poliedro duale.}
\end{definition}

\pol{Bryły dualne do wielościanów platońskich są platońskie, a dualne do Keplera-Poinsota są bryłami Keplera-Poinsota, więc nie dają żadnych nowych obiektów.}
\ita{I duali dei poliedri platonici sono ancora poliedri platonici, mentre i duali dei poliedri di Keplero-Poinsot sono ancora poliedri di Keplero-Poinsot, perciò non producono nuovi oggetti.}
\pol{Bryła dualna do bryły dualnej jest tego samego kształtu, co wyjściowa.}
\ita{Il duale del duale di un poliedro ha la stessa forma del poliedro iniziale.}

\begin{definition}[\pol{wielościan Catalana}\ita{poliedro catalano}]
    \pol{Wielościan dualny do wielościanu archimedesowego nazywamy wielościanem Catalana.}
    \ita{Un poliedro duale di un poliedro archimedeo si chiama poliedro catalano.}
\end{definition}

\pol{Pierwsze wielościany Catalana znajdzie Kepler, ale dopiero Eugene Catalan skompletuje je w 1865 roku.}
\ita{I primi poliedri catalani saranno scoperti da Keplero, ma sarà Eugène Catalan a completarne la classificazione nel 1865.} % Some Catalan solids were discovered by Johannes Kepler during his study of zonohedra
\pol{Nazwę sześćdziesciościan deltoidowy zaproponuje profesor Roman Duda podczas tłumaczenia książki ,,Modele matematyczne'' Cundy'ego i Rolleta; brzmi to trochę dziwnie, ale przyjmie się.}
\ita{Il nome \emph{esaecontaedro deltoidale} sarà proposto dal professor Roman Duda durante la traduzione del libro \emph{Modelli matematici} di Cundy e Rollett; suona un po' strano, ma finirà per affermarsi.}

\begin{proposition}
    \pol{Wielościany Catalana są ściennie przechodnie, ale nie wierzchołkowo przechodnie.}
    \ita{I poliedri catalani sono transitivi sulle facce, ma non sui vertici.}
\end{proposition}

\begin{proposition}
    \pol{W wielościanach Catalana wszystkie kąty dwuścienne są równe.}
    \ita{Nei poliedri catalani tutti gli angoli diedri sono uguali.} % Each Catalan solid has constant dihedral angles, meaning the angle between any two adjacent faces is the same.[1]
\end{proposition}
