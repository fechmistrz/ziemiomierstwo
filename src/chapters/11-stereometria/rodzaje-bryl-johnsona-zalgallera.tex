\pol{\subsection{Wielościany Johnsona-Zalgallera}}
\ita{\subsection{Poliedri di Johnson-Zalgaller}}

\begin{definition}
    \pol{Wielościan wypukły, który nie jest jednorodny, ale wszystkie jego ściany są foremne, nazywamy wielościanem Johnsona albo Johnsona-Zalgallera.}
    \ita{Un poliedro convesso che non è uniforme, ma le cui facce sono tutte regolari, si chiama poliedro di Johnson o di Johnson-Zalgaller.}
\end{definition}

\pol{Norman Woodason Johnson \cite{johnson_1966} opublikuje w pracy doktorskiej z 1966 roku, a spisanej pod opieką samego Coxetera, listę 92 brył, które spełniają ten warunek, zaś Wiktor Abramowicz Zalgaller \cite{zalgaller_1969} udowodni po trzech latach (1969), że lista ta jest kompletna.}
\ita{Norman Woodason Johnson \cite{johnson_1966} pubblicherà nella sua tesi di dottorato del 1966, scritta sotto la supervisione dello stesso Coxeter, un elenco di 92 solidi che soddisfano questa condizione; tre anni dopo (1969) Viktor Abramovič Zalgaller \cite{zalgaller_1969} dimostrerà che l'elenco è completo.}

\begin{proposition}
    \pol{Jeśli $n \neq 72, 73, 74, 75, 77$, to wielościan Johnsona-Zalgallera $J_{n}$ ma własność Ruperta: można wyciąć w nim otwór, przez który przechodzi powiększona kopia tego wielościanu.}
    \ita{Se $n \neq 72, 73, 74, 75, 77$, allora il poliedro di Johnson-Zalgaller $J_{n}$ possiede la proprietà di Rupert: è possibile praticarvi un foro attraverso il quale passa una copia ingrandita del poliedro stesso.}
\end{proposition}
