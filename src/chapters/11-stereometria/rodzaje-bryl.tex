\pol{\section{Rodzaje brył}}
\ita{\section{Tipi di solidi}}

\pol{Panuje ogromny bałagan, jeśli chodzi o nazewnictwo wielościanów.}
\ita{Regna una grande confusione nella nomenclatura dei poliedri.}
\pol{Spróbujemy więc zaprowadzić porządek.}
\ita{Cercheremo quindi di fare un po' d'ordine.}

\begin{definition}
    \pol{Wielościan nazywamy przechodnim wierzchołkowo (odpowiednio: krawędziowo, ściennie, flagowo), jeśli działanie grupy symetrii tego wielościanu na zbiorze jego wierzchołków (odpowiednio: krawędzi, ścian, flag) jest przechodnie.}
    \ita{Un poliedro si dice transitivo sui vertici (rispettivamente sugli spigoli, sulle facce, sulle bandiere) se l'azione del suo gruppo di simmetria sull'insieme dei vertici (rispettivamente degli spigoli, delle facce, delle bandiere) è transitiva.}
\end{definition}

\begin{definition}
    \pol{Wielościany platońskie to te, które są flagowo (lub równoważnie, wierzchołkowo, krawędziowo i ściennie) przechodnie, zaś ich ściany są wielokątami foremnymi.}
    \ita{I poliedri platonici sono quelli che sono transitivi sulle bandiere (o equivalentemente sui vertici, sugli spigoli e sulle facce) e le cui facce sono poligoni regolari.}
\end{definition}

\begin{definition}
    \pol{Wielościany Keplera-Poinsota (gwiaździste foremne) to te, które są flagowo przechodnie, zaś ich ściany są wielokątami gwiaździstymi foremnymi.}
    \ita{I poliedri di Keplero-Poinsot (stellati regolari) sono quelli che sono transitivi sulle bandiere e le cui facce sono poligoni stellati regolari.}
\end{definition}

\begin{definition}
\label{wieloscian_archimedesowy}%
    \pol{Wielościany archimedesowe to te, których ściany są pewnymi wielokątami foremnymi, zaś w~każdym wierzchołku cykliczne ułożenie ścian jest takie samo.}
    \ita{I poliedri archimedei sono quelli le cui facce sono poligoni regolari e nei quali la disposizione ciclica delle facce attorno a ogni vertice è la stessa.}
\end{definition}

\pol{Inna definicja głosi, że wielościany półforemne to wielościany wierzchołkowo przechodnie, ale nie ściennie przechodnie, których wszystkie ściany są wielokątami foremnymi.}
\ita{Un'altra definizione afferma che i poliedri semiregolari sono poliedri transitivi sui vertici ma non sulle facce, le cui facce sono tutte poligoni regolari.}
\pol{Pseudo sześcio-ośmiościan rombowy mały spełnia definicję \ref{wieloscian_archimedesowy}, bo w każdym wierzchołku spotykają się trzy kwadraty i jeden trójkąt, ale nie jest półforemny.}
\ita{Lo pseudo-rombicubottaedro piccolo soddisfa la definizione \ref{wieloscian_archimedesowy}, poiché in ogni vertice si incontrano tre quadrati e un triangolo, ma non è semiregolare.}

\pol{Stanowczo za dużo osób nie widzi różnicy między ,,archimedesowym'' i ,,półforemnym'', ale my idziemy w ślad za Grünbaumem \cite{grunbaum_2009} i widzimy.}
\ita{Decisamente troppe persone non vedono la differenza tra ``archimedeo'' e ``semiregolare'', ma noi seguiamo Grünbaum \cite{grunbaum_2009} e la vediamo.}
\pol{Grünbaum pokusi się o prześledzenie \emph{,,enduring error''} w niedawno opublikowanych pracach.}
\ita{Grünbaum si prenderà la briga di seguire la traccia di un \emph{``enduring error''} nei lavori pubblicati di recente.}

\begin{definition}
    \pol{Wielościany Catalana to te, które są dualne do wielościanów archimedesowych.}
    \ita{I poliedri catalani sono quelli duali dei poliedri archimedei.}
\end{definition}

\begin{definition}
    \pol{Wielościany jednorodne to te, które są wierzchołkowo przechodnie i mają foremne ściany. Dzielą się na foremne (ściennie i krawędziowo przechodnie), kwasiforemne (krawędziowo, ale nie ściennie przechodnie) i semiforemne (ani krawędziowo, ani ściennie przechodnie).}
    \ita{I poliedri uniformi sono quelli transitivi sui vertici e con facce regolari. Si dividono in regolari (transitivi su facce e spigoli), quasiregolari (transitivi sugli spigoli ma non sulle facce) e semiregolari (né transitivi sugli spigoli né sulle facce).}
\end{definition}

\begin{definition}
    \pol{Wielościany Johnsona-Zalgallera to te, które są wypukłe, mają foremne ściany i nie są jednorodne.}
    \ita{I poliedri di Johnson-Zalgaller sono quelli che sono convessi, hanno facce regolari e non sono uniformi.}
\end{definition}
