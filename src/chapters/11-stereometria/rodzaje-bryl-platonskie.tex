\pol{\subsubsection{Wielościany platońskie}}
\ita{\subsubsection{Poliedri platonici}}

\todofoot{Coxeter 166-167.}
\todofoot{OCTAEDRON ELEVATUM}

\pol{Wszystkie ściany wielościanów platońskich są przystającymi wielokątami foremnymi, w każdym wierzchołku spotyka się taka sama liczba ścian.}
\ita{Tutte le facce dei poliedri platonici sono poligoni regolari congruenti e in ogni vertice si incontra lo stesso numero di facce.}
\pol{Pozostaje zagadką, kto pierwszy odkryje wszystkie foremne wielościany wypukłe, ale dwunastościan zostanie odkryty jako ostatni.}
\ita{Rimane un mistero chi abbia scoperto per primo tutti i poliedri regolari convessi, ma il dodecaedro sarà l'ultimo a essere scoperto.}
\pol{Teajtet, uczeń Platona udowodni, że jest ich dokładnie pięć:}
\ita{Teeteto, allievo di Platone, dimostrerà che ve ne sono esattamente cinque:}
\pol{\index[persons]{Teajtet, uczeń Platona}}
\ita{\index[persons]{Teeteto}}%
\begin{itemize}
\item \pol{czworościan foremny,}\ita{tetraedro regolare,}
\item \pol{sześciościan foremny (znany lepiej jako sześcian),}\ita{esaedro regolare (più noto come cubo),}
\item \pol{ośmiościan foremny,}\ita{ottaedro regolare,}
\item \pol{dwunastościan foremny,}\ita{dodecaedro regolare,}
\item \pol{dwudziestościan foremny.}\ita{icosaedro regolare.}
\end{itemize}

\pol{Niektórzy będą mylić wielościany foremne z platońskimi (!).}
\ita{Alcuni confonderanno i poliedri regolari con quelli platonici (!).}
\pol{Więcej informacji u Hartshorne'a \cite[rozdział 8, 45]{hartshorne2000}.}
\ita{Per maggiori informazioni si veda Hartshorne \cite[capitolo 8, 45]{hartshorne2000}.}

\begin{proposition}
    \pol{Największy sześcian, jaki można zmieścić w dwudziestościanie foremnym, wypełnia}
    \ita{Il più grande cubo che può essere inscritto in un icosaedro regolare occupa}
    \begin{equation}
        \frac{219 \sqrt 5 + 15}{1331} \approx 38\,\%
    \end{equation}
    \pol{objętości.}
    \ita{del volume.}
\end{proposition}
