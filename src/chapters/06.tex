\pol{Trygonometria (od greckiego τρίγωνον (trójkąt) oraz μέτρον (miara)) to dział geometrii, który bada relacje między długościami boków trójkąta, a miarami jego kątów.}
\ita{La trigonometria (dal greco τρίγωνον, triangolo, e μέτρον, misura) è il ramo della geometria che studia le relazioni tra le lunghezze dei lati di un triangolo e le ampiezze dei suoi angoli.}

\pol{O funkcjach trygonometrycznych kąta ostrego pisze Guzicki \cite[s. 240-254]{guzicki_2021}, Bogdańska, Neugebauer \cite[s. 88]{neugebauer_2018}.}
\ita{Delle funzioni trigonometriche di un angolo acuto trattano Guzicki \cite[pp.~240--254]{guzicki_2021} e Bogdańska, Neugebauer \cite[p.~88]{neugebauer_2018}.}

\begin{theorem}[\pol{nierówność Arystarcha}\ita{disuguaglianza di Aristarco}]
\index{nierówność!Arystarcha}
\ita{\index{disuguaglianza!di Aristarco}}
    \pol{Niech $\alpha, \beta$ będą kątami ostrymi takimi, że $\beta < \alpha$.}
    \ita{Siano $\alpha$ e $\beta$ angoli acuti tali che $\beta < \alpha$.}
    \pol{Wtedy}
    \ita{Allora}
    \begin{equation}
        \frac{\sin \alpha}{\sin \beta} < \frac \alpha \beta < \frac{\tan \alpha}{\tan \beta}.
    \end{equation}
	% https://en.wikipedia.org/wiki/Aristarchus's_inequality
\end{theorem}
