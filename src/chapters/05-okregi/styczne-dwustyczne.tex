\pol{\section{Styczne i dwustyczne}}
\ita{\section{Tangenti e bitangenti}}

\begin{proposition}
    \pol{Niech $\Gamma$ będzie okręgiem o środku $O$ oraz promieniu $OA$.}
    \ita{Sia $\Gamma$ una circonferenza di centro $O$ e raggio $OA$.}
    \pol{Wtedy prosta prostopadła do $OA$, która przechodzi przez $A$, jest styczną do okręgu, leżącą (poza punktem $A$) na zewnątrz okręgu $\Gamma$.}
    \ita{Allora la retta perpendicolare a $OA$ passante per $A$ è tangente alla circonferenza e, a eccezione del punto $A$, giace all'esterno di $\Gamma$.}
    \pol{Odwrotnie, każda prosta, która jest styczna w punkcie $A$ do okręgu $\Gamma$, musi być prostopadła do prostej $OA$.}
    \ita{Viceversa, ogni retta tangente alla circonferenza $\Gamma$ nel punto $A$ deve essere perpendicolare alla retta $OA$.}
    \pol{\index{styczność}}
    \ita{\index{tangenza}}
\end{proposition} % Hartshorne 105

\begin{corollary}
    \pol{Przez każdy punkt okręgu przechodzi dokładnie jedna styczna do tego okręgu.}
    \ita{Per ogni punto della circonferenza passa una e una sola tangente alla circonferenza stessa.}
\end{corollary} % Hartshorne 105

\pol{Przez każdy punkt leżący na zewnątrz okręgu przechodzą dwie proste -- Neugebauer \cite[s. 34]{neugebauer_2018} z rozmachem nazywa to \emph{,,zasadniczym twierdzeniem planimetrii''}\footnote{Ciąg dalszy następuje w zbiorze zadań Kiezy \cite[s. 28]{kieza_2017}: załóżmy, że pewne dwie płaszczyzny, przecinające się wzdłuż prostej $AB$, są styczne do sfery w punktach $P$ i $Q$. Wtedy $AP = AQ$, $BP = BQ$, zaś trójkąty $APB$ i $AQB$ są przystające. Kieza nazywa to najmocniejszym twierdzeniem stereometrii.}.}
\ita{Per ogni punto esterno a una circonferenza passano due tangenti; Neugebauer \cite[p. 34]{neugebauer_2018} chiama con enfasi questo fatto il \emph{``teorema fondamentale della planimetria''}\footnote{Il seguito compare nella raccolta di problemi di Kieza \cite[p. 28]{kieza_2017}: supponiamo che due piani, intersecantisi lungo la retta $AB$, siano tangenti a una sfera nei punti $P$ e $Q$. Allora $AP = AQ$, $BP = BQ$ e i triangoli $APB$ e $AQB$ sono congruenti. Kieza lo chiama il più potente teorema della stereometria.}.}

\begin{corollary}
    \pol{Prosta, która nie jest styczna do okręgu i nie jest z nim rozłączna, musi przecinać go w dokładnie dwóch punktach.}
    \ita{Una retta che non sia tangente a una circonferenza e non sia disgiunta da essa la interseca in esattamente due punti.}
\end{corollary} % Hartshorne 106

\begin{proposition}
    \pol{Niech $O_1, O_2, A$ będą trzema punktami.}
    \ita{Siano $O_1$, $O_2$ e $A$ tre punti.}
    \pol{Następujące warunki są równoważne: punkty $A, O_1, O_2$ są współliniowe; okręgi o promieniach $O_1A$, $O_2A$ są styczne.}
    \ita{Le seguenti condizioni sono equivalenti: i punti $A$, $O_1$, $O_2$ sono allineati; le circonferenze di raggi $O_1A$ e $O_2A$ sono tangenti.}
\end{proposition} % Hartshorne 105

\begin{corollary}
    \pol{Dwa okręgi, które nie są rozłączne i nie są styczne, mają dokładnie dwa punkty wspólne.}
    \ita{Due circonferenze che non siano disgiunte e non siano tangenti hanno esattamente due punti in comune.}
\end{corollary} % Hartshorne 106

\ita{Esiste inoltre il teorema delle secanti e delle tangenti, che conduce ancora una volta al concetto di potenza di un punto rispetto a una circonferenza...}

\begin{proposition}[\pol{twierdzenie o siecznych i stycznych}\ita{teorema delle secanti e delle tangenti}]
	\pol{Jeżeli przechodząca przez dany punkt $S$ sieczna $k$ przecina okrąg $\Omega$ o środku $O$ i promieniu $r$ w punktach $A$ i $B$, to iloczyn $|SA| \cdot |SB|$ nie zależy od wyboru prostej $k$; ma on wartość $|s^2 - r^2|$, gdzie $s = |SO|$.}
	\ita{Se una secante $k$ passante per un punto assegnato $S$ interseca la circonferenza $\Omega$, di centro $O$ e raggio $r$, nei punti $A$ e $B$, allora il prodotto $|SA| \cdot |SB|$ non dipende dalla scelta della retta $k$; esso vale $|s^2-r^2|$, dove $s = |SO|$.}
	\pol{\index{twierdzenie!o siecznych i stycznych}}
	\ita{\index{teorema!delle secanti e delle tangenti}}
\end{proposition}
