\pol{\subsection{Okręgi dopisane}}
\ita{\subsection{Circonferenze exinscritte}}
\pol{\index{okrąg dopisany}}%
\ita{\index{circonferenza exinscritta}}%

\pol{Oprócz okręgu wpisanego w trójkąt są jeszcze trzy inne okręgi styczne do wszystkich trzech prostych, na których leżą boki trójkąta.}
\ita{Oltre alla circonferenza inscritta in un triangolo esistono altre tre circonferenze tangenti a tutte e tre le rette contenenti i lati del triangolo.}
\pol{Nazywamy je okręgami dopisanymi (\emph{excircle} albo \emph{escribed circle} po angielsku), pisze o nich Audin \cite[s. 98]{audin_2003}.}
\ita{Sono dette circonferenze exinscritte (\emph{excircle} oppure \emph{escribed circle} in inglese); Audin ne parla a \cite[p. 98]{audin_2003}.}

\pol{Środek okręgu wpisanego znajduje się na przecięciu dwusiecznych wewnętrznych kątów trójkąta.}
\ita{Il centro della circonferenza inscritta è il punto d'incontro delle bisettrici interne del triangolo.}
\pol{Analogicznie środki okręgów dopisanych znajdują się na przecięciu dwusiecznej kąta wewnętrznego przy ustalonym wierzchołku i dwusiecznych kątów zewnętrznych przy pozostałych wierzchołkach.}
\ita{Analogamente, i centri delle circonferenze exinscritte si trovano all'intersezione della bisettrice dell'angolo interno in un vertice fissato e delle bisettrici degli angoli esterni negli altri due vertici.}

\pol{Ich promienie można łatwo wyznaczyć przez porównanie pól trójkątów, co prowadzi do wzoru:}
\ita{I loro raggi si determinano facilmente confrontando aree di triangoli, ottenendo la formula}
\begin{equation}
	r_a = \frac{S_{ABC}}{p - a},
\end{equation}
\pol{oraz dwóch kolejnych dla $r_b$ i $r_c$, jak pokazuje Zetel \cite[s. 18]{zetel_2020}.}
\ita{e le analoghe formule per $r_b$ e $r_c$, come mostra Zetel \cite[p. 18]{zetel_2020}.}
\pol{Tutaj $p$ oznacza tak jak zawsze połowę obwodu.}
\ita{Qui $p$ indica, come di consueto, il semiperimetro.}
\pol{Czynnik $p - a$ sugeruje związek ze wzorem Herona i tak istotnie jest, mamy}
\ita{Il fattore $p-a$ suggerisce un legame con la formula di Erone, e infatti}
\begin{equation}
	S_{ABC}^2 = r \cdot r_a \cdot r_b \cdot r_c.
\end{equation}

\pol{Zależności między czterema promieniami jest bez liku, mamy też:}
\ita{Le relazioni tra questi quattro raggi sono numerosissime; ad esempio}
\begin{equation}
	\frac{1}{r} = \frac {1} {r_a} + \frac {1} {r_b} + \frac {1} {r_c}.
\end{equation}
