\pol{\section{Potęga punktu względem okręgu}}
\ita{\section{Potenza di un punto rispetto a una circonferenza}}

\pol{Zdefiniujemy potęgę punktu względem okręgu, a potem potęgową oś i środek.}
\ita{Definiremo la potenza di un punto rispetto a una circonferenza e successivamente l'asse radicale e il centro radicale.}
\todofoot{\pol{Inwersja: Audin \cite[s. 84]{audin_2003}.}\ita{Inversione: Audin \cite[p. 84]{audin_2003}.}}

\todofoot{Coxeter s. 85, coaxial circles}

\begin{definition}[\pol{potęga punktu względem okręgu}\ita{potenza di un punto rispetto a una circonferenza}]
	\label{def_power_point}
	\pol{Dane są okrąg $\Gamma$ o środku $O$ i promieniu $r$ oraz dowolny punkt $A$.}
	\ita{Siano dati una circonferenza $\Gamma$ di centro $O$ e raggio $r$, nonché un punto arbitrario $A$.}
	\pol{Liczbę rzeczywistą}
	\ita{Il numero reale}
	\begin{equation}
		\Pi(A) := |OA|^2 - r^2
	\end{equation}
	\pol{nazywamy potęgą punktu $A$ względem okręgu $\Gamma$.}
	\ita{si chiama potenza del punto $A$ rispetto alla circonferenza $\Gamma$.}
\end{definition}

\begin{proposition}[\pol{oś potęgowa istnieje}\ita{esistenza dell'asse radicale}]
\label{guzicki_6_11}%
    \pol{Dane są dwa niewspółśrodkowe okręgi $\Gamma_1(O_1, r_1)$ oraz $\Gamma_2(O_2, r_2)$.}
    \ita{Siano date due circonferenze non concentriche $\Gamma_1(O_1, r_1)$ e $\Gamma_2(O_2, r_2)$.}
    \pol{Wtedy miejscem geometrycznym punktów $P$ mających równe potęgi względem obu okręgów:}
    \ita{Allora il luogo geometrico dei punti $P$ aventi la stessa potenza rispetto a entrambe le circonferenze:}
	\begin{equation}
		\{S : \Pi(S, \Gamma_1) = \Pi(S, \Gamma_2)\}
	\end{equation}
	\pol{jest prosta prostopadła do prostej przechodzącej przez środki obu okręgów.}
	\ita{è una retta perpendicolare alla retta passante per i centri delle due circonferenze.}
	\pol{Nazywamy ją \emph{osią potęgową} okręgów $\Gamma_1$, $\Gamma_2$.}
	\ita{La chiamiamo \emph{asse radicale} delle circonferenze $\Gamma_1$, $\Gamma_2$.}
	\pol{\index{oś potęgowa}}
	\ita{\index{asse radicale}}%
	\pol{\index{potęgowa!oś|see{oś potęgowa}}}
	\ita{\index{radicale!asse|see{asse radicale}}}
\end{proposition}

\pol{O osi potęgowej piszą Guzicki \cite[s. 173, 174]{guzicki_2021}, Bogdańska, Neugebauer \cite[s. 69]{neugebauer_2018}, Audin \cite[s. 89]{audin_2003}, Eves \cite[s. 92]{eves1_1972}; patrz też \cite[s. 11]{komisarczyk_2024}.}
\ita{Dell'asse radicale scrivono Guzicki \cite[p. 173, 174]{guzicki_2021}, Bogdańska e Neugebauer \cite[p. 69]{neugebauer_2018}, Audin \cite[p. 89]{audin_2003}, Eves \cite[p. 92]{eves1_1972}; si veda anche \cite[p. 11]{komisarczyk_2024}.}
\pol{Oś potęgowa okręgów stycznych to prosta styczna do obydwu okręgów; oś potęgowa okręgów, które się przecinają w dwóch punktach, to prosta przez punkty przecięcia (to ćwiczenie u Hartshorne'a \cite[s. 182]{hartshorne2000}).}
\ita{L'asse radicale di due circonferenze tangenti è la loro tangente comune; l'asse radicale di due circonferenze che si intersecano in due punti è la retta passante per tali punti di intersezione (si tratta di un esercizio in Hartshorne \cite[p. 182]{hartshorne2000}).}

\begin{proposition}[\pol{środek potęgowy}\ita{centro radicale}]
	\pol{Dane są trzy parami niewspółśrodkowe okręgi $\Gamma_1, \Gamma_2, \Gamma_3$.}
	\ita{Siano date tre circonferenze $\Gamma_1, \Gamma_2, \Gamma_3$ a due a due non concentriche.}
	\pol{Wtedy albo środki tych okręgów są współliniowe (i trzy osie potęgowe każdej pary są równoległe), albo wszystkie trzy osie przecinają się w jednym punkcie: \emph{środku potęgowym} okręgów.}
	\ita{Allora o i centri delle tre circonferenze sono allineati (e i tre assi radicali delle coppie di circonferenze sono paralleli), oppure tutti e tre gli assi si incontrano in un unico punto, detto \emph{centro radicale} delle circonferenze.}
	\pol{\index{środek potęgowy}}
	\ita{\index{centro radicale}}%
	\pol{\index{potęgowy!środek|see{środek potęgowy}}}
	\ita{\index{radicale!centro|see{centro radicale}}}
\end{proposition}

\ita{Anche i termini ``asse radicale'' e ``centro radicale'' si devono al già citato Gaultier.}

\ita{Non è chiaro perché alcuni chiamino questo risultato teorema di Monge (come Bogdańska e Neugebauer \cite[p. 70]{neugebauer_2018}) oppure soluzione del problema di Monge (come Dörrie \cite[p. 151]{dorrie_1965}).}

\begin{theorem}[\pol{Auberta}\ita{di Aubert}]
	\pol{Dane są cztery proste w położeniu ogólnym (żadne trzy nie przechodzą przez jeden punkt, żadne dwie nie są równoległe).}
	\ita{Siano date quattro rette in posizione generale (nessuna terna passa per uno stesso punto e nessuna coppia è parallela).}
	\pol{Wówczas ortocentra czterech trójkątów wyznaczonych przez trójki tych prostych leża na jednej prostej, zwanej prostą Auberta.}
	\ita{Allora gli ortocentri dei quattro triangoli determinati dalle terne di tali rette giacciono su una stessa retta, detta retta di Aubert.}
	\pol{\index{twierdzenie!Auberta}}
	\ita{\index{teorema!di Aubert}}
\end{theorem}

\pol{\subsection{Pęki okręgów}}
\ita{\subsection{Fasci di circonferenze}}

\pol{Pękiem (z angielskiego \emph{pencil}) będziemy nazywać dowolną rodziną geometrycznych obiektów (przede wszystkim prostych lub okręgów) o pewnej wspólnej cesze, którą można odtworzyć z każdej pary obiektów wspomnianej rodziny.}
\ita{Chiameremo fascio (in inglese \emph{pencil}) una qualunque famiglia di oggetti geometrici (soprattutto rette o circonferenze) caratterizzata da una proprietà comune che può essere ricostruita a partire da una qualsiasi coppia di oggetti della famiglia.}

\begin{example}
	\pol{Do pęku prostych należą wszystkie proste, które przechodzą przez ustalony punkt.}
	\ita{Appartengono a un fascio di rette tutte le rette che passano per un punto fissato.}
\end{example}

\begin{definition}
	\pol{Dowolne dwa niewspółśrodkowe okręgi $K_1, K_2$ wyznaczają jedną maksymalną rodzinę okręgów współosiowych z okręgami $K_1, K_2$, zwaną pękiem okręgów.}
	\ita{Due qualunque circonferenze non concentriche $K_1$, $K_2$ determinano un'unica massima famiglia di circonferenze coassiali con $K_1$ e $K_2$, detta fascio di circonferenze.}
\end{definition}

\pol{O pękach okręgów pisze Neugebauer, Bogdańska \cite[s. 80]{neugebauer_2018}.}
\ita{Dei fasci di circonferenze scrivono Neugebauer e Bogdańska \cite[p. 80]{neugebauer_2018}.}
\pol{Dla ustalonch punktów $P, Q$ wprowadzają tam rodzinę}
\ita{Per punti fissati $P$, $Q$ introducono la famiglia}
\begin{equation}
	\mathcal H(PQ) := \{A(\lambda) : 0 \le \lambda \le + \infty, \lambda \neq 1 \},
\end{equation}
\pol{gdzie $A(\lambda)$ to okrąg Apoloniusza z definicji \ref{definicja_okregu_apoloniusza}.}
\ita{dove $A(\lambda)$ è la circonferenza di Apollonio della definizione \ref{definicja_okregu_apoloniusza}.}
\pol{Okręgi $A(\lambda)$ są parami rozłączne, wszystkie są prostopadłe do wszystkich okręgów przechodzących przez punkty $P$ i $Q$.}
\ita{Le circonferenze $A(\lambda)$ sono a due a due disgiunte e tutte ortogonali a tutte le circonferenze passanti per i punti $P$ e $Q$.}
\pol{Uznają, że $A(\infty)$ to punkt $Q$, $A(0)$ to punkt $P$, zaś $A(1)$ to symetralna odcinka $PQ$.}
\ita{Assumono che $A(\infty)$ sia il punto $Q$, che $A(0)$ sia il punto $P$ e che $A(1)$ sia l'asse del segmento $PQ$.}
\pol{Wtedy przez każdy punkt płaszczyzny przechodzi dokładnie jeden okrąg $A(\lambda)$.}
\ita{Allora per ogni punto del piano passa esattamente una circonferenza $A(\lambda)$.}

\pol{Maksymalną współosiową rodzinę okręgów parami rozłącznych nazywają pękiem hiperbolicznym okręgów, każdy jest postaci $\mathcal H(PQ)$ i punkty $P \neq Q$ są wyznaczone jednoznacznie.}
\ita{Chiamano fascio iperbolico di circonferenze una massima famiglia coassiale di circonferenze a due a due disgiunte; ciascuna è della forma $\mathcal H(PQ)$ e i punti $P \neq Q$ sono determinati univocamente.}

\begin{definition}
	\pol{Punkty $P, Q$ nazywamy punktami Ponceleta, prostą $PQ$ linią środków, zaś symetralną odcinka $PQ$ -- osią pęku hiperbolicznego $\mathcal H(PQ)$.}
	\ita{I punti $P$, $Q$ si chiamano punti di Poncelet, la retta $PQ$ si chiama linea dei centri e l'asse del segmento $PQ$ si chiama asse del fascio iperbolico $\mathcal H(PQ)$.}
\end{definition}

\pol{Nazwisko Ponceleta pojawia się tu nieprzypadkowo, ma związek z (małym) twierdzeniem Ponceleta.}
\ita{Il nome di Poncelet non compare qui per caso: è legato al (piccolo) teorema di Poncelet.}

\pol{(Nie jest trudno się domyślić, że istnieją też pęki paraboliczne i eliptyczne).}
\ita{(Non è difficile immaginare che esistano anche fasci parabolici ed ellittici).}
