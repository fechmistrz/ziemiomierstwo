\begin{proposition}[\pol{o pięciu okręgach}\ita{dei cinque cerchi}]
    \pol{Środki $O_1, O_2, \ldots, O_5$ okręgów $\Gamma_1, \Gamma_2, \ldots, \Gamma_5$ leżą na okręgu $\Gamma$.}
    \ita{I centri $O_1, O_2, \ldots, O_5$ dei cerchi $\Gamma_1, \Gamma_2, \ldots, \Gamma_5$ giacciono sul cerchio $\Gamma$.}
    \pol{Następnie: okręgi $\Gamma_{k}$ i $\Gamma_{k+1}$ dla $k = 1, 2, \ldots, 5$ przecinają się w punktach $A_k$, $B_k$ (przy czym $\Gamma_6$ definiujemy jako $\Gamma_1$).}
    \ita{Inoltre: i cerchi $\Gamma_{k}$ e $\Gamma_{k+1}$ per $k = 1, 2, \ldots, 5$ si intersecano nei punti $A_k$, $B_k$ (dove definiamo $\Gamma_6$ come $\Gamma_1$).}
    \pol{Punkty przecięcia $A_1, A_2, \ldots, A_5$ także leżą na okręgu $\Gamma$.}
    \ita{Anche i punti di intersezione $A_1, A_2, \ldots, A_5$ giacciono sul cerchio $\Gamma$.}
    \pol{Wreszcie: proste $B_kB_{k+1}$ i $B_{k+3}B_{k+4}$ dla przecinają się w punkcie $C_k$}
    \ita{Infine: le rette $B_kB_{k+1}$ e $B_{k+3}B_{k+4}$ si intersecano nel punto $C_k$}

    \pol{Wówczas punkty $C_1, C_2, \ldots, C_5$ leżą na okręgach $\Gamma_1, \Gamma_2, \ldots, \Gamma_5$.}
    \ita{Allora i punti $C_1, C_2, \ldots, C_5$ giacciono sui cerchi $\Gamma_1, \Gamma_2, \ldots, \Gamma_5$.}
\end{proposition}

\begin{proposition}[\pol{o sześciu okręgach}\ita{dei sei cerchi}]
\pol{\index{twierdzenie!o sześciu okręgach}}\ita{\index{teorema!dei sei cerchi}}%
    \pol{Dany jest trójkąt $\triangle ABC$ oraz sześć okręgów $\Gamma_1, \Gamma_2, \ldots, \Gamma_6$ wpisanych kolejno w kąty przy wierzchołkach $A$, $B$, $C$, $A$, $B$, $C$, $A$ tak, że każdy jest styczny do poprzedniego.}
    \ita{Sono dati un triangolo $\triangle ABC$ e sei cerchi $\Gamma_1, \Gamma_2, \ldots, \Gamma_6$ inscritti successivamente negli angoli presso i vertici $A$, $B$, $C$, $A$, $B$, $C$, $A$ in modo che ciascuno sia tangente al precedente.}
    \pol{Wtedy okręgi $\Gamma_6$ i $\Gamma_1$ też są do siebie styczne.}
    \ita{Allora anche i cerchi $\Gamma_6$ e $\Gamma_1$ sono tangenti tra loro.}
\end{proposition}

\pol{Tabacznikow, Iwanow \cite{ivanov_tabachnikov_2016} pokażą, że jeśli osłabimy założenia: okręgi nie muszą zawierać się w~trójkącie i~wystarczy, że będą styczne do prostych zawierających boki trójkąta, to nadal ciąg okręgów jest od pewnego miejsca okresowy z okresem równym sześć, ale osiągnięcie tego stanu może wymagać dowolnie wielu kroków.}
\ita{Tabacznikow e Iwanow \cite{ivanov_tabachnikov_2016} mostreranno che, se indeboliamo le ipotesi: i cerchi non devono essere contenuti nel triangolo ed è sufficiente che siano tangenti alle rette contenenti i lati del triangolo, allora la successione di cerchi è comunque, da un certo punto in poi, periodica con periodo uguale a sei, ma il raggiungimento di questo stato può richiedere un numero arbitrariamente grande di passi.}

\begin{proposition}[\pol{o siedmiu okręgach}\ita{dei sette cerchi}]
\pol{\index{twierdzenie!o siedmiu okręgach}}\ita{\index{teorema!dei sette cerchi}}%
    \pol{Dany jest okrąg $\Gamma$ oraz sześć okręgów stycznych do niego tak, że każdy jest też styczny do swoich dwóch sąsiadów.}
    \ita{Sono dati un cerchio $\Gamma$ e sei cerchi tangenti a esso, in modo che ciascuno sia anche tangente ai suoi due vicini.}
    \pol{Wtedy trzy proste łączące przeciwległe punkty styczności są współpękowe.}
    \ita{Allora le tre rette che congiungono punti di tangenza opposti sono concorrenti.}
\end{proposition}

\begin{proposition}[\pol{o dziewięciu okręgach}\ita{dei nove cerchi}]
\pol{\index{twierdzenie!o dziewięciu okręgach}}\ita{\index{teorema!dei nove cerchi}}%
    \pol{Dane są cztery okręgi $\Gamma_1$, $\Gamma_2$, $\Gamma_3$, $\Gamma_4$ takie, że czwarty jest styczny do drugiego i trzeciego.}
    \ita{Sono dati quattro cerchi $\Gamma_1$, $\Gamma_2$, $\Gamma_3$, $\Gamma_4$ tali che il quarto sia tangente al secondo e al terzo.}
    \pol{Kreślimy ciąg okręgów stycznych do poprzednich:}
    \ita{Tracciamo una successione di cerchi tangenti ai precedenti:}
    \pol{$\Gamma_5$ do $\Gamma_1$, $\Gamma_3$ i $\Gamma_4$,}
    \ita{$\Gamma_5$ a $\Gamma_1$, $\Gamma_3$ e $\Gamma_4$,}
    \pol{$\Gamma_6$ do $\Gamma_1$, $\Gamma_2$ i $\Gamma_5$,}
    \ita{$\Gamma_6$ a $\Gamma_1$, $\Gamma_2$ e $\Gamma_5$,}
    \pol{$\Gamma_7$ do $\Gamma_2$, $\Gamma_3$ i $\Gamma_6$,}
    \ita{$\Gamma_7$ a $\Gamma_2$, $\Gamma_3$ e $\Gamma_6$,}
    \pol{$\Gamma_8$ do $\Gamma_1$, $\Gamma_3$ i $\Gamma_7$,}
    \ita{$\Gamma_8$ a $\Gamma_1$, $\Gamma_3$ e $\Gamma_7$,}
    \pol{$\Gamma_9$ do $\Gamma_1$, $\Gamma_2$ i $\Gamma_8$.}
    \ita{$\Gamma_9$ a $\Gamma_1$, $\Gamma_2$ e $\Gamma_8$.}
    \pol{Wtedy przy dobrym wyborze, który okrąg styczny wziąć, okrąg $\Gamma_4$ jest też styczny do $\Gamma_2$, $\Gamma_3$ i $\Gamma_9$.}
    \ita{Allora, scegliendo opportunamente quale cerchio tangente prendere, il cerchio $\Gamma_4$ è anche tangente a $\Gamma_2$, $\Gamma_3$ e $\Gamma_9$.}
\end{proposition}
