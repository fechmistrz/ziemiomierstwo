\pol{\section{Przybliżenia $\pi$}}
\ita{\section{Approssimazioni di $\pi$}}
\label{sekcja_przyblizenia_pi}%

% Eves s. 7
\pol{W papirusie Rhinda pojawi się wskazówka, że pole koła to w przybliżeniu pole kwadratu o boku tak długim jak $8/9$ średnica koła, co jest równoważne przybliżeniu $\pi \approx (4/3)^4 \approx 3.16049$.}
\ita{Nel papiro di Rhind compare l'indicazione che l'area del cerchio è approssimativamente uguale all’area di un quadrato con lato pari a $8/9$ del diametro del cerchio, il che è equivalente all’approssimazione $\pi \approx (4/3)^4 \approx 3{,}16049$.}
\pol{Następne przybliżenie (dokładniejsze za cenę większego mianownika) to dopiero $31/35$, więć Egipcjanie będą mieć nosa.}
\ita{L'approssimazione successiva (più accurata, ma al prezzo di un denominatore maggiore) è solo $31/35$, quindi gli Egizi avevano davvero buon intuito.}

\pol{Śulbasūtras, starożytny tekst hinduistyczny, przybliży $\pi$ jako $36 / (6 + 4 \sqrt 2) \approx 3.08831$.}
\ita{Gli Śulbasūtra, un antico testo hindu, approssimano $\pi$ come $36 / (6 + 4 \sqrt{2}) \approx 3{,}08831$.}

\pol{W 260 roku przed naszą erą Archimedes znajdzie przybliżenie}
\ita{Nel 260 a.C. Archimede troverà un'approssimazione}
\begin{equation}
    3 + \frac{10}{71} = 3.14084 \ldots < \pi < 3 + \frac 1 7 = 3.14285 \ldots
\end{equation}

\pol{Mikołaj z Kuzy (znany też jako Kuzańczyk) w wielu pismach zajmie się problemem kwadratury koła i obliczania obwodu koła.}
\ita{Niccolò Cusano in molti scritti si occuperà del problema della quadratura del cerchio e del calcolo della circonferenza.}
\index[persons]{Mikołaj z Kuzy}%
\pol{Jego przybliżenie $\pi \approx 6\sqrt{48/7}/5 = 3.1423...$ (z \emph{De Geometricis Transmutationibus} z 1445 roku) będzie zgodne z~oszacowaniem $223/71 < \pi < 22/7$ pochodzącym jeszcze od Archimedesa.}
\ita{La sua approssimazione $\pi \approx 6\sqrt{48/7}/5 = 3{,}1423\ldots$ (da \emph{De Geometricis Transmutationibus} del 1445) sarà conforme alla stima $223/71 < \pi < 22/7$, risalente ancora ad Archimede.}
\pol{Niestety, późniejsze pomysły będą mniej trafione;  Regiomontanus omówi po kolei wszystkie konstrukcje Mikołaja i~przejdzie do liczb (,,\emph{nunc ad numeros descendendum}'').}
\ita{Purtroppo, le idee successive saranno meno riuscite; Regiomontano esaminerà una per una tutte le costruzioni di Niccolò e passerà ai numeri (``\emph{nunc ad numeros descendendum}'').}
\pol{Policzy na przykład, że Mikołaj w \emph{Quadratura Circuli} z 1450 roku sugeruje $\pi = 3.154\ldots > 22/7$.}
\ita{Calcolerà, per esempio, che Niccolò nel \emph{Quadratura Circuli} del 1450 suggerisce $\pi = 3{,}154\ldots > 22/7$.}

\pol{Joseph Ignatius Carl von Leistner, rycerz służący Franciszkowi I, księciu Lotaryngii, znajdzie ,,wzór''}
\ita{Joseph Ignatius Carl von Leistner, cavaliere al servizio di Francesco I, duca di Lorena, troverà una ,,formula''}
\begin{equation}
    \pi = \frac{3844}{1225} = 3.13795\,91836\ldots,
\end{equation}
\pol{więcej o wzorze napisze Hartshorne \cite[s. 224]{hartshorne2000}.}
\ita{di cui scriverà più diffusamente Hartshorne \cite[p. 224]{hartshorne2000}.}

\pol{Pełni zapału ludzie będą szukać kolejnych cyfr liczby $\pi$ na długo przed wynalezieniem kalkulatorów; w roku 1600 będą znać 15 cyfr po przecinku, w 1700 --- 71 cyfr, natomiast w 1873 ponad 500 cyfr.}
\ita{Persone piene di entusiasmo cercheranno ulteriori cifre del numero $\pi$ molto prima dell’invenzione delle calcolatrici: nel 1600 si conosceranno 15 cifre decimali, nel 1700 --- 71 cifre, mentre nel 1873 oltre 500 cifre.}
\pol{Wreszcie w 1883 roku Lindemann pokaże, że $\pi$ jest liczbą przestępną.}
\ita{Infine, nel 1883 Lindemann dimostrerà che $\pi$ è un numero trascendente.}
